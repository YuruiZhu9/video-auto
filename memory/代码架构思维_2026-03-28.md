```python
import threading

class ModelConfig:
    _instance = None
    _lock = threading.Lock()
    
    def __new__(cls):
        if cls._instance is None:

                if cls._instance is None:

                    cls._instance._initialized = False
        return cls._instance
```

#### 🔹 工厂模式 (Factory)
**场景：** 不同模型类型（CF/DeepFM/DIN）的统一创建入口

```python
from abc import ABC, abstractmethod

class BaseRecommender(ABC):
    @abstractmethod
    def predict(self, user_id, item_ids):
        pass

class CFRecommender(BaseRecommender):
    def predict(self, user_id, item_ids):
        # 协同过滤逻辑
        return [0.5, 0.3, 0.8]

class DINRecommender(BaseRecommender):
    def predict(self, user_id, item_ids):
        # DIN注意力模型逻辑
        return [0.6, 0.4, 0.9]

class RecommenderFactory:
    @staticmethod
    def create(model_type: str) -> BaseRecommender:
        factories = {
            "cf": CFRecommender,
            "din": DINRecommender,
        }
        if model_type not in factories:
            raise ValueError(f"Unknown model type: {model_type}")
        return factories[model_type]()

# 使用：新增模型只需添加一个类，不用改调用方
rec = RecommenderFactory.create("din")
scores = rec.predict(user_id=123, item_ids=[1, 2, 3])
```

#### 🔹 策略模式 (Strategy)
**场景：** 多种排序策略（A/B测试、EE探索、多目标融合）的灵活切换

```python
from abc import ABC, abstractmethod

class RankingStrategy(ABC):
    @abstractmethod
    def rank(self, candidates: list, context: dict) -> list:
        pass

class CTROnlyStrategy(RankingStrategy):
    def rank(self, candidates, context):
        return sorted(candidates, key=lambda x: x["ctr"], reverse=True)

class DiversityStrategy(RankingStrategy):
    def rank(self, candidates, context):
        # MMR (Maximal Marginal Relevance) 策略
        return self._mmr_rank(candidates, context)
    
    def _mmr_rank(self, candidates, context):
        result = []
        seen_categories = set()
        remaining = candidates.copy()
        while remaining:
            best_item = max(remaining, 
                key=lambda x: x["ctr"] * (1.0 if x.get("category") not in seen_categories else 0.3))
            result.append(best_item)
            seen_categories.add(best_item.get("category"))
            remaining.remove(best_item)
        return result

class Ranker:
    def __init__(self, strategy: RankingStrategy):
        self._strategy = strategy
    
    def set_strategy(self, strategy: RankingStrategy):
        self._strategy = strategy
    
    def recommend(self, candidates, context):
        return self._strategy.rank(candidates, context)

# 使用：运行时切换策略，无需修改 Ranker 类
ranker = Ranker(CTROnlyStrategy())
results = ranker.recommend(candidates, context)
ranker.set_strategy(DiversityStrategy())  # 运行时切换
```

#### 🔹 装饰器模式 (Decorator)
**场景：** 给推荐函数添加缓存、监控、重试等横切关注点

```python
import time
import functools

def log_execution(func):
    """日志装饰器：记录函数执行时间和参数"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"[LOG] {func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

def cache(ttl=300):
    """缓存装饰器：避免重复计算"""
    def decorator(func):
        cache_store = {}
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            key = str(args) + str(kwargs)
            if key in cache_store:
                cached_time, cached_val = cache_store[key]
                if time.time() - cached_time < ttl:
                    return cached_val
            result = func(*args, **kwargs)
            cache_store[key] = (time.time(), result)
            return result
        return wrapper
    return decorator

@log_execution
@cache(ttl=60)
def get_user_embedding(user_id: int) -> list:
    """获取用户向量：自动日志+缓存"""
    return [0.1] * 128
```

```python
class TrainingObserver:
    def on_train_complete(self, metrics: dict): pass
    def on_train_fail(self, error: Exception): pass

class SlackNotifier(TrainingObserver):
    def on_train_complete(self, metrics):
        print(f"[Slack] AUC={metrics.get('auc')}")
    def on_train_fail(self, error):
        print(f"[Slack] Failed: {error}")

class ModelPusher(TrainingObserver):
    def on_train_complete(self, metrics):
        if metrics.get("auc", 0) > 0.75:
            print("[Pusher] Model passed, pushing online...")

class TrainingManager:
    def __init__(self):
        self._observers: list[TrainingObserver] = []
    
    def subscribe(self, obs: TrainingObserver):
        self._observers.append(obs)
    
    def run_training(self):
        try:
            metrics = {"auc": 0.78}
            for obs in self._observers:
                obs.on_train_complete(metrics)
        except Exception as e:
            for obs in self._observers:
                obs.on_train_fail(e)
```

#### 🔹 迭代器模式 (Iterator)
**场景：** 遍历大规模用户行为日志、批处理训练数据

```python
class UserBehaviorIterator:
    """大数据量场景下的内存友好迭代器"""
    def __init__(self, user_ids: list, batch_size: int = 1000):
        self.user_ids = user_ids
        self.batch_size = batch_size
        self.index = 0
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if self.index >= len(self.user_ids):
            raise StopIteration
        batch = self.user_ids[self.index: self.index + self.batch_size]
        self.index += self.batch_size
        return batch

# 使用：分批处理百万用户，不用一次性加载
for user_batch in UserBehaviorIterator(all_user_ids, batch_size=5000):
    features = extract_features_batch(user_batch)

```

#### 🔹 模板方法模式 (Template Method)
**场景：** 定义推荐流程骨架，具体步骤由子类实现

```python
class BaseRecommenderPipeline:
    """推荐流程模板：数据→特征→召回→排序→后处理"""
    
    def run(self, user_id: int) -> list:
        raw_data = self.load_data(user_id)
        features = self.build_features(raw_data)
        candidates = self.recall_candidates(user_id, raw_data)
        scores = self.rank(candidates, features)
        return self.post_process(scores)
    
    # 抽象方法（子类必须实现）
    def load_data(self, user_id):
        raise NotImplementedError
    
    def build_features(self, raw_data):
        raise NotImplementedError
    
    # 钩子方法（子类可选重写）
    def post_process(self, results):
        return results[:10]

class CFRecommenderPipeline(BaseRecommenderPipeline):
    def load_data(self, user_id):
        return {"interactions": [(1,2), (2,3)]}
    
    def build_features(self, raw_data):
        return {"user_vector": [0.2]*64}
    
    def recall_candidates(self, user_id, raw_data):
        return [{"item_id": 1, "score": 0.9}]
    
    def rank(self, candidates, features):
        return sorted(candidates, key=lambda x: x["score"], reverse=True)
```

**场景：** 构建复杂配置对象（模型超参数、特征配置）

```python
class RecommenderConfig:
    def __init__(self):
        self.model_type = None
        self.embedding_dim = None
        self.learning_rate = None
        self.batch_size = None
        self.layers = []
        self.dropout = None

class RecommenderConfigBuilder:
    def __init__(self):
        self._config = RecommenderConfig()
    
    def model_type(self, t: str):
        self._config.model_type = t
        return self  # 链式调用
    
    def embedding_dim(self, dim: int):
        self._config.embedding_dim = dim
        return self
    
    def learning_rate(self, lr: float):
        self._config.learning_rate = lr
        return self
    
    def add_layer(self, units: int):
        self._config.layers.append(units)
        return self
    
    def build(self):
        if not self._config.model_type:
            raise ValueError("model_type is required")
        return self._config

# 使用：链式构建复杂配置
config = (RecommenderConfigBuilder()
          .model_type("din")
          .embedding_dim(64)
          .learning_rate(0.001)
          .add_layer(256).add_layer(128).add_layer(64)
```

## 二、架构风格：从单体到分布式

### 2.1 主要架构风格对比

```
单体架构 → 模块化架构 → 微服务架构 → 服务网格
   ↓          ↓            ↓            ↓
  简单      按需拆分      解耦独立      治理分离
```

| 架构风格 | 特点 | 适用场景 | 推荐系统中的例子 |
|---------|------|---------|----------------|
| **单体架构** | 所有模块在一个进程 | 早期/小规模/POC | 快速验证推荐算法demo |
| **微服务架构** | 服务独立部署、REST通信 | 中大型系统 | 召回→粗排→精排→重排 拆分 |
| **事件驱动架构(EDA)** | 异步消息总线、松耦合 | 实时性要求高 | 用户行为→Kafka→实时特征→推荐 |
| **CQRS** | 读写分离 | 读写压力差异大 | 写：日志写入；读：预计算推荐结果 |
| **Lambda架构** | 批处理+流处理双通道 | 兼顾精度和实时性 | 批模型全量训练+流模型实时更新 |
| **Kappa架构** | 纯流处理 | 资源受限 | 纯Kafka+Flink流处理 |

### 2.2 微服务架构详解

#### 为什么推荐系统天然适合微服务？

```

│           推荐系统微服务拆分示例                      │
├─────────────────────────────────────────────────────┤
│                                                     │
│  用户服务   物品服务   特征服务   召回服务   排序服务 │
```

**关键理解：**
- **每个服务职责单一**：召回服务只负责候选生成，不关心排序
- **独立扩缩容**：大促时排序服务扩容3倍，其他服务不动
- **技术栈灵活**：Python写算法服务，Go写高并发网关

### 2.3 事件驱动架构 (EDA)

事件驱动是推荐系统实时化的核心架构。

```
┌──────────┐   行为事件    ┌─────────────┐   写入事件   ┌──────────────┐
│  前端APP  │ ──────────→ │ Kafka队列    │ ─────────→ │ 特征服务      │
└──────────┘              └─────────────┘             └──────────────┘
                              │                              │
                              ↓                              ↓
                         ┌─────────────┐             ┌──────────────┐
                         │  Flink流处理  │             │   Redis      │
                         │  (窗口聚合/   │             │ (用户短期特征) │
                         │   反作弊/实时)│             └──────────────┘
                         └─────────────┘                    │
                              │                              │ 读取
                              ↓                              ↓
                         ┌─────────────┐             ┌──────────────┐
                         │ Hologres OLAP│             │ 推荐推理服务  │
                         └─────────────┘             └──────────────┘
```

**核心组件：**
- **Kafka**：分布式消息队列，事件持久化、支持回放
- **Flink**：流处理引擎，支持事件时间、窗口聚合、状态管理
- **Redis**：高性能KV存储，缓存实时特征

### 2.4 CQRS（命令查询职责分离）

CQRS在推荐系统中天然适用：**写请求（用户行为）和读请求（推荐结果）模式完全不同。**

```
┌──────────────────────────────────────────────────────────┐
│                        CQRS 架构                         │
├─────────────────────────┬────────────────────────────────┤
│       命令端(Write)      │           查询端(Read)           │
│  用户点击/曝光/购买事件    │    推荐结果查询                  │
│       │                 │         │                      │
│       ↓                 │         ↓                      │
│  写入行为日志             │    预计算好的推荐列表             │
│       ↓                 │    (Redis/ES缓存)              │
│  更新特征存储              │                                │
│  (Kafka→Flink→HBase)    │    毫秒级响应(<50ms)            │
└─────────────────────────┴────────────────────────────────┘
```

**为什么推荐系统适合CQRS：**
- 写端：海量小数据（每次点击事件，几KB）
- 读端：少量大数据（推荐列表，几MB一次性返回）
- 写优化：追加写日志，分批聚合计算
- 读优化：预计算好推荐结果，缓存命中

### 2.5 Lambda vs Kappa 架构

| 维度 | Lambda架构 | Kappa架构 |
|-----|----------|---------|
| 核心思想 | 批处理层+速度层双通道 | 纯流处理，统一入口 |
| 复杂度 | 较高（两套代码） | 较低（统一） |
| 推荐系统应用 | 批模型（长期兴趣）+流模型（短期兴趣） | Flink全量流处理 |

```python
# Lambda架构示例：推荐系统中的双通道融合
class HybridRecommender:
    def __init__(self):
        # 批通道：全量训练的协同过滤模型（每天更新）
        self.batch_model = load_batch_cf_model("s3://models/cf_daily")
        # 流通道：Flink实时计算的 item CF 分数
        self.stream_scores = Redis("realtime:item_cf_scores")
    
    def recommend(self, user_id: int, k: int = 10) -> list:
        # 融合：0.7 * 批通道分 + 0.3 * 流通道分
        batch_scores = self.batch_model.predict(user_id)
        stream_scores = self.stream_scores.get(user_id, {})
        
        fused_scores = {}
        for item_id, batch_score in batch_scores.items():
            stream_score = stream_scores.get(item_id, 0.0)
            fused_scores[item_id] = 0.7 * batch_score + 0.3 * stream_score
        
        return sorted(fused_scores.items(), key=lambda x: x[1], reverse=True)[:k]
```

## 三、代码重构技巧：让烂代码起死回生

### 3.1 坏代码的典型症状（算法工程师版）

```
❌ 症状1: 一个.py文件写了3000行
         （特征工程+模型训练+评估全在一起）
❌ 症状2: 硬编码满天飞
         （batch_size=1024, embedding_dim=128 散落各处）
❌ 症状3: 函数副作用太多
         （训练函数偷偷改了全局配置）
❌ 症状4: 重复代码
         （三个实验脚本复制粘贴了80%相同代码）
❌ 症状5: 注释和代码不一致
         （注释说做归一化，实际做了标准化）
❌ 症状6: 全局可变状态
         （多线程训练时结果不一致）
```

### 3.2 SOLID原则：重构的理论基础

#### S — 单一职责原则（Single Responsibility Principle）
**一个类/函数只做一件事。**

```python
# ❌ 违反SRP：一个函数干三件事
def train_and_evaluate(model_type, data_path, output_path):
    data = load_csv(data_path)
    model = train_model(model_type, data)
    metrics = evaluate(model, data)
    save_results(metrics, output_path)

# ✅ 符合SRP：每个函数职责单一
def load_data(path: str) -> pd.DataFrame:
    return pd.read_csv(path)

def train_model(model_type: str, data: pd.DataFrame):
    if model_type == "din":
        return DINModel().fit(data)
    return WideDeepModel().fit(data)

def evaluate(model, data: pd.DataFrame) -> dict:
    predictions = model.predict(data)
    return {"auc": roc_auc_score(data["label"], predictions)}
```

#### O — 开闭原则（Open-Closed Principle）
**对扩展开放，对修改封闭。**

```python
# ❌ 违反OCP：新增损失函数要改主逻辑
def compute_loss(model_type, predictions, labels):
    if model_type == "bce":
        return binary_cross_entropy(predictions, labels)
    elif model_type == "focal":
        return focal_loss(predictions, labels)

# ✅ 符合OCP：新增损失函数只需添加新类，不改主逻辑
class LossFunction(ABC):
    @abstractmethod
    def __call__(self, predictions, labels) -> torch.Tensor:
        pass

class BCELoss(LossFunction):
    def __call__(self, predictions, labels):
        return F.binary_cross_entropy(predictions, labels)

class FocalLoss(LossFunction):
    def __call__(self, predictions, labels, gamma=2.0):
        pt = torch.where(labels == 1, predictions, 1 - predictions)
        return -((1 - pt) ** gamma * torch.log(pt)).mean()
```

#### L — 里氏替换原则（Liskov Substitution Principle）
**子类必须能替换父类而不破坏程序行为。**

```python
# ❌ 违反LSP：子类改变了父类的行为契约
class BaseTrainer:
    def train(self, data):
        for epoch in range(100):  # 默认训练100个epoch
            self._train_epoch(data)
        return self.model

class QuickTrainer(BaseTrainer):
    def train(self, data):
        self._train_epoch(data)  # 改变行为：只训练1个epoch！
        return self.model

# ✅ 符合LSP：子类增强但不改变行为契约
class BaseTrainer:
    def __init__(self, model, config):
        self.model = model
        self.epochs = config.get("epochs", 100)
    
    def train(self, data):
        for epoch in range(self.epochs):
            self._train_epoch(data)
        return self.model

class EarlyStoppingTrainer(BaseTrainer):
    """子类增强：添加早停，但train()行为不变"""
    def __init__(self, model, config):
        super().__init__(model, config)
        self.patience = config.get("patience", 10)
        self.best_auc = 0
        self.wait = 0
    
    def _train_epoch(self, data):
        self.model.fit_epoch(data)
        val_auc = self.model.evaluate(data["val"])
        if val_auc > self.best_auc:
            self.best_auc = val_auc
            self.wait = 0
        else:
            self.wait += 1
            if self.wait >= self.patience:
                raise StopIteration("Early stopping")
```

## 四、架构决策记录（ADR）：让决策有据可查

### 4.1 什么是 ADR？

ADR（Architecture Decision Records，架构决策记录）是一种**记录架构决策的轻量级文档实践**，由 Michael Nygard 在《Documenting Architecture Decisions》中提出。

**为什么算法工程师需要了解 ADR？**
- 团队协同时，技术选型决策需要透明化
- 面试中能展示"我在项目中做了哪些架构决策，为什么"
- 避免重复踩坑，新人也能理解历史决策

### 4.2 ADR 模板

```markdown
# ADR-001: 推荐系统引入实时特征服务

### 状态
已接受 (Accepted)

### 背景
- 问题：当前推荐系统完全依赖离线批处理特征，用户行为延迟1小时才能反映到推荐中
- 影响：大促期间用户刚点击的商品，1小时后仍在推荐中
- 需求：引入实时特征，将行为反馈延迟降至1分钟以内

### 决策
- 引入 Kafka + Flink 流处理架构
- 实时特征存储使用 Redis
- 特征服务独立部署，提供 gRPC 接口

### 选项对比

| 选项 | 优点 | 缺点 |
|-----|------|------|
| A. 纯离线（现状）| 简单，无额外基础设施 | 延迟高 |
| B. Lambda架构 | 兼顾实时和全量 | 两套代码，复杂 |
| **C. Kappa架构** | **代码统一，Flink全量处理** | **资源消耗较高** |
| D. 完全实时 | 最低延迟 | 无历史数据回溯能力 |

**选择C的原因：** 团队已有Flink使用经验，Kappa架构可复用现有组件。

### 后果
- **正面：** 实时特征延迟从1小时降至30秒；大促期间用户体验提升
- **负面：** 新增Kafka+Flink运维成本；需要处理流处理特有的乱序/迟到事件
- **待办：** 补充流处理监控告警；编写Flink作业SOP

### 日期
2026-01-15
```

### 4.3 ADR 的使用场景

```
场景1: 技术选型
→ "我们选择用Redis而不是Memcached存实时特征，原因记录在ADR-003"

场景2: 架构变更
→ "从单体拆分为微服务，决策过程记录在ADR-010至ADR-015"

场景3: 新人 onboarding
→ "先读ADR-001到ADR-009，了解系统架构演进历史"

场景4: 回滚决策
→ "ADR-007的方案失败了，需要重新讨论，记录为ADR-007-Revised"
```

### 4.4 ADR 生命周期

```
提议 (Proposed) → 已接受 (Accepted) / 已拒绝 (Rejected)
     ↓
已接受 → 已废弃 (Deprecated)  ←  被新ADR替代时
     ↓
记录演进 (Superseded)
```

## 五、技术债务管理：不欠下未来的债

### 5.1 什么是技术债务？

技术债务（Technical Debt）最早由 Ward Cunningham 提出，比喻软件开发中的**短期便利换取长期成本**。就像金融债务一样，技术债务也会产生"利息"——每次在烂代码上添加新功能，都要额外付出维护成本。

```
技术债务 = 刻意债务（明知故犯） + 意外债务（无知不觉）

债务利息 = 在低质量代码上开发新功能的速度损耗
```

### 5.2 算法工程师常见的 6 类技术债务

| 债务类型 | 具体表现 | 影响 |
|---------|---------|------|
| **模型债务** | 模型无版本管理，硬编码在代码中；超参数无记录 | 无法复现实验，新人接手困难 |
| **数据债务** | 特征无文档，来源不清晰，口径不一致 | 特征雪崩，模型效果无法保证 |
| **代码债务** | 训练和推理代码风格割裂，无单测 | 线上线下不一致，Bug频发 |
| **依赖债务** | 锁死旧版本框架不敢升级 | 安全漏洞，无法使用新特性 |
| **运维债务** | 模型上线手动操作，无CI/CD | 发布风险高，回滚困难 |
| **文档债务** | 代码无注释，架构设计无文档 | 知识传承断裂 |

### 5.3 技术债务管理四象限

```
                │
    紧急程度     │    ① 立即偿还          ② 计划偿还
      高        │   （热点代码区）       （定期技术改进）
                │    高影响+高频变更      高影响+低频变更
                │    优先重构             排入迭代计划
                │
                ├───────────────────────────────
                │
      紧急程度   │    ③ 暂不处理          ④ 立即修复
       低       │   （低价值代码）        （隐藏炸弹）
                │    低影响+高频变更      低影响+低频变更
                │    接受债务             小步快跑消除
                │
                └────────────────────────────────────
                              业务价值
```

### 6.2 无状态服务设计：水平扩展的基础

**核心原则：服务本身不存储状态，状态外置到 Redis/DB。**

```python
# ❌ 有状态服务：无法水平扩展
class RecommendationService:
    def __init__(self):
        self.user_cache = {}  # 内存缓存，无法跨实例共享
        self.model = load_model()  # 每个实例各自加载一份
    
    def recommend(self, user_id):
        if user_id not in self.user_cache:
            self.user_cache[user_id] = self.compute(user_id)
        return self.user_cache[user_id]

# ✅ 无状态服务：可以无限水平扩展
class RecommendationService:
    def __init__(self, redis_client, model_path):
        self.redis = redis_client
        self.model = load_model()  # 模型仍然加载到内存（通常GPU显存）
        # 但用户级别状态全部外置
    
    def recommend(self, user_id):
        # 从Redis获取缓存（所有实例共享同一个Redis集群）
        cache_key = f"rec:result:{user_id}"
        cached = self.redis.get(cache_key)
        if cached:
            return json.loads(cached)
        
        # 实时计算
        result = self._compute_recommendation(user_id)
        # 写入Redis（所有实例共享）
        self.redis.setex(cache_key, 300, json.dumps(result))
        return result
```

### 6.3 数据分片与负载均衡

#### 数据分片（Sharding）：解决单库存储瓶颈

```python
# 简单哈希分片示例：用户数据按user_id分片
class UserDataSharding:
    def __init__(self, num_shards: int, shards: list):
        self.num_shards = num_shards
        self.shards = shards  # 每个分片对应的存储节点
    
    def get_shard(self, user_id: int) -> int:
        return user_id % self.num_shards
    
    def get_connection(self, user_id: int):
        shard_id = self.get_shard(user_id)
        return self.shards[shard_id]
    
    def save_user_features(self, user_id: int, features: dict):
        shard = self.get_connection(user_id)
        shard.set(f"user:{user_id}", json.dumps(features))
    
    def get_user_features(self, user_id: int) -> dict:
        shard = self.get_connection(user_id)
        return json.loads(shard.get(f"user:{user_id}"))

# 推荐系统实际分片策略
# - 用户特征：按 user_id % N 哈希分片
# - 物品特征：按 item_id % N 哈希分片
# - 推荐结果：按 user_id 路由到同一个分片（保证同一用户的推荐结果一致性）
```

#### 负载均衡：让请求均匀分布

```
                    ┌─────────────┐
             请求1  │ 排序服务-1  │  ← 实例1
    ┌──────────┐    ├─────────────┤
    │  用户请求  │──→│  负载均衡器   │  请求2  │ 排序服务-2  │  ← 实例2
    └──────────┘    ├─────────────┤
                    │ 排序服务-3  │  ← 实例3（暂时不活跃）
                    └─────────────┘

负载均衡策略：
- Round Robin（轮询）：简单但不感知实例负载
- Least Connections（最少连接）：动态感知
- 加权轮询：给高性能实例分配更多请求
- 一致性哈希：同类用户路由到同类实例（缓存友好）
```

### 6.4 推荐系统的扩展性设计要点

#### 核心扩展策略

| 扩展维度 | 问题 | 解决方案 |
|---------|------|---------|
| **特征存储扩展** | 用户/物品特征量太大 | Redis Cluster分片，特征分层（热/温/冷） |
| **推理吞吐扩展** | QPS太高，模型推理慢 | 模型量化+多实例部署+GPU共享 |
| **召回规模扩展** | 候选集太大 | 多级召回：粗召回(万级)→精召回(千级)→精排(百级) |
| **数据处理扩展** | 训练数据太大 | 分布式训练（Parameter Server/AllReduce） |
| **实时性扩展** | Kafka数据积压 | Flink并行度调优+背压机制 |

#### 多级漏斗：控制计算量

```
┌──────────────────────────────────────────────────────────────┐
│                   推荐系统多级漏斗                             │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│  全量商品  →  多路召回   →  粗排(T-1)  →  精排(T-2)  → 重排   │
│  (1000万)    (2000个)     (500个)       (100个)     (Top10)  │
│                                                              │
│  召回阶段：多种策略并行（CF/Embedding/热榜/地理）              │
│  ↓                                                          │
│  粗排阶段：轻量级模型（如LightGBM），快速过滤到500             │
│  ↓                                                          │
│  精排阶段：复杂深度模型（如DIN），精打细算Top100               │
│  ↓                                                          │
│  重排阶段：业务规则（去重、频控、多样性、商品组合）             │
│                                                              │
│  为什么这样设计？                                             │
```

```python
async def recommend(req: RecommendRequest):
    """
    推荐主流程：用户理解 → 多路召回 → 精排 → 重排 → 返回
    """
    import time
    start = time.time()
    
    # 1. 用户理解：获取用户实时+离线特征
    user_features = await clients.feature_store.get_user_features(req.user_id)
    
    # 2. 多路召回：并行调用多个召回通道
    recall_tasks = [
        recall(user_id=req.user_id, user_features=user_features, channel=ch)
        for ch in ["cf", "swing", "hot", "tag_match"]
    ]
    recall_results = await asyncio.gather(*recall_tasks)
    all_candidates = merge_recall_results(recall_results)
    
    # 3. 精排：调用排序服务打分
    ranked_items = await clients.ranker_service.rank(
        user_id=req.user_id,
        candidates=all_candidates,
        features=user_features,
        top_k=100
    )
    
    # 4. 重排：多样性、业务规则
    final_items = diversity_rerank(ranked_items, user_features)
    final_items = apply_business_rules(final_items)
    
    # 5. 分页返回
    start_idx = (req.page_num - 1) * req.page_size
    result_items = final_items[start_idx: start_idx + req.page_size]
    
    latency = (time.time() - start) * 1000
    return RecommendResponse(
        user_id=req.user_id,
        items=result_items,
        request_id=f"req_{req.user_id}_{int(time.time())}",
```

## 八、面试高频问题汇总

### Q1: 你在项目中用过哪些设计模式？解决了什么问题？

**参考回答框架：**
> "我在推荐系统的重构中用到了策略模式。原来排序策略硬编码在代码中，每次想尝试新的排序方法都要改核心逻辑。后来我抽象出一个 RankingStrategy 接口，把 CTR 排序、多样性排序、多目标排序分别实现，运行时通过配置切换，代码可维护性大幅提升。"

### Q2: 你们的推荐系统是怎么做扩展的？

**参考回答框架：**
> "我们做了几个层面的扩展：① 服务无状态化，把用户状态外置到 Redis，这样排序服务可以水平扩容；② 多级漏斗，全量商品→召回(2000)→粗排(500)→精排(100)→重排(10)，控制计算成本；③ 读写分离，读特征走 Redis 缓存，写特征异步更新。"

### Q3: 你们怎么处理技术债务的？

**参考回答框架：**
> "我们用四象限管理技术债务，区分紧急程度和业务价值。高影响+高频变更的债务优先偿还。同时每个迭代抽出 10% 时间专门做技术债务清理，所有决策记录在 ADR 中。"

### Q4: 什么情况下会选择微服务，什么情况选择单体？

**参考回答框架：**
> "推荐系统达到日活 100 万以上时，单体架构的部署和扩展会成为瓶颈，这时适合拆微服务。但对于早期 POC 或小团队，优先选择模块化单体，先跑通业务，再逐步拆分。不建议为了微服务而微服务。"

### Q5: CQRS 和传统读写分离有什么区别？

**参考回答框架：**
> "传统读写分离是同一个数据源的读写分离；CQRS 则是读写使用完全不同的数据模型。推荐系统中，写端是用户行为日志，读端是预计算好的推荐列表——这两者的数据模型和使用场景完全不同，CQRS 更适合这种场景。"

### 数据层（Repository）

```python
# repositories/database.py
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, Session
from contextlib import contextmanager

engine = create_engine(settings.database_url, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

@contextmanager
def get_db() -> Session:
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# repositories/user_repository.py
from sqlalchemy.orm import Session
from models.schemas import UserCreate, UserUpdate

class UserRepository:
    def __init__(self, db: Session):
        self.db = db
    
    def get_by_id(self, user_id: int):
        return self.db.query(User).filter(User.id == user_id).first()
    
    def get_by_email(self, email: str):
        return self.db.query(User).filter(User.email == email).first()
    
    def list(self, skip: int = 0, limit: int = 100):
        return self.db.query(User).offset(skip).limit(limit).all()
    
    def create(self, user_data: UserCreate) -> User:
        user = User(**user_data.model_dump())
        self.db.add(user)
        self.db.commit()
        self.db.refresh(user)
        return user
    
    def update(self, user: User, user_data: UserUpdate) -> User:
        for key, value in user_data.model_dump(exclude_unset=True).items():
            setattr(user, key, value)
        self.db.commit()
        self.db.refresh(user)
        return user
    
    def delete(self, user: User):
        self.db.delete(user)
        self.db.commit()
```

### 业务层（Service）

```python
# services/user_service.py
from sqlalchemy.orm import Session
from models.schemas import UserCreate, UserUpdate, UserResponse
from repositories.user_repository import UserRepository
from core.security import hash_password, verify_password

class UserService:
    def __init__(self, repo: UserRepository):
        self.repo = repo
    
    def get_user(self, user_id: int) -> User:
        user = self.repo.get_by_id(user_id)
        if not user:
            raise ValueError("用户不存在")
        return user
    
    def get_user_by_email(self, email: str) -> User:
        return self.repo.get_by_email(email)
    
    def list_users(self, skip: int = 0, limit: int = 100):
        return self.repo.list(skip, limit)
    
    def create_user(self, user_data: UserCreate) -> User:
        # 业务逻辑：检查邮箱唯一性
        if self.repo.get_by_email(user_data.email):
            raise ValueError("邮箱已被注册")
        
        # 业务逻辑：密码加密
        user_data.password = hash_password(user_data.password)
        
        return self.repo.create(user_data)
    
    def update_user(self, user_id: int, user_data: UserUpdate) -> User:
        user = self.get_user(user_id)
        
        # 业务逻辑：如果改邮箱，检查唯一性
        if user_data.email and user_data.email != user.email:
            if self.repo.get_by_email(user_data.email):
                raise ValueError("邮箱已被注册")
        
        return self.repo.update(user, user_data)
    
    def authenticate(self, email: str, password: str) -> User:
        user = self.get_user_by_email(email)
        if not user or not verify_password(password, user.hashed_password):
            raise ValueError("邮箱或密码错误")
        return user
```

### 表现层（API）

```python
# api/v1/users.py
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from typing import List
from api.deps import get_db, get_current_user
from services.user_service import UserService
from repositories.user_repository import UserRepository
from models.schemas import UserCreate, UserUpdate, UserResponse, Token

router = APIRouter(prefix="/users", tags=["用户"])

def get_user_service(db: Session = Depends(get_db)) -> UserService:
    return UserService(UserRepository(db))

@router.post("/", response_model=UserResponse, status_code=status.HTTP_201_CREATED)
def create_user(
    user_data: UserCreate,
    service: UserService = Depends(get_user_service)
):
    try:
        user = service.create_user(user_data)
        return user
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/{user_id}", response_model=UserResponse)
def get_user(
    user_id: int,
    service: UserService = Depends(get_user_service)
):
    try:
        user = service.get_user(user_id)
        return user
    except ValueError as e:
        raise HTTPException(status_code=404, detail=str(e))

@router.get("/", response_model=List[UserResponse])
def list_users(
    skip: int = 0,
    limit: int = 100,
    service: UserService = Depends(get_user_service)
):
    return service.list_users(skip, limit)

@router.put("/{user_id}", response_model=UserResponse)
def update_user(
    user_id: int,
    user_data: UserUpdate,
    service: UserService = Depends(get_user_service),
    current_user: User = Depends(get_current_user)  # 需要登录
):
    try:
        return service.update_user(user_id, user_data)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.delete("/{user_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_user(
    user_id: int,
    service: UserService = Depends(get_user_service),
    current_user: User = Depends(get_current_user)
):
    try:
        user = service.get_user(user_id)
        service.delete_user(user)
    except ValueError as e:
        raise HTTPException(status_code=404, detail=str(e))
```

## 三层架构优势

```
┌─────────────────────────────────────────────┐
│              API Layer（表现层）              │
│  - 接收 HTTP 请求                           │
│  - 参数验证                                 │
│  - 权限检查                                 │
│  - 返回响应格式                             │
├─────────────────────────────────────────────┤
│            Service Layer（业务层）            │
│  - 业务逻辑                                 │
│  - 数据校验                                 │
│  - 事务处理                                 │
│  - 异常处理                                 │
├─────────────────────────────────────────────┤
│          Repository Layer（数据层）           │
│  - 数据库操作                               │
│  - CRUD                                     │
│  - 查询优化                                 │
└─────────────────────────────────────────────┘

✅ 优点：
1. 职责清晰，每层只做自己的事
2. 易于测试，每层可以单独测试
3. 易于替换，数据库换了你只需要改 Repository
4. 易于扩展，加新功能只需要加 Service

❌ 缺点：
1. 代码量增加
2. 小项目可能过度设计
3. 需要更多文件管理
```

## 踩坑记录

### 坑 1：循环导入
```
问题：api/v1/users.py 导入 services.user_service
     services.user_service.py 导入 api.deps
     api.deps.py 导入 api.v1.users
     → 循环导入！
```

**解决方案**：
- 使用依赖注入，不在 import 时加载
- 把类型提示放到字符串中：`def get_user(service: "UserService")`
- 重构模块边界

### 坑 2：事务泄漏
```python
# ❌ 错误：异常时未回滚事务
def create_user(user_data):
    user = User(**user_data)
    db.add(user)
    send_welcome_email()  # 如果这里抛异常...
    db.commit()  # 用户已添加，但邮件没发
    return user

# ✅ 正确：使用 try-finally 或 context manager
def create_user(user_data):
    try:
        user = User(**user_data)
        db.add(user)
        db.commit()  # 先提交
        send_welcome_email()  # 再发邮件
        return user
    except:
        db.rollback()  # 失败时回滚
        raise
```

### 坑 3：N+1 查询问题
```python
# ❌ 错误：N+1 查询
users = db.query(User).all()
for user in users:
    print(user.orders)  # 每查一个用户，多查一次订单

# ✅ 正确：使用 joinedload
from sqlalchemy.orm import joinedload

users = db.query(User).options(joinedload(User.orders)).all()
for user in users:
    print(user.orders)  # 不再额外查询
```

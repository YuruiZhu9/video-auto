# Python 代码架构案例

## 案例：从简单Flask到分层架构

### 原始代码（无架构）

```python
# app.py - 所有代码混在一起
from flask import Flask, request, jsonify
import sqlite3

app = Flask(__name__)

@app.route('/users', methods=['GET'])
def get_users():
    conn = sqlite3.connect('app.db')
    cursor = conn.cursor()
    cursor.execute("SELECT id, name, email FROM users")
    users = [{'id': row[0], 'name': row[1], 'email': row[2]} for row in cursor.fetchall()]
    conn.close()
    return jsonify(users)

@app.route('/users', methods=['POST'])
def create_user():
    data = request.get_json()
    conn = sqlite3.connect('app.db')
    cursor = conn.cursor()
    cursor.execute("INSERT INTO users (name, email) VALUES (?, ?)", (data['name'], data['email']))
    conn.commit()
    user_id = cursor.lastrowid
    conn.close()
    return jsonify({'id': user_id, 'name': data['name'], 'email': data['email']}), 201

if __name__ == '__main__':
    app.run(debug=True)
```

### 重构后（三层架构）

```python
# config.py
import os

class Config:
    DATABASE = os.getenv('DATABASE', 'app.db')
    DEBUG = os.getenv('DEBUG', 'True') == 'True'

# models/database.py
import sqlite3
from config import Config

class Database:
    @staticmethod
    def get_connection():
        return sqlite3.connect(Config.DATABASE)
    
    @staticmethod
    def execute(query, params=None):
        conn = Database.get_connection()
        cursor = conn.cursor()
        if params:
            cursor.execute(query, params)
        else:
            cursor.execute(query)
        conn.commit()
        return cursor
    
    @staticmethod
    def fetch_all(query, params=None):
        cursor = Database.execute(query, params)
        return cursor.fetchall()
    
    @staticmethod
    def close(conn):
        conn.close()

# models/user.py
class User:
    def __init__(self, id=None, name=None, email=None):
        self.id = id
        self.name = name
        self.email = email
    
    def to_dict(self):
        return {'id': self.id, 'name': self.name, 'email': self.email}
    
    @staticmethod
    def from_row(row):
        return User(id=row[0], name=row[1], email=row[2])

# services/user_service.py
from models.user import User
from models.database import Database

class UserService:
    @staticmethod
    def get_all_users():
        rows = Database.fetch_all("SELECT id, name, email FROM users")
        return [User.from_row(row).to_dict() for row in rows]
    
    @staticmethod
    def create_user(name, email):
        cursor = Database.execute(
            "INSERT INTO users (name, email) VALUES (?, ?)",
            (name, email)
        )
        return {'id': cursor.lastrowid, 'name': name, 'email': email}

# controllers/user_controller.py
from flask import jsonify
from services.user_service import UserService

class UserController:
    @staticmethod
    def get_users():
        return jsonify(UserService.get_all_users()), 200
    
    @staticmethod
    def create_user(request):
        data = request.get_json()
        user = UserService.create_user(data['name'], data['email'])
        return jsonify(user), 201

# app.py
from flask import Flask, request
from controllers.user_controller import UserController

app = Flask(__name__)

@app.route('/users', methods=['GET'])
def get_users():
    return UserController.get_users()

@app.route('/users', methods=['POST'])
def create_user():
    return UserController.create_user(request)

if __name__ == '__main__':
    app.run(debug=True)
```

## 案例：应用SOLID原则

### 依赖倒置原则示例

```python
# 不符合DIP
class MySQLUserRepository:
    def get_users(self):
        # MySQL查询逻辑
        pass

class UserService:
    def __init__(self):
        self.repository = MySQLUserRepository()  # 依赖具体实现
    
    def find_users(self):
        return self.repository.get_users()

# 符合DIP
from abc import ABC, abstractmethod

class UserRepository(ABC):
    @abstractmethod
    def get_users(self):
        pass

class MySQLUserRepository(UserRepository):
    def get_users(self):
        # MySQL查询逻辑
        pass

class PostgreSQLUserRepository(UserRepository):
    def get_users(self):
        # PostgreSQL查询逻辑
        pass

class UserService:
    def __init__(self, repository: UserRepository):  # 依赖抽象
        self.repository = repository
    
    def find_users(self):
        return self.repository.get_users()

# 使用
mysql_repo = MySQLUserRepository()
service = UserService(mysql_repo)
```

## 踩坑记录

### 踩坑1：过早抽象

**问题**：为了"以后可能用到"而创建不必要的接口和抽象

**教训**：
- 遵循YAGNI（You Aren't Gonna Need It）原则
- 等到确实需要抽象时再抽象
- 简单代码优于复杂架构

---

### 踩坑2：分层过度

**问题**：为小项目创建太多层级，导致代码跳转复杂

**教训**：
- 小项目保持简单
- 中大型项目才需要完整分层
- 架构服务于项目，而非相反

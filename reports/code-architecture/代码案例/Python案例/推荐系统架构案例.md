## 版本1：能跑就行（反例）

```python
# bad_recommend.py - 所有代码写在一个文件

from flask import Flask, request, jsonify
import pymysql

app = Flask(__name__)

@app.route('/recommend/<user_id>')
def recommend(user_id):
    # 1. 直接查数据库
    conn = pymysql.connect(host='localhost', user='root', 
                          password='123456', database='recsys')
    cursor = conn.cursor()
    
    # 2. 查用户历史
    cursor.execute("SELECT item_id FROM user_history WHERE user_id=%s", 
                  (user_id,))
    history = [row[0] for row in cursor.fetchall()]
    
    # 3. 查所有商品
    cursor.execute("SELECT id, name, category FROM items")
    all_items = cursor.fetchall()
    
    # 4. 简单推荐逻辑（写死在函数里）
    recommendations = []
    for item in all_items:
        score = 0
        if item[2] in ['electronics', 'phone']:  # 魔法数字
            score += 0.5
        if item[0] not in history:
            score += 0.3
        if len(history) > 10:
            score += 0.2
        recommendations.append({'item': item, 'score': score})
    
    # 5. 排序返回
    recommendations.sort(key=lambda x: x['score'], reverse=True)
    
    cursor.close()
    conn.close()
    
    return jsonify(recommendations[:10])

if __name__ == '__main__':
    app.run(debug=True)
```

### 配置层 config.py

```python
import os

class Config:
    # 数据库配置
    DB_HOST = os.getenv('DB_HOST', 'localhost')
    DB_USER = os.getenv('DB_USER', 'root')
    DB_PASSWORD = os.getenv('DB_PASSWORD', '123456')
    DB_NAME = os.getenv('DB_NAME', 'recsys')
    
    # 推荐配置
    DEFAULT_RECOMMEND_COUNT = 10
    CATEGORY_BOOST_SCORE = 0.5
    NO_HISTORY_BOOST_SCORE = 0.3
    ACTIVE_USER_BOOST_SCORE = 0.2
    ACTIVE_USER_THRESHOLD = 10
```

### 数据层 models/user.py

```python
from dataclasses import dataclass
from typing import List

@dataclass
class User:
    user_id: str
    name: str
    history: List[str] = None
    
    def __post_init__(self):
        if self.history is None:
            self.history = []
    
    @property
    def is_active(self) -> bool:
        return len(self.history) > 10
```

### 数据访问层 repositories/user_repo.py

```python
import pymysql
from typing import List
from models.user import User

class UserRepository:
    def __init__(self, config):
        self.config = config
    
    def _get_connection(self):
        return pymysql.connect(
            host=self.config.DB_HOST,
            user=self.config.DB_USER,
            password=self.config.DB_PASSWORD,
            database=self.config.DB_NAME
        )
    
    def get_user(self, user_id: str) -> User:
        conn = self._get_connection()
        cursor = conn.cursor()
        
        cursor.execute("SELECT user_id, name FROM users WHERE user_id=%s", 
                      (user_id,))
        row = cursor.fetchone()
        
        if not row:
            cursor.close()
            conn.close()
            return None
        
        # 获取用户历史
        cursor.execute(
            "SELECT item_id FROM user_history WHERE user_id=%s", 
            (user_id,)
        )
        history = [row[0] for row in cursor.fetchall()]
        
        cursor.close()
        conn.close()
        
        return User(user_id=row[0], name=row[1], history=history)
```

### 业务逻辑层 services/recommend_service.py

```python
from typing import List
from dataclasses import dataclass

@dataclass
class RecommendationItem:
    item_id: str
    name: str
    category: str
    score: float

class RecommendService:
    def __init__(self, user_repo, item_repo, config):
        self.user_repo = user_repo
        self.item_repo = item_repo
        self.config = config
    
    def recommend(self, user_id: str, count: int = None) -> List[RecommendationItem]:
        if count is None:
            count = self.config.DEFAULT_RECOMMEND_COUNT
        
        # 1. 获取用户
        user = self.user_repo.get_user(user_id)
        if not user:
            return []
        
        # 2. 获取所有商品
        items = self.item_repo.get_all_items()
        
        # 3. 计算推荐分数
        scored_items = []
        for item in items:
            score = self._calculate_score(user, item)
            scored_items.append(RecommendationItem(
                item_id=item['id'],
                name=item['name'],
                category=item['category'],
                score=score
            ))
        
        # 4. 排序返回
        scored_items.sort(key=lambda x: x.score, reverse=True)
        return scored_items[:count]
    
    def _calculate_score(self, user, item):
        """计算推荐分数"""
        score = 0.0
        
        # 分类加权
        if item['category'] in ['electronics', 'phone']:
            score += self.config.CATEGORY_BOOST_SCORE
        
        # 过滤历史（不在历史中的商品）
        if item['id'] not in user.history:
            score += self.config.NO_HISTORY_BOOST_SCORE
        
        # 活跃用户加权
        if len(user.history) > self.config.ACTIVE_USER_THRESHOLD:
            score += self.config.ACTIVE_USER_BOOST_SCORE
        
        return score
```

### API层 controllers/recommend_api.py

```python
from flask import Blueprint, request, jsonify

recommend_bp = Blueprint('recommend', __name__)

@recommend_bp.route('/recommend/<user_id>')
def recommend(user_id):
    count = request.args.get('count', type=int)
    items = recommend_service.recommend(user_id, count)
    
    return jsonify([
        {
            'item_id': item.item_id,
            'name': item.name,
            'category': item.category,
            'score': item.score
        }
        for item in items
    ])
```

### 入口 app.py

```python
from flask import Flask
from config import Config
from repositories.user_repo import UserRepository
from repositories.item_repo import ItemRepository
from services.recommend_service import RecommendService
from controllers.recommend_api import recommend_bp

def create_app():
    app = Flask(__name__)
    
    # 配置
    config = Config()
    
    # 初始化各层
    user_repo = UserRepository(config)
    item_repo = ItemRepository(config)
    recommend_service = RecommendService(user_repo, item_repo, config)
    
    # 注册API
    app.register_blueprint(recommend_bp)
    
    # 注入service（简化处理）
    global recommend_service as module-level
    import sys
    sys.modules[__name__].recommend_service = recommend_service
    
    return app

if __name__ == '__main__':
    app = create_app()
    app.run(debug=True)
```

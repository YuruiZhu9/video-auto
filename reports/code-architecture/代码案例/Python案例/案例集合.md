**3. 数据验证** (`schemas/user_schema.py`)
```python
from dataclasses import dataclass
from typing import Optional

@dataclass
class CreateUserSchema:
    name: str
    email: str
    age: Optional[int] = 0
    
    @classmethod
    def from_dict(cls, data: dict) -> 'CreateUserSchema':
        if not data.get('name'):
            raise ValueError("name is required")
        if not data.get('email'):
            raise ValueError("email is required")
        if '@' not in data.get('email', ''):
            raise ValueError("invalid email format")
        return cls(
            name=data['name'],
            email=data['email'],
            age=data.get('age', 0)
        )
```

**4. 数据访问层** (`repositories/user_repo.py`)
```python
import sqlite3
from typing import List, Optional
from models.user import User

class UserRepository:
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    def _get_connection(self):
        return sqlite3.connect(self.db_path)
    
    def find_all(self) -> List[User]:
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute('SELECT id, name, email, age FROM users')
        rows = cursor.fetchall()
        conn.close()
        return [User(id=row[0], name=row[1], email=row[2], age=row[3]) 
                for row in rows]
    
    def find_by_id(self, user_id: int) -> Optional[User]:
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute('SELECT id, name, email, age FROM users WHERE id = ?', 
                      (user_id,))
        row = cursor.fetchone()
        conn.close()
        if row:
            return User(id=row[0], name=row[1], email=row[2], age=row[3])
        return None
    
    def create(self, user: User) -> int:
        conn = self._get_connection()
        cursor = conn.cursor()
        cursor.execute(
            'INSERT INTO users (name, email, age) VALUES (?, ?, ?)',
            (user.name, user.email, user.age)
        )
        conn.commit()
        user_id = cursor.lastrowid
        conn.close()
        return user_id
```

**5. 业务逻辑层** (`services/user_service.py`)
```python
from typing import List, Optional
from models.user import User
from repositories.user_repo import UserRepository

class UserService:
    def __init__(self, repo: UserRepository):
        self.repo = repo
    
    def get_all_users(self) -> List[User]:
        return self.repo.find_all()
    
    def get_user_by_id(self, user_id: int) -> Optional[User]:
        return self.repo.find_by_id(user_id)
    
    def create_user(self, name: str, email: str, age: int = 0) -> User:
        # 业务逻辑：检查邮箱是否已存在
        existing = self.repo.find_by_email(email)
        if existing:
            raise ValueError(f"Email {email} already exists")
        
        # 创建用户
        user = User(name=name, email=email, age=age)
        user.id = self.repo.create(user)
        return user
```

**6. 请求处理层** (`controllers/user_controller.py`)
```python
from flask import Blueprint, request, jsonify
from services.user_service import UserService
from schemas.user_schema import CreateUserSchema

user_bp = Blueprint('users', __name__)

# 依赖注入
user_service = None

def init_controller(user_service: UserService):
    global user_service
    user_service = user_service

@user_bp.route('/users', methods=['GET'])
def get_users():
    users = user_service.get_all_users()
    return jsonify([u.to_dict() for u in users])

@user_bp.route('/users', methods=['POST'])
def create_user():
    try:
        data = request.json
        schema = CreateUserSchema.from_dict(data)
        user = user_service.create_user(schema.name, schema.email, schema.age)
        return jsonify(user.to_dict()), 201
    except ValueError as e:
        return jsonify({'error': str(e)}), 400

@user_bp.route('/users/<int:user_id>', methods=['GET'])
def get_user(user_id):
    user = user_service.get_user_by_id(user_id)
    if not user:
        return jsonify({'error': 'User not found'}), 404
    return jsonify(user.to_dict())
```

**7. 应用入口** (`app.py`)
```python
from flask import Flask
from config import Config
from models.user import User
from repositories.user_repo import UserRepository
from services.user_service import UserService
from controllers.user_controller import user_bp, init_controller
import sqlite3

def create_app():
    app = Flask(__name__)
    
    # 初始化数据库
    conn = sqlite3.connect(Config.DATABASE_URI)
    conn.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT,
            email TEXT,
            age INTEGER
        )
    ''')
    conn.close()
    
    # 依赖注入
    repo = UserRepository(Config.DATABASE_URI)
    service = UserService(repo)
    init_controller(service)
    
    app.register_blueprint(user_bp)
    return app

if __name__ == '__main__':
    app = create_app()
    app.run(debug=Config.DEBUG == 'True')
```

## 案例2：支付系统（策略模式实现开闭原则）

### 需求
支持多种支付方式：支付宝、微信、银行卡

### 原始代码

```python
# bad_payment.py
def pay(order_id, amount, payment_type):
    if payment_type == 'alipay':
        # 支付宝支付逻辑
        print(f"调用支付宝API，订单{order_id}，金额{amount}")
        return True
    elif payment_type == 'wechat':
        # 微信支付逻辑
        print(f"调用微信API，订单{order_id}，金额{amount}")
        return True
    elif payment_type == 'bankcard':
        # 银行卡支付逻辑
        print(f"调用银行API，订单{order_id}，金额{amount}")
        return True
    else:
        raise ValueError(f"不支持的支付方式: {payment_type}")
```

### 重构后（策略模式）

```python
# payment/
# ├── __init__.py
# ├── interface.py
# ├── strategies/
# │   ├── __init__.py
# │   ├── alipay.py
# │   ├── wechat.py
# │   └── bankcard.py
# └── factory.py

# interface.py
from abc import ABC, abstractmethod

class PaymentStrategy(ABC):
    @abstractmethod
    def pay(self, order_id: str, amount: float) -> bool:
        """执行支付"""
        pass
    
    @abstractmethod
    def refund(self, order_id: str, amount bool:
        """执行退款"""
       : float) -> pass

# strategies/alipay.py
class AlipayStrategy(PaymentStrategy):
    def pay(self, order_id: str, amount: float) -> bool:
        print(f"支付宝支付: 订单{order_id}, 金额{amount}")
        # 实际调用支付宝API
        return True
    
    def refund(self, order_id: str, amount: float) -> bool:
        print(f"支付宝退款: 订单{order_id}, 金额{amount}")
        return True

# strategies/wechat.py  
class WechatPayStrategy(PaymentStrategy):
    def pay(self, order_id: str, amount: float) -> bool:
        print(f"微信支付: 订单{order_id}, 金额{amount}")
        return True
    
    def refund(self, order_id: str, amount: float) -> bool:
        print(f"微信退款: 订单{order_id}, 金额{amount}")
        return True

# strategies/bankcard.py
class BankCardStrategy(PaymentStrategy):
    def pay(self, order_id: str, amount: float) -> bool:
        print(f"银行卡支付: 订单{order_id}, 金额{amount}")
        return True
    
    def refund(self, order_id: str, amount: float) -> bool:
        print(f"银行卡退款: 订单{order_id}, 金额{amount}")
        return True

# factory.py
class PaymentFactory:
    _strategies = {
        'alipay': AlipayStrategy,
        'wechat': WechatPayStrategy,
        'bankcard': BankCardStrategy
    }
    
    @classmethod
    def get_strategy(cls, payment_type: str) -> PaymentStrategy:
        strategy_class = cls._strategies.get(payment_type)
        if not strategy_class:
            raise ValueError(f"不支持的支付方式: {payment_type}")
        return strategy_class()

# 使用
def process_payment(order_id, amount, payment_type):
    strategy = PaymentFactory.get_strategy(payment_type)
    return strategy.pay(order_id, amount)

# 测试
if __name__ == '__main__':
    process_payment('ORDER001', 99.9, 'alipay')
    process_payment('ORDER002', 88.8, 'wechat')
    process_payment('ORDER003', 77.7, 'bankcard')
```

## 案例3：依赖注入实践

### 问题：硬编码依赖

```python
# bad_example.py
class UserService:
    def __init__(self):
        self.db = MySQLDatabase()  # 硬编码
    
    def get_user(self, user_id):
        return self.db.query(f"SELECT * FROM users WHERE id = {user_id}")
```

### 解决：依赖注入

```python
# good_example.py
from abc import ABC, abstractmethod
from typing import Optional

# 抽象接口
class Database(ABC):
    @abstractmethod
    def query(self, sql: str) -> Optional[dict]:
        pass

# 具体实现
class MySQLDatabase(Database):
    def query(self, sql: str) -> Optional[dict]:
        # MySQL 查询逻辑
        print(f"MySQL: {sql}")
        return {}

class PostgreSQLDatabase(Database):
    def query(self, sql: str) -> Optional[dict]:
        # PostgreSQL 查询逻辑
        print(f"PostgreSQL: {sql}")
        return {}

# 依赖注入
class UserService:
    def __init__(self, db: Database):  # 注入抽象，不依赖具体
        self.db = db
    
    def get_user(self, user_id: int):
        return self.db.query(f"SELECT * FROM users WHERE id = {user_id}")

# 使用时注入具体实现
mysql_db = MySQLDatabase()
service = UserService(mysql_db)

# 换成 PostgreSQL 只需要：
pg_db = PostgreSQLDatabase()
service = UserService(pg_db)
```

## 代码对比总结

| 维度 | 重构前 | 重构后 |
|------|--------|--------|
| 可测试性 | 难，依赖真实数据库 | 易，可注入mock |
| 可扩展性 | 加支付方式改原代码 | 加类就行 |
| 可读性 | 一大坨 | 职责清晰 |
| 可维护性 | 牵一发而动全身 | 各层独立修改 |

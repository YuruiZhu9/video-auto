# Python项目架构案例

## 案例：用户管理系统

### 初始状态（无架构）

```python
# app.py - 所有代码在一个文件
from flask import Flask, request, jsonify

app = Flask(__name__)

# 模拟数据库
users = []

@app.route('/users', methods=['GET'])
def get_users():
    return jsonify(users)

@app.route('/users', methods=['POST'])
def create_user():
    data = request.json
    user = {
        'id': len(users) + 1,
        'name': data.get('name'),
        'email': data.get('email'),
        'password': data.get('password')  # 密码明文存储！
    }
    users.append(user)
    return jsonify(user), 201

@app.route('/users/<int:user_id>', methods=['GET'])
def get_user(user_id):
    for user in users:
        if user['id'] == user_id:
            return jsonify(user)
    return jsonify({'error': 'User not found'}), 404

@app.route('/users/<int:user_id>', methods=['PUT'])
def update_user(user_id):
    data = request.json
    for user in users:
        if user['id'] == user_id:
            user['name'] = data.get('name', user['name'])
            user['email'] = data.get('email', user['email'])
            return jsonify(user)
    return jsonify({'error': 'User not found'}), 404

@app.route('/users/<int:user_id>', methods=['DELETE'])
def delete_user(user_id):
    global users
    users = [u for u in users if u['id'] != user_id]
    return '', 204

# 问题：
# 1. 数据、逻辑、路由全混在一起
# 2. 没有密码加密
# 3. 没有数据验证
# 4. 无法测试
# 5. 难以扩展
```

#### 1. Model层

```python
# models/user.py
from dataclasses import dataclass
from typing import Optional

@dataclass
class User:
    id: Optional[int] = None
    name: str = ""
    email: str = ""
    password_hash: str = ""
    
    def to_dict(self):
        return {
            'id': self.id,
            'name': self.name,
            'email': self.email
            # 不返回password_hash
        }
```

#### 2. Schema层（数据验证）

```python
# schemas/user_schema.py
from dataclasses import dataclass
from typing import Optional
import re

@dataclass
class CreateUserSchema:
    name: str
    email: str
    password: str
    
    def validate(self) -> tuple[bool, Optional[str]]:
        if not self.name or len(self.name) < 2:
            return False, "Name must be at least 2 characters"
        
        if not re.match(r'^[\w\.-]+@[\w\.-]+\.\w+$', self.email):
            return False, "Invalid email format"
        
        if len(self.password) < 6:
            return False, "Password must be at least 6 characters"
        
        return True, None

@dataclass  
class UpdateUserSchema:
    name: Optional[str] = None
    email: Optional[str] = None
```

#### 3. Repository层（数据访问）

```python
# repositories/user_repository.py
from typing import Optional, List
from models.user import User

class UserRepository:
    def __init__(self):
        self._users = []
        self._next_id = 1
    
    def find_all(self) -> List[User]:
        return self._users.copy()
    
    def find_by_id(self, user_id: int) -> Optional[User]:
        for user in self._users:
            if user.id == user_id:
                return user
        return None
    
    def find_by_email(self, email: str) -> Optional[User]:
        for user in self._users:
            if user.email == email:
                return user
        return None
    
    def save(self, user: User) -> User:
        if user.id is None:
            user.id = self._next_id
            self._next_id += 1
            self._users.append(user)
        return user
    
    def delete(self, user_id: int) -> bool:
        for i, user in enumerate(self._users):
            if user.id == user_id:
                self._users.pop(i)
                return True
        return False
```

#### 4. Service层（业务逻辑）

```python
# services/user_service.py
from typing import Optional, List
from models.user import User
from repositories.user_repository import UserRepository
from schemas.user_schema import CreateUserSchema, UpdateUserSchema
from utils.security import hash_password, verify_password

class UserService:
    def __init__(self, repository: UserRepository):
        self._repository = repository
    
    def get_all_users(self) -> List[User]:
        return self._repository.find_all()
    
    def get_user_by_id(self, user_id: int) -> Optional[User]:
        return self._repository.find_by_id(user_id)
    
    def create_user(self, name: str, email: str, password: str) -> tuple[Optional[User], Optional[str]]:
        # 验证
        schema = CreateUserSchema(name=name, email=email, password=password)
        valid, error = schema.validate()
        if not valid:
            return None, error
        
        # 检查邮箱是否已存在
        if self._repository.find_by_email(email):
            return None, "Email already exists"
        
        # 创建用户（密码加密）
        user = User(
            name=name,
            email=email,
            password_hash=hash_password(password)
        )
        
        return self._repository.save(user), None
    
    def update_user(self, user_id: int, name: str = None, email: str = None) -> tuple[Optional[User], Optional[str]]:
        user = self._repository.find_by_id(user_id)
        if not user:
            return None, "User not found"
        
        if name:
            user.name = name
        if email:
            user.email = email
        
        return self._repository.save(user), None
    
    def delete_user(self, user_id: int) -> bool:
        return self._repository.delete(user_id)
```

#### 5. Utils层（工具函数）

```python
# utils/security.py
import hashlib
import secrets

def hash_password(password: str) -> str:
    """密码哈希"""
    salt = secrets.token_hex(16)
    pwd_hash = hashlib.pbkdf2_hmac('sha256', password.encode(), salt.encode(), 100000)
    return f"{salt}${pwd_hash.hex()}"

def verify_password(password: str, password_hash: str) -> bool:
    """验证密码"""
    try:
        salt, pwd_hash = password_hash.split('$')
        new_hash = hashlib.pbkdf2_hmac('sha256', password.encode(), salt.encode(), 100000)
        return new_hash.hex() == pwd_hash
    except:
        return False
```

#### 6. 主应用（组装）

```python
# app.py
from flask import Flask, request, jsonify
from models.user import User
from repositories.user_repository import UserRepository
from services.user_service import UserService

app = Flask(__name__)

# 依赖注入
repository = UserRepository()
user_service = UserService(repository)

@app.route('/users', methods=['GET'])
def get_users():
    users = user_service.get_all_users()
    return jsonify([u.to_dict() for u in users])

@app.route('/users', methods=['POST'])
def create_user():
    data = request.json
    user, error = user_service.create_user(
        name=data.get('name'),
        email=data.get('email'),
        password=data.get('password')
    )
    
    if error:
        return jsonify({'error': error}), 400
    
    return jsonify(user.to_dict()), 201

@app.route('/users/<int:user_id>', methods=['GET'])
def get_user(user_id):
    user = user_service.get_user_by_id(user_id)
    if not user:
        return jsonify({'error': 'User not found'}), 404
    return jsonify(user.to_dict())

@app.route('/users/<int:user_id>', methods=['PUT'])
def update_user(user_id):
    data = request.json
    user, error = user_service.update_user(
        user_id=user_id,
        name=data.get('name'),
        email=data.get('email')
    )
    
    if error:
        return jsonify({'error': error}), 404
    
    return jsonify(user.to_dict())

@app.route('/users/<int:user_id>', methods=['DELETE'])
def delete_user(user_id):
    success = user_service.delete_user(user_id)
    if not success:
        return jsonify({'error': 'User not found'}), 404
    return '', 204
```

## 重构收益

| 方面 | 重构前 | 重构后 |
|------|--------|--------|
| 代码行数 | ~100行 | ~200行（但更清晰） |
| 可测试性 | 难 | 各层可单独测试 |
| 可扩展性 | 困难 | 依赖注入易替换 |
| 安全性 | 明文密码 | PBKDF2加密 |
| 数据验证 | 无 | 统一Schema验证 |
| 维护性 | 差 | 好 |

---

## 关键原则应用

1. **单一职责**：每层只做自己的事
2. **依赖倒置**：Service依赖Repository抽象
3. **开闭原则**：新增验证规则只需改Schema
4. **DRY**：密码工具复用

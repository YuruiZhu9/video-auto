// app.ts

interface Task {
    id: number;
    title: string;
    description: string;
    status: 'pending' | 'in-progress' | 'completed';
    assigneeId: number;
    createdAt: Date;
}

const tasks: Task[] = [];
let taskIdCounter = 1;

// 路由和处理逻辑混在一起
app.get('/tasks', (req, res) => {
    const { status, assigneeId } = req.query;
    
    let filteredTasks = tasks;
    if (status) {
        filteredTasks = filteredTasks.filter(t => t.status === status);
    }
    if (assigneeId) {
        filteredTasks = filteredTasks.filter(t => t.assigneeId === Number(assigneeId));
    }
    
    res.json(filteredTasks);
});

app.post('/tasks', (req, res) => {
    const { title, description, assigneeId } = req.body;
    
    // 验证混在路由中
    if (!title || title.trim() === '') {
        res.status(400).json({ error: 'Title is required' });
        return;
    }
    
    if (title.length > 100) {
        res.status(400).json({ error: 'Title too long' });
        return;
    }
    
    const task: Task = {
        id: taskIdCounter++,
        title,
        description: description || '',
        status: 'pending',
        assigneeId: assigneeId || null,
        createdAt: new Date()
    };
    
    tasks.push(task);
    res.status(201).json(task);
});

app.put('/tasks/:id', (req, res) => {
    const taskId = Number(req.params.id);
    const task = tasks.find(t => t.id === taskId);
    
    if (!task) {
        res.status(404).json({ error: 'Task not found' });
        return;
    }
    
    const { title, description, status, assigneeId } = req.body;
    
    if (title !== undefined) task.title = title;
    if (description !== undefined) task.description = description;
    if (status !== undefined) task.status = status;
    if (assigneeId !== undefined) task.assigneeId = assigneeId;
    
    res.json(task);
});

app.delete('/tasks/:id', (req, res) => {
    const taskId = Number(req.params.id);
    const index = tasks.findIndex(t => t.id === taskId);
    
    if (index === -1) {
        res.status(404).json({ error: 'Task not found' });
        return;
    }
    
    tasks.splice(index, 1);
    res.json({ success: true });
});

### 重构后（分层架构 + SOLID）

```typescript
// ✅ models/task.model.ts - 数据模型
export enum TaskStatus {
    PENDING = 'pending',
    IN_PROGRESS = 'in-progress',
    COMPLETED = 'completed'
}

export interface Task {
    id: number;
    title: string;
    description: string;
    status: TaskStatus;
    assigneeId: number | null;
    createdAt: Date;
}

// models/user.model.ts
export interface User {
    id: number;
    name: string;
    email: string;
}

// dtos/create-task.dto.ts - 数据传输对象
export interface CreateTaskDto {
    title: string;
    description?: string;
    assigneeId?: number;
}

export interface UpdateTaskDto {
    title?: string;
    description?: string;
    status?: TaskStatus;
    assigneeId?: number | null;
}

// repositories/task.repository.ts - 数据访问层
import { Task, TaskStatus } from '../models/task.model';

export interface TaskFilter {
    status?: TaskStatus;
    assigneeId?: number;
}

export class TaskRepository {
    private tasks: Task[] = [];
    private idCounter = 1;
    
    findAll(filter?: TaskFilter): Task[] {
        let result = [...this.tasks];
        
        if (filter?.status) {
            result = result.filter(t => t.status === filter.status);
        }
        if (filter?.assigneeId !== undefined) {
            result = result.filter(t => t.assigneeId === filter.assigneeId);
        }
        
        return result;
    }
    
    findById(id: number): Task | undefined {
        return this.tasks.find(t => t.id === id);
    }
    
    create(task: Omit<Task, 'id' | 'createdAt'>): Task {
        const newTask: Task = {
            ...task,
            id: this.idCounter++,
            createdAt: new Date()
        };
        this.tasks.push(newTask);
        return newTask;
    }
    
    update(id: number, updates: Partial<Task>): Task | undefined {
        const task = this.findById(id);
        if (!task) return undefined;
        
        Object.assign(task, updates);
        return task;
    }
    
    delete(id: number): boolean {
        const index = this.tasks.findIndex(t => t.id === id);
        if (index === -1) return false;
        
        this.tasks.splice(index, 1);
        return true;
    }
}

// services/task.service.ts - 业务逻辑层
import { Task, TaskStatus } from '../models/task.model';
import { CreateTaskDto, UpdateTaskDto } from '../dtos/create-task.dto';
import { TaskRepository, TaskFilter } from '../repositories/task.repository';

// 依赖倒置：依赖抽象接口
export interface ITaskRepository {
    findAll(filter?: TaskFilter): Task[];
    findById(id: number): Task | undefined;
    create(task: Omit<Task, 'id' | 'createdAt'>): Task;
    update(id: number, updates: Partial<Task>): Task | undefined;
    delete(id: number): boolean;
}

// 单一职责：验证逻辑
export class TaskValidator {
    static validateCreate(dto: CreateTaskDto): string[] {
        const errors: string[] = [];
        
        if (!dto.title || dto.title.trim() === '') {
            errors.push('Title is required');
        }
        
        if (dto.title && dto.title.length > 100) {
            errors.push('Title must be less than 100 characters');
        }
        
        return errors;
    }
    
    static validateUpdate(dto: UpdateTaskDto): string[] {
        const errors: string[] = [];
        
        if (dto.title !== undefined && dto.title.trim() === '') {
            errors.push('Title cannot be empty');
        }
        
        if (dto.status !== undefined && !Object.values(TaskStatus).includes(dto.status)) {
            errors.push('Invalid status');
        }
        
        return errors;
    }
}

export class TaskService {
    constructor(private readonly repository: ITaskRepository) {}
    
    getAllTasks(filter?: TaskFilter): Task[] {
        return this.repository.findAll(filter);
    }
    
    getTaskById(id: number): Task | undefined {
        return this.repository.findById(id);
    }
    
    createTask(dto: CreateTaskDto): Task {
        // 验证
        const errors = TaskValidator.validateCreate(dto);
        if (errors.length > 0) {
            throw new ValidationError(errors.join(', '));
        }
        
        // 业务逻辑
        return this.repository.create({
            title: dto.title.trim(),
            description: dto.description || '',
            status: TaskStatus.PENDING,
            assigneeId: dto.assigneeId || null
        });
    }
    
    updateTask(id: number, dto: UpdateTaskDto): Task {
        // 验证
        const errors = TaskValidator.validateUpdate(dto);
        if (errors.length > 0) {
            throw new ValidationError(errors.join(', '));
        }
        
        // 检查是否存在
        const existing = this.repository.findById(id);
        if (!existing) {
            throw new NotFoundError('Task not found');
        }
        
        // 业务逻辑
        return this.repository.update(id, dto) as Task;
    }
    
    deleteTask(id: number): void {
        const deleted = this.repository.delete(id);
        if (!deleted) {
            throw new NotFoundError('Task not found');
        }
    }
}

// 自定义错误类
export class ValidationError extends Error {
    constructor(message: string) {
        super(message);
        this.name = 'ValidationError';
    }
}

export class NotFoundError extends Error {
    constructor(message: string) {
        super(message);
        this.name = 'NotFoundError';
    }
}

// controllers/task.controller.ts - 控制器层
import { Task } from '../models/task.model';
import { CreateTaskDto, UpdateTaskDto } from '../dtos/create-task.dto';
import { TaskService, ValidationError, NotFoundError } from '../services/task.service';
import { TaskStatus } from '../models/task.model';

// 依赖注入
const taskRepository = new TaskRepository();
const taskService = new TaskService(taskRepository);

interface Request {
    params: { id: string };
    query: { status?: TaskStatus; assigneeId?: string };
    body: any;
}

interface Response {
    status(code: number): Response;
    json(data: any): Response;
}

// 控制器处理请求和响应
class TaskController {
    getAllTasks(req: Request, res: Response): void {
        const filter = {
            status: req.query.status as TaskStatus,
            assigneeId: req.query.assigneeId ? Number(req.query.assigneeId) : undefined
        };
        
        const tasks = taskService.getAllTasks(filter);
        res.status(200).json(tasks);
    }
    
    getTaskById(req: Request, res: Response): void {
        const id = Number(req.params.id);
        const task = taskService.getTaskById(id);
        
        if (!task) {
            res.status(404).json({ error: 'Task not found' });
            return;
        }
        
        res.status(200).json(task);
    }
    
    createTask(req: Request, res: Response): void {
        try {
            const dto: CreateTaskDto = req.body;
            const task = taskService.createTask(dto);
            res.status(201).json(task);
        } catch (error) {
            if (error instanceof ValidationError) {
                res.status(400).json({ error: error.message });
                return;
            }
            throw error;
        }
    }
    
    updateTask(req: Request, res: Response): void {
        try {
            const id = Number(req.params.id);
            const dto: UpdateTaskDto = req.body;
            const task = taskService.updateTask(id, dto);
            res.status(200).json(task);
        } catch (error) {
            if (error instanceof ValidationError) {
                res.status(400).json({ error: error.message });
                return;
            }
            if (error instanceof NotFoundError) {
                res.status(404).json({ error: error.message });
                return;
            }
            throw error;
        }
    }
    
    deleteTask(req: Request, res: Response): void {
        try {
            const id = Number(req.params.id);
            taskService.deleteTask(id);
            res.status(200).json({ success: true });
        } catch (error) {
            if (error instanceof NotFoundError) {
                res.status(404).json({ error: error.message });
                return;
            }
            throw error;
        }
    }
}

// main.ts - 入口文件
const taskController = new TaskController();

// 路由绑定
app.get('/tasks', (req, res) => taskController.getAllTasks(req, res));
app.get('/tasks/:id', (req, res) => taskController.getTaskById(req, res));
app.post('/tasks', (req, res) => taskController.createTask(req, res));
app.put('/tasks/:id', (req, res) => taskController.updateTask(req, res));
app.delete('/tasks/:id', (req, res) => taskController.deleteTask(req, res));
```

### 重构收益

| 方面 | 重构前 | 重构后 |
|------|--------|--------|
| 代码组织 | 1个文件，200+行 | 7个文件，职责清晰 |
| 单元测试 | 困难 | 容易（各层可独立测试） |
| 扩展性 | 差 | 好（依赖注入） |
| 代码复用 | 低 | 高 |
| 类型安全 | 无 | 完整（TypeScript） |
| 错误处理 | 分散 | 集中（自定义错误类） |
| 可读性 | 差 | 好 |

## 实现层

### 召回策略实现

```typescript
// modules/recommend/services/category-recall.strategy.ts

import { Injectable } from '@nestjs/common';
import { RecallStrategy } from '../strategies/recall.strategy.interface';
import { RecommendationItem } from '../../../common/interfaces/recommend.interface';
import { User } from '../../user/interfaces/user.interface';
import { ItemRepository } from '../../item/repositories/item.repository';

@Injectable()
export class CategoryRecallStrategy implements RecallStrategy {
  readonly strategyName = 'CategoryRecall';
  
  constructor(private readonly itemRepo: ItemRepository) {}
  
  async recall(user: User, count: number): Promise<RecommendationItem[]> {
    // 获取用户历史中的分类
    const userCategories = this.extractUserCategories(user);
    
    // 根据用户偏好分类召回
    const items = await this.itemRepo.findByCategories(
      userCategories, 
      count
    );
    
    return items.map(item => ({
      itemId: item.id,
      itemName: item.name,
      category: item.category,
      price: item.price,
      score: this.calculateScore(user, item),
      reason: `推荐${item.category}类商品`
    }));
  }
  
  private extractUserCategories(user: User): string[] {
    // 从用户历史提取偏好分类
    const categoryMap = new Map<string, number>();
    for (const item of user.history) {
      const count = categoryMap.get(item.category) || 0;
      categoryMap.set(item.category, count + 1);
    }
    return Array.from(categoryMap.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 3)
      .map(([cat]) => cat);
  }
  
  private calculateScore(user: User, item: any): number {
    let score = 0.5;
    if (user.preferredCategories?.includes(item.category)) {
      score += 0.5;
    }
    return score;
  }
}
```

## 业务服务层

```typescript
// modules/recommend/services/recommend.service.ts

import { Injectable, Inject } from '@nestjs/common';
import { ConfigType } from '@nestjs/config';
import { v4 as uuidv4 } from 'uuid';

import { RecommendServiceInterface } from './recommend.service.interface';
import { RecallStrategy } from '../strategies/recall.strategy.interface';
import { RankStrategy } from '../strategies/rank.strategy.interface';
import { UserService } from '../../user/user.service';
import { 
  RecommendRequest, 
  RecommendResponse,
  RecommendationItem 
} from '../../../common/interfaces/recommend.interface';
import recommendationConfig from '../recommendation.config';

@Injectable()
export class RecommendServiceImpl implements RecommendServiceInterface {
  
  constructor(
    private readonly userService: UserService,
    @Inject('CategoryRecallStrategy')
    private readonly categoryRecall: RecallStrategy,
    @Inject('CFRecallStrategy')
    private readonly cfRecall: RecallStrategy,
    @Inject('DefaultRankStrategy')
    private readonly rankStrategy: RankStrategy,
    @Inject(recommendationConfig.KEY)
    private readonly config: ConfigType<typeof recommendationConfig>,
  ) {}
  
  async recommend(request: RecommendRequest): Promise<RecommendResponse> {
    const { userId, count, scene } = request;
    const recommendCount = count || this.config.defaultCount;
    
    // 1. 获取用户信息
    const user = await this.userService.getUserById(userId);
    if (!user) {
      return { items: [], requestId: uuidv4() };
    }
    
    // 2. 召回（根据场景选择策略）
    const recallStrategy = this.selectRecallStrategy(scene);
    const candidates = await recallStrategy.recall(user, recommendCount * 3);
    
    // 3. 过滤已购买/已推荐
    const filteredCandidates = this.filterCandidates(candidates, user);
    
    // 4. 排序
    const rankedItems = await this.rankStrategy.rank(filteredCandidates, user);
    
    // 5. 返回结果
    return {
      items: rankedItems.slice(0, recommendCount),
      requestId: uuidv4(),
      exp: this.config.cacheTTL
    };
  }
  
  private selectRecallStrategy(scene?: string): RecallStrategy {
    switch (scene) {
      case 'home':
        return this.categoryRecall;
      case 'detail':
        return this.cfRecall;
      default:
        return this.categoryRecall;
    }
  }
  
  private filterCandidates(
    candidates: RecommendationItem[], 
    user: any
  ): RecommendationItem[] {
    const purchasedIds = new Set(user.purchasedItems || []);
    return candidates.filter(item => !purchasedIds.has(item.itemId));
  }
}
```

## 控制层

```typescript
// modules/recommend/recommend.controller.ts

import { Controller, Post, Body, Get, Query } from '@nestjs/common';
import { RecommendService } from './interfaces/recommend.service.interface';
import { RecommendRequest, RecommendResponse } from '../../common/interfaces/recommend.interface';

@Controller('api/recommend')
export class RecommendController {
  constructor(private readonly recommendService: RecommendService) {}
  
  @Post()
  async recommend(@Body() dto: RecommendRequest): Promise<RecommendResponse> {
    return this.recommendService.recommend(dto);
  }
  
  @Get('home')
  async getHomeRecommend(
    @Query('userId') userId: string,
    @Query('count') count?: string
  ): Promise<RecommendResponse> {
    return this.recommendService.recommend({
      userId,
      count: count ? parseInt(count, 10) : undefined,
      scene: 'home'
    });
  }
}
```

## 架构优势

| 特性 | 说明 |
|------|------|
| **依赖注入** | 通过NestJS的DI系统管理依赖 |
| **策略模式** | 轻松切换召回/排序策略 |
| **接口契约** | 定义清晰的数据结构 |
| **模块化** | 各模块职责分离 |
| **可测试** | 每个Strategy可单独测试 |

**这种架构非常适合推荐系统**：算法可以灵活切换，数据层和业务层解耦。

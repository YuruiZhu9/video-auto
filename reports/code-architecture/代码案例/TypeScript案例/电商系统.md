### 3. 单一职责原则（SRP）- 仓储层

```typescript
// repositories/order-repository.ts
import { Order } from '../models/order';

export class OrderRepository {
  private orders: Map<string, Order> = new Map();
  
  async findById(id: string): Promise<Order | null> {
    return this.orders.get(id) || null;
  }
  
  async findByUserId(userId: string): Promise<Order[]> {
    return Array.from(this.orders.values()).filter(
      order => order.userId === userId
    );
  }
  
  async save(order: Order): Promise<Order> {
    this.orders.set(order.id, order);
    return order;
  }
  
  async update(order: Order): Promise<boolean> {
    if (!this.orders.has(order.id)) {
      return false;
    }
    this.orders.set(order.id, order);
    return true;
  }
}
```

### 4. 业务服务层

```typescript
// services/order-service.ts
import { Order, OrderItem } from '../models/order';
import { OrderRepository } from '../repositories/order-repository';
import { PaymentStrategy } from '../strategies/payment-strategy';

export class OrderService {
  constructor(
    private orderRepository: OrderRepository,
    private paymentStrategy: PaymentStrategy
  ) {}
  
  // 创建订单
  async createOrder(userId: string, items: OrderItem[]): Promise<Order> {
    // 业务验证
    if (!items || items.length === 0) {
      throw new Error('订单至少需要一个商品');
    }
    
    // 计算总价
    const totalAmount = items.reduce(
      (sum, item) => sum + item.price * item.quantity, 
      0
    );
    
    // 创建订单
    const order: Order = {
      id: this.generateOrderId(),
      userId,
      items,
      totalAmount,
      status: 'pending',
      createdAt: new Date()
    };
    
    await this.orderRepository.save(order);
    return order;
  }
  
  // 支付订单 - 使用依赖注入的支付策略
  async payOrder(orderId: string): Promise<boolean> {
    const order = await this.orderRepository.findById(orderId);
    
    if (!order) {
      throw new Error('订单不存在');
    }
    
    if (order.status !== 'pending') {
      throw new Error('订单状态不正确');
    }
    
    // 调用支付策略
    const success = await this.paymentStrategy.pay(order.totalAmount);
    
    if (success) {
      order.status = 'paid';
      order.paidAt = new Date();
      await this.orderRepository.update(order);
    }
    
    return success;
  }
  
  // 切换支付方式（开闭原则）
  setPaymentStrategy(strategy: PaymentStrategy): void {
    // 运行时切换
    // 但这里需要新实例，看下文依赖注入容器
  }
  
  private generateOrderId(): string {
    return `ORD${Date.now()}${Math.random().toString(36).substr(2, 9)}`;
  }
}
```

### 5. 依赖注入容器

```typescript
// container.ts
import { OrderService } from './services/order-service';
import { OrderRepository } from './repositories/order-repository';
import { AlipayStrategy, WechatPayStrategy, CardPayStrategy } from './strategies/strategies';

// 简单的DI容器
class Container {
  private services: Map<string, any> = new Map();
  
  register<T>(name: string, instance: T): void {
    this.services.set(name, instance);
  }
  
  get<T>(name: string): T {
    const service = this.services.get(name);
    if (!service) {
      throw new Error(`Service ${name} not found`);
    }
    return service;
  }
}

// 创建全局容器
export const container = new Container();

// 初始化服务
const orderRepository = new OrderRepository();
const alipayStrategy = new AlipayStrategy();

container.register('orderRepository', orderRepository);
container.register('alipay', alipayStrategy);
container.register('wechatPay', new WechatPayStrategy());
container.register('cardPay', new CardPayStrategy());

// 初始化订单服务（使用支付宝）
export const orderService = new OrderService(
  container.get('orderRepository'),
  container.get('alipay')
);
```

### 6. 使用示例

```typescript
// index.ts
import { orderService } from './container';
import { WechatPayStrategy } from './strategies/strategies/wechat-pay-strategy';
import { container } from './container';

async function main() {
  // 创建订单
  const order = await orderService.createOrder('user123', [
    { productId: 'p1', name: 'iPhone 15', price: 6999, quantity: 1 },
    { productId: 'p2', name: 'AirPods Pro', price: 1999, quantity: 1 }
  ]);
  
  console.log('创建订单:', order);
  
  // 使用支付宝支付
  await orderService.payOrder(order.id);
  
  // 切换支付方式（运行时改变行为，不改代码）
  const wechatService = new OrderService(
    container.get('orderRepository'),
    container.get('wechatPay')
  );
  
  const order2 = await wechatService.createOrder('user456', [
    { productId: 'p3', name: 'MacBook Pro', price: 14999, quantity: 1 }
  ]);
  
  await wechatService.payOrder(order2);
}

main();
```

## 总结

| 原则 | 实践 |
|------|------|
| SRP | OrderRepository只管数据、OrderService只管业务 |
| OCP | 新增支付方式只需添加新类，不改原有代码 |
| LSP | 所有支付策略都实现PaymentStrategy接口 |
| ISP | PaymentStrategy接口只定义支付相关方法 |
| DIP | OrderService依赖抽象PaymentStrategy，不依赖具体实现 |

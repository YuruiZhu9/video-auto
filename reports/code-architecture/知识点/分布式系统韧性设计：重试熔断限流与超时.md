# 分布式系统韧性设计：重试、熔断、限流与超时

## 概念解释

分布式系统最怕什么？**不确定性和级联失败**。一次网络抖动、一个服务超时，可能引发整个系统雪崩。

韧性（Resilience）设计的核心思想是：**承认故障必然发生，然后设计容错机制**。

四大核心模式：

| 模式 | 解决的问题 | 类比 |
|------|-----------|------|
| **超时（Timeout）** | 无限等待 | 打电话超过N秒就挂断 |
| **重试（Retry）** | 瞬时故障 | 第一次没接通，再拨一次 |
| **熔断（Circuit Breaker）** | 防止级联失败 | 保险丝，跳闸保护 |
| **限流（Rate Limiting）** | 保护系统不被压垮 | 景区限流，防止踩踏 |

## 一、超时（Timeout）

### 为什么超时最重要？

> 慢的响应 = 坏的响应。在高并发系统中，一个永不返回的请求会耗尽所有线程资源。

### 反例（无超时设置）

```python
# ❌ 问题：请求没有超时控制，可能永久阻塞
import requests

def get_recommendations(user_id: str):
    # 如果下游服务挂了，这个请求会永远等待
    response = requests.get(f"http://recommend-service/api/{user_id}/items")
    return response.json()
```

### 正例（设置超时）

```python
# ✅ 正确：设置连接超时 + 读取超时
import requests
from requests.exceptions import ConnectTimeout, ReadTimeout

def get_recommendations(user_id: str, timeout: tuple = (3, 5)):
    """
    timeout: (connect_timeout, read_timeout)，单位：秒
    connect_timeout = 3：建立连接超过3秒就放弃
    read_timeout = 5：服务器响应超过5秒就放弃
    """
    try:
        response = requests.get(
            f"http://recommend-service/api/{user_id}/items",
            timeout=timeout
        )
        response.raise_for_status()
        return response.json()
    except (ConnectTimeout, ReadTimeout):
        # 触发降级：返回默认推荐
        return get_default_recommendations(user_id)
    except requests.exceptions.RequestException as e:
        logger.error(f"推荐服务请求失败: {e}")
        return get_default_recommendations(user_id)


def get_default_recommendations(user_id: str):
    """降级方案：返回热门推荐，不依赖算法服务"""
    return {"items": [{"id": "hot_1", "score": 0.9}], "source": "fallback"}
```

### 超时设置原则

```
超时设置 = 基准延迟 × 2 + 缓冲

示例：
├─ 正常P99延迟 = 50ms  → 超时设为 150-200ms
├─ 正常P99延迟 = 200ms → 超时设为 500-800ms
└─ 正常P99延迟 = 1s    → 超时设为 3s

原则：超时不要设太长，否则失去保护意义
```

## 二、重试（Retry）

### 什么情况适合重试？

| ✅ 适合重试 | ❌ 不适合重试 |
|-----------|-------------|
| 网络抖动（瞬时不可达） | 业务逻辑错误（如余额不足） |
| 超时（服务临时过载） | 幂等性无法保证的写操作 |
| 503 Service Unavailable | 资源不足（会加重负担） |
| 409 Conflict（轻微冲突） | 认证失败（Token无效） |

### 反例（暴力重试）

```python
# ❌ 问题1：无限重试，可能引发雪崩
def get_user_profile(user_id: str):
    for _ in range(100):  # 无限重试！
        try:
            return call_service(user_id)
        except:
            continue

# ❌ 问题2：固定间隔重试，退化严重
def get_user_profile(user_id: str):
    for attempt in range(5):
        try:
            return call_service(user_id)
        except:
            time.sleep(2)  # 每次都等2秒，高并发时堆积
            continue
```

### 正例（指数退避 + 抖动）

```python
# ✅ 正确：指数退避 + 随机抖动（Exponential Backoff with Jitter）
import random
import time
import logging
from functools import wraps
from typing import Type, Tuple

logger = logging.getLogger(__name__)


def retry_with_backoff(
    max_attempts: int = 3,
    base_delay: float = 0.5,       # 基础延迟：0.5秒
    max_delay: float = 10.0,       # 最大延迟：10秒
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
    jitter: bool = True             # 是否加随机抖动
):
    """
    重试装饰器：指数退避 + 随机抖动

    退避序列（加抖动）：
      第1次: 0.5 + random(0~0.5) ≈ 0.5~1s
      第2次: 1.0 + random(0~1)   ≈ 1~2s
      第3次: 2.0 + random(0~2)   ≈ 2~4s
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_attempts:
                        logger.error(
                            f"重试{attempt}次全部失败: {func.__name__}: {e}"
                        )
                        raise

                    delay = min(base_delay * (2 ** (attempt - 1)), max_delay)
                    if jitter:
                        delay = delay * (0.5 + random.random())

                    logger.warning(
                        f"第{attempt}次尝试失败: {func.__name__}, "
                        f"{delay:.2f}s后重试... 错误: {e}"
                    )
                    time.sleep(delay)
        return wrapper
    return decorator


# ============ 使用示例 ============
class RecommendationService:
    """推荐服务：带重试的客户端封装"""

    def __init__(self, base_url: str):
        self.base_url = base_url

    @retry_with_backoff(
        max_attempts=3,
        base_delay=0.5,
        max_delay=8.0,
        exceptions=(ConnectionError, TimeoutError,),
    )
    def get_recall_candidates(self, user_id: str, count: int = 100):
        """获取召回候选集，网络异常自动重试"""
        response = requests.get(
            f"{self.base_url}/recall/{user_id}",
            params={"count": count},
            timeout=(2, 5)
        )
        response.raise_for_status()
        return response.json()

    @retry_with_backoff(
        max_attempts=3,
        base_delay=1.0,
        exceptions=(requests.exceptions.RequestException,),
    )
    def get_ranking_scores(self, candidate_items: list):
        """获取排序分数，只对网络异常重试"""
        response = requests.post(
            f"{self.base_url}/rank",
            json={"item_ids": candidate_items},
            timeout=(3, 10)
        )
        if response.status_code >= 400 and response.status_code < 500:
            raise BusinessError(f"业务错误: {response.status_code}")
        response.raise_for_status()
        return response.json()
```

```typescript
interface RetryOptions {
  maxAttempts?: number;
  baseDelay?: number;
  maxDelay?: number;
  shouldRetry?: (error: unknown) => boolean;
}

async function retryWithBackoff<T>(
  fn: () => Promise<T>,
  options: RetryOptions = {}
): Promise<T> {
  const {
    maxAttempts = 3,
    baseDelay = 500,
    maxDelay = 8000,
    shouldRetry = () => true,
  } = options;

  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt === maxAttempts || !shouldRetry(error)) throw error;

      // 指数退避 + 随机抖动
      const delay = Math.min(baseDelay * 2 ** (attempt - 1), maxDelay);
      const jitter = delay * (0.5 + Math.random() * 0.5);
      const waitMs = Math.min(jitter, maxDelay);

      console.warn(`第${attempt}次失败，${waitMs.toFixed(0)}ms后重试...`);
      await new Promise((r) => setTimeout(r, waitMs));
    }
  }
  throw new Error('重试耗尽');
}

// 使用示例
async function getRecallCandidates(userId: string, count: number = 100) {
  return retryWithBackoff(
    async () => {
      const response = await fetch(
        `http://recommend-svc/recall/${userId}?count=${count}`,
        { signal: AbortSignal.timeout(5000) }
      );
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      return response.json() as Promise<RecallResult>;
    },
    {
      maxAttempts: 3,
      baseDelay: 500,
      shouldRetry: (e) => e instanceof TypeError || (e as any)?.name === 'TimeoutError',
    }
  );
}
```

### 推荐系统的重试策略

| 场景 | 重试策略 | 原因 |
|------|---------|------|
| 用户特征服务 | 立即重试1次，超时降级 | 实时性要求高 |
| 模型推理服务 | 不重试，快速降级 | 推理耗时固定，重试无意义 |
| 缓存预热 | 指数退避重试3次 | 异步任务，可容忍延迟 |
| Kafka消费 | 不重试，用幂等性保证 | 防止重复处理 |
| 外部API（微信/支付） | 不重试，业务层处理 | 幂等性无法保证 |

## 三、熔断器（Circuit Breaker）

### 为什么需要熔断？

假设推荐服务调用用户画像服务，用户画像服务挂了：
- 第1个请求超时 → 等
- 第100个请求超时 → 全部等 → 线程池耗尽 → **推荐服务也挂 → 级联失败**

熔断器的核心思想：**快速失败，保留资源，给下游恢复的机会**

### 熔断器状态机

```
    ┌───────┐  失败率>阈值     ┌──────────┐  恢复超时   ┌────────┐
    │CLOSED│ ──────────────▶ │   OPEN   │ ─────────▶ │ HALF-  │
    │ 正常  │                │   熔断中  │            │  OPEN  │
    └───────┘  ◀────────────── └──────────┘            │  半开   │
     ▲      探测成功              ▲                     └────────┘
     │      连续成功N次           │                         │
     └────────────────────────────┘                         │ 成功
          熔断失败率达标，自动打开                            │ 失败
                                                          ▼
                                                        (OPEN)
```

| 状态 | 行为 | 适用场景 |
|------|------|---------|
| **CLOSED**（关闭） | 正常请求通过，失败计入计数器 | 系统正常运行 |
| **OPEN**（打开） | 所有请求直接失败（快速返回），不调用下游 | 下游已确认故障 |
| **HALF_OPEN**（半开） | 允许部分探测请求通过，探测下游是否恢复 | 试探性恢复 |

### 完整熔断器实现（Python）

```python
# ✅ 熔断器完整实现
import time
import threading
from enum import Enum
from functools import wraps
from typing import Callable, Any, Optional


class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"


class CircuitBreaker:
    """
    熔断器实现

    参数:
        failure_threshold: 失败多少次后打开熔断（默认: 5）
        success_threshold: 半开状态下成功多少次后关闭（默认: 2）
        timeout: 熔断打开后的恢复探测时间（秒，默认: 30）
    """

    def __init__(
        self,
        name: str = "default",
        failure_threshold: int = 5,
        success_threshold: int = 2,
        timeout: float = 30.0,
    ):
        self.name = name
        self.failure_threshold = failure_threshold
        self.success_threshold = success_threshold
        self.timeout = timeout
        self._state = CircuitState.CLOSED
        self._failure_count = 0
        self._success_count = 0
        self._last_failure_time: Optional[float] = None
        self._lock = threading.Lock()

    @property
    def state(self) -> CircuitState:
        with self._lock:
            if self._state == CircuitState.OPEN:
                if time.time() - self._last_failure_time >= self.timeout:
                    self._state = CircuitState.HALF_OPEN
                    self._success_count = 0
            return self._state

    def record_success(self):
        with self._lock:
            self._failure_count = 0
            if self._state == CircuitState.HALF_OPEN:
                self._success_count += 1
                if self._success_count >= self.success_threshold:
                    self._state = CircuitState.CLOSED
                    print(f"[CircuitBreaker {self.name}] ✅ 熔断关闭，恢复正常")

    def record_failure(self):
        with self._lock:
            self._failure_count += 1
            self._last_failure_time = time.time()
            if self._state == CircuitState.HALF_OPEN:
                self._state = CircuitState.OPEN
                print(f"[CircuitBreaker {self.name}] ⚠️ 半开探测失败，重新打开熔断")
            elif self._failure_count >= self.failure_threshold:
                self._state = CircuitState.OPEN
                print(f"[CircuitBreaker {self.name}] 🔴 失败{self._failure_count}次，打开熔断")

    def call(self, func: Callable, *args, **kwargs) -> Any:
        if self.state == CircuitState.OPEN:
            raise CircuitOpenError(
                f"CircuitBreaker[{self.name}] 处于OPEN状态，拒绝请求"
            )
        try:
            result = func(*args, **kwargs)
            self.record_success()
            return result
        except Exception as e:
            self.record_failure()
            raise


class CircuitOpenError(Exception):
    """熔断器打开时抛出的异常"""
    pass


def circuit_breaker(
    name: str = "default",
    failure_threshold: int = 5,
    success_threshold: int = 2,
    timeout: float = 30.0,
    fallback: Callable = None,
):
    """熔断器装饰器，fallback为降级函数"""
    breaker = CircuitBreaker(name, failure_threshold, success_threshold, timeout)

    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return breaker.call(func, *args, **kwargs)
            except CircuitOpenError:
                if fallback:
                    return fallback(*args, **kwargs)
                raise
        wrapper.circuit_breaker = breaker
        return wrapper
    return decorator


# ============ 推荐系统使用示例 ============
class UserProfileService:
    """用户画像服务（带熔断保护）"""

    def __init__(self):
        self._breaker = CircuitBreaker(
            name="UserProfileService",
            failure_threshold=5,
            success_threshold=2,
            timeout=30.0,
        )

    def get_user_features(self, user_id: str) -> dict:
        """获取用户特征，熔断保护"""
        def _fetch():
            return requests.get(
                f"http://user-profile-svc/api/{user_id}/features",
                timeout=(1, 3)
            ).json()

        def _fallback(user_id: str) -> dict:
            """降级方案：返回默认特征（历史统计值）"""
            print(f"[降级] 用户画像服务不可用，返回默认特征: user_id={user_id}")
            return {
                "user_id": user_id,
                "age_group": "unknown",
                "gender": "unknown",
                "city_level": 2,
                "preference_tags": [],
                "_source": "fallback",
                "_fallback_time": time.time(),
            }

        try:
            return self._breaker.call(_fetch)
        except CircuitOpenError:
            return _fallback(user_id)
```

## 四、限流（Rate Limiting）

### 为什么需要限流？

- 防止突发流量压垮服务
- 防止恶意爬虫/刷接口
- 保护下游依赖不被击穿

### 限流算法对比

| 算法 | 原理 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|---------|
| **固定窗口** | 固定时间段内限制请求数 | 简单 | 边界突发问题 | 简单场景 |
| **滑动窗口** | 滚动时间窗口 | 平滑 | 实现稍复杂 | 推荐系统入口 |
| **令牌桶** | 以固定速率放入令牌，取令牌处理 | 支持突发 | - | API限流 |
| **漏桶** | 以固定速率消费请求 | 平滑限流 | 不支持突发 | 支付/订单 |

### 滑动窗口限流器（Redis + Lua）

```python
# ✅ 滑动窗口限流器（Redis + Lua 保证原子性）
import redis
import time


class SlidingWindowRateLimiter:
    """
    滑动窗口限流器
    支持按 user_id / ip / api_key 限流
    """

    def __init__(self, redis_client: redis.Redis, key_prefix: str = "rl"):
        self.redis = redis_client
        self.key_prefix = key_prefix

    def is_allowed(
        self,
        identifier: str,
        max_requests: int,
        window_seconds: int,
    ) -> dict:
        """
        检查是否允许请求
        """
        key = f"{self.key_prefix}:{identifier}"
        now = time.time()
        window_start = now - window_seconds

        # Lua脚本：原子性执行滑动窗口计数
        lua_script = """
        local key = KEYS[1]
        local now = tonumber(ARGV[1])
        local window_start = tonumber(ARGV[2])
        local max_requests = tonumber(ARGV[3])
        local window_seconds = tonumber(ARGV[4])

        redis.call('ZREMRANGEBYSCORE', key, 0, window_start)
        local current = redis.call('ZCARD', key)

        if current < max_requests then
            redis.call('ZADD', key, now, now .. '-' .. math.random())
            redis.call('EXPIRE', key, window_seconds)
            return {1, max_requests - current - 1, window_seconds}
        else
            return {0, 0, window_seconds}
        end
        """

        result = self.redis.eval(
            lua_script, 1, key, now, window_start, max_requests, window_seconds,
        )
        return {
            "allowed": bool(result[0]),
            "remaining": int(result[1]),
            "reset_at": time.time() + int(result[2]),
            "limit": max_requests,
        }
```

### Flask 中间件集成

```python
from flask import Flask, request, jsonify, g
from functools import wraps
import redis

app = Flask(__name__)
redis_client = redis.Redis(host='localhost', port=6379, decode_responses=True)
limiter = SlidingWindowRateLimiter(redis_client)


def rate_limit(max_requests: int, window_seconds: int):
    """限流装饰器"""
    def decorator(f):
        @wraps(f)
        def wrapper(*args, **kwargs):
            identifier = getattr(g, 'user_id', request.remote_addr or 'anonymous')
            result = limiter.is_allowed(identifier, max_requests, window_seconds)

            headers = {
                "X-RateLimit-Limit": str(result["limit"]),
                "X-RateLimit-Remaining": str(result["remaining"]),
                "X-RateLimit-Reset": str(int(result["reset_at"])),
            }

            if not result["allowed"]:
                return jsonify({
                    "error": "rate_limit_exceeded",
                    "message": "请求过于频繁，请稍后重试",
                    "retry_after": int(result['reset_at'] - time.time()),
                }), 429, headers

            response = f(*args, **kwargs)
            if isinstance(response, tuple) and len(response) == 2:
                return response[0], response[1], headers
            return response, 200, headers
        return wrapper
    return decorator


# ============ 推荐系统限流示例 ============
@app.route("/api/recommend/<user_id>")
@rate_limit(max_requests=100, window_seconds=60)
def recommend(user_id: str):
    """推荐接口：每用户每分钟100次"""
    items = recommendation_service.get_items(user_id)
    return {"user_id": user_id, "items": items}


@app.route("/api/recall/<user_id>")
@rate_limit(max_requests=10, window_seconds=60)
def recall(user_id: str):

    candidates = recall_service.get_candidates(user_id)
    return {"user_id": user_id, "candidates": candidates}
```

## 五、综合实战：推荐系统全链路韧性

```python
# ✅ 完整示例：推荐服务全链路韧性设计
import time
import requests
from circuit_breaker import CircuitBreaker, CircuitOpenError
from sliding_window import SlidingWindowRateLimiter
from retry import retry_with_backoff
import redis


class RecommendationService:
    """
    推荐服务：综合应用超时 + 重试 + 熔断 + 限流

    推荐系统典型调用链：
    API → 限流 → 用户特征(熔断) → 召回(熔断) → 排序(熔断) → 重排 → 返回
    """

    def __init__(self):
        self.redis_client = redis.Redis(host='localhost', port=6379)
        self.limiter = SlidingWindowRateLimiter(self.redis_client)

        # 各下游服务的熔断器（独立配置）
        self.user_profile_breaker = CircuitBreaker(
            name="user_profile", failure_threshold=5, timeout=30.0,
        )
        self.recall_breaker = CircuitBreaker(
            name="recall_service", failure_threshold=3, timeout=15.0,
        )
        self.rank_breaker = CircuitBreaker(
            name="rank_service", failure_threshold=3, timeout=10.0,
        )

    # ===== 降级策略 =====
    def _get_fallback_features(self, user_id: str) -> dict:
        return {"user_id": user_id, "tags": ["fallback"], "age_group": 3}

    def _get_fallback_recall(self, user_id: str) -> list:
        return [{"item_id": "hot_1", "score": 0.9}, {"item_id": "hot_2", "score": 0.8}]

    def _get_fallback_rank(self, candidates: list) -> list:
        return candidates[:10]

    # ===== 核心推荐逻辑 =====
    def recommend(self, user_id: str, scene: str = "home", count: int = 20) -> dict:
        """
        推荐主流程

        故障处理策略：
        1. 超时：各服务独立设置，超时后触发降级
        2. 重试：召回层支持指数退避重试
        3. 熔断：下游任一服务故障不影响其他服务
        4. 降级：每层都有降级兜底方案
        """
        start_time = time.time()
        fallback_stages = []

        # Step 1: 限流检查
        limit_result = self.limiter.is_allowed(
            identifier=f"user:{user_id}", max_requests=100, window_seconds=60,
        )
        if not limit_result["allowed"]:
            return {"error": "rate_limited", "retry_after": 30}

        try:
            # Step 2: 获取用户特征（熔断保护）
            try:
                features = self.user_profile_breaker.call(
                    lambda: requests.get(
                        f"http://profile-svc/{user_id}", timeout=(1, 3)
                    ).json()
                )
            except CircuitOpenError:
                features = self._get_fallback_features(user_id)
                fallback_stages.append("user_profile")

            # Step 3: 多路召回（重试 + 熔断保护）
            try:
                candidates = self._multi_recall(user_id, features, count * 3)
            except (CircuitOpenError, Exception):
                candidates = self._get_fallback_recall(user_id)
                fallback_stages.append("recall")

            # Step 4: 粗排过滤（独立超时）
            candidates = self._coarse_filter(candidates, scene, timeout=5)

            # Step 5: 精排（熔断保护）
            try:
                ranked = self.rank_breaker.call(
                    lambda: requests.post(
                        "http://rank-svc/batch_score",
                        json={"items": candidates, "user_features": features},
                        timeout=(2, 8)
                    ).json()
                )
            except CircuitOpenError:
                ranked = self._get_fallback_rank(candidates)
                fallback_stages.append("rank")

            # Step 6: 重排（业务规则过滤）
            final_items = self._rerank(ranked, scene, count)

            return {
                "user_id": user_id,
                "scene": scene,
                "items": final_items,
                "metrics": {
                    "latency_ms": int((time.time() - start_time) * 1000),
                    "fallback_stages": fallback_stages,
                },
            }

        except Exception as e:
            logger.error(f"推荐服务异常", exc_info=True)
            return self._emergency_fallback(user_id)

    @retry_with_backoff(
        max_attempts=2, base_delay=0.3,
        exceptions=(ConnectionError, TimeoutError)
    )
    def _multi_recall(self, user_id: str, features: dict, count: int) -> list:
        """多路召回：支持重试"""
        response = requests.get(
            f"http://recall-svc/{user_id}/candidates",
            params={"count": count, "features": features},
            timeout=(1, 5)
        )
        response.raise_for_status()
        return response.json()["candidates"]

    def _coarse_filter(self, items: list, scene: str, timeout: int) -> list:
        """粗排过滤，带超时保护"""
        response = requests.post(
            "http://coarse-svc/filter",
            json={"items": items, "scene": scene},
            timeout=timeout
        )
        response.raise_for_status()
        return response.json()["filtered"]

    def _rerank(self, items: list, scene: str, count: int) -> list:
        """重排：多样性打散 + 业务规则"""
        response = requests.post(
            "http://rerank-svc/reorder",
            json={"items": items, "scene": scene, "count": count},
            timeout=(1, 3)
        )
        if response.status_code == 200:
            return response.json()["reordered"]
        return items[:count]

    def _emergency_fallback(self, user_id: str) -> dict:
        """紧急降级：兜底返回热门推荐"""
        return {
            "user_id": user_id,
            "items": self._get_fallback_recall(user_id),
            "source": "emergency_fallback",
        }
```

## 六、推荐系统三级降级方案

```
正常流程（所有服务可用）
└─ 用户特征 → 多路召回 → 粗排过滤 → 精排 → 重排

── 一级降级（精排服务挂了）──
└─ 用户特征 → 多路召回 → 粗排过滤 → [降级] 直接返回粗排分数
  → 用户感知：推荐质量下降，但功能可用

── 二级降级（召回服务挂了）──
└─ 用户特征 → [降级] 返回热门候选集 → 粗排 → 精排 → 重排
  → 用户感知：推荐内容大众化，无个性化

── 三级降级（整个推荐链路挂了）──
└─ [降级] 直接返回编辑推荐的热门内容
  → 用户感知：看到的是热门内容，系统仍在运行
```

## 常见误区

### 误区1：超时设为 0 或无限
> ❌ `timeout=None` = 无限等待 = 系统hang死的元凶

### 误区2：重试次数过多
> ❌ 重试3次 = 基础延迟1s × 7 = 最大等待7秒，用户早就放弃了
> ✅ 重试1-2次 + 指数退避，用户感知不明显

### 误区3：所有异常都重试
> ❌ `except Exception` → 业务异常也重试 → 雪上加霜
> ✅ 只对**网络异常**重试，业务异常立即返回

### 误区4：熔断阈值设得太宽松
> ❌ 失败100次才熔断 → 线程池早就耗尽了
> ✅ 失败3-5次就熔断，快速失败是关键

### 误区5：限流后不告诉用户
> ❌ 返回 500 错误，用户莫名其妙
> ✅ 返回 429 + `Retry-After` 头，告知用户何时重试

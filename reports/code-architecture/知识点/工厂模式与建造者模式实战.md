# 工厂模式与建造者模式实战

## 概念解释

### 工厂模式：把"创建"封装起来

**解决的问题**：对象创建散落各处，改一个类名要改 N 个地方。

**核心思想**：不要直接 `new Something()`，而是调用工厂函数/类，让创建逻辑集中在一个地方。

**三种形态**：
- **简单工厂**：一个函数根据参数返回不同子类（最常用，最易理解）
- **工厂方法**：父类定义工厂接口，子类决定创建哪个类
- **抽象工厂**：创建一系列相关对象，而不用指定具体类

### 建造者模式：一步一步构建复杂对象

**解决的问题**：构造函数参数太多、可选参数混乱、对象创建步骤不清晰。

**核心思想**：把"如何组装"从"最终产品"中分离出来，通过链式调用逐步构建。

**典型特征**：
- `builder.设置A().设置B().设置C().build()`
- 有 `Director` 时：Director 负责定义标准组装流程

### 正例（改进后）：建造者模式

```python
# ✅ 建造者：每个参数都有 setter，最后一步 build()
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class ModelConfig:
    # 必填字段
    name: str
    version: str
    model_type: str
    # 可选字段，有默认值
    embedding_dim: int = 64
    learning_rate: float = 0.001
    batch_size: int = 256
    epochs: int = 10
    optimizer: str = "adam"
    regularization: float = 0.0
    feature_store_url: Optional[str] = None
    use_gpu: bool = True
    early_stopping: bool = False
    log_dir: Optional[str] = None
    tags: list[str] = field(default_factory=list)


class ModelConfigBuilder:
    """建造者：链式设置参数，最后 build() 生成不可变对象"""

    def __init__(self, name: str, version: str, model_type: str):
        self._name = name
        self._version = version
        self._model_type = model_type
        # 可选参数默认值在 build 时统一填充
        self._embedding_dim = 64
        self._learning_rate = 0.001
        self._batch_size = 256
        self._epochs = 10
        self._optimizer = "adam"
        self._regularization = 0.0
        self._feature_store_url = None
        self._use_gpu = True
        self._early_stopping = False
        self._log_dir = None
        self._tags = []

    # 链式调用接口
    def embedding_dim(self, value: int) -> "ModelConfigBuilder":
        self._embedding_dim = value
        return self

    def learning_rate(self, value: float) -> "ModelConfigBuilder":
        self._learning_rate = value
        return self

    def batch_size(self, value: int) -> "ModelConfigBuilder":
        self._batch_size = value
        return self

    def epochs(self, value: int) -> "ModelConfigBuilder":
        self._epochs = value
        return self

    def use_gpu(self, enabled: bool) -> "ModelConfigBuilder":
        self._use_gpu = enabled
        return self

    def tags(self, *tags: str) -> "ModelConfigBuilder":
        self._tags = list(tags)
        return self

    def build(self) -> ModelConfig:
        """最终校验 + 生成不可变配置"""
        if self._embedding_dim <= 0:
            raise ValueError("embedding_dim must be positive")
        return ModelConfig(
            name=self._name,
            version=self._version,
            model_type=self._model_type,
            embedding_dim=self._embedding_dim,
            learning_rate=self._learning_rate,
            batch_size=self._batch_size,
            epochs=self._epochs,
            optimizer=self._optimizer,
            regularization=self._regularization,
            feature_store_url=self._feature_store_url,
            use_gpu=self._use_gpu,
            early_stopping=self._early_stopping,
            log_dir=self._log_dir,
            tags=self._tags,
        )


# 调用方：代码即文档，可读性极强
config = (
    ModelConfigBuilder("DIN", "v2.1", "deep Interest")
    .embedding_dim(128)
    .learning_rate(0.0005)
    .batch_size(1024)
    .epochs(20)
    .use_gpu(True)
    .tags("ctr", "e-commerce")
    .build()
)
```

### 简单工厂 vs 建造者：适用场景对比

| 场景 | 模式 | 说明 |
|------|------|------|
| 创建统一接口的不同算法实例 | 简单工厂 | 根据字符串返回不同推荐算法对象 |
| 构建复杂配置对象 | 建造者 | 模型超参、ETL 管道配置、API 请求体 |
| 创建一族相关对象 | 抽象工厂 | 创建不同数据库的 DAO + Repository 组合 |
| 需要 Fluent API（链式调用） | 建造者 | 几乎所有复杂对象都适合 |

### 简单工厂实战：推荐算法工厂

```python
# ✅ 简单工厂：将创建逻辑集中在一个地方
from abc import ABC, abstractmethod


class Recommender(ABC):
    @abstractmethod
    def recommend(self, user_id: str, top_k: int) -> list[str]:
        """返回 top_k 个 item id"""
        pass


class CFRecommender(Recommender):
    """协同过滤"""
    def recommend(self, user_id: str, top_k: int) -> list[str]:
        return [f"item_{i}" for i in range(top_k)]


class DeepRecommender(Recommender):
    """深度学习推荐模型"""
    def __init__(self, model_path: str):
        self.model_path = model_path

    def recommend(self, user_id: str, top_k: int) -> list[str]:
        return [f"deep_item_{i}" for i in range(top_k)]


class TwoTowerRecommender(Recommender):
    """双塔模型"""
    def __init__(self, embedding_dim: int = 128):
        self.embedding_dim = embedding_dim

    def recommend(self, user_id: str, top_k: int) -> list[str]:
        return [f"two_tower_item_{i}" for i in range(top_k)]


class RecommenderFactory:
    """简单工厂：注册式，支持扩展"""
    _registry: dict[str, type[Recommender]] = {}

    @classmethod
    def register(cls, name: str, recommender_cls: type[Recommender]):
        cls._registry[name] = recommender_cls

    @classmethod
    def create(cls, name: str, **kwargs) -> Recommender:
        if name not in cls._registry:
            available = ", ".join(cls._registry.keys())
            raise ValueError(f"Unknown recommender '{name}'. Available: {available}")
        return cls._registry[name](**kwargs)


# 注册推荐算法
RecommenderFactory.register("cf", CFRecommender)
RecommenderFactory.register("deep", DeepRecommender)
RecommenderFactory.register("two_tower", TwoTowerRecommender)

# 创建实例：业务代码完全不依赖具体类
rec: Recommender = RecommenderFactory.create("deep", model_path="/models/din_v2.pt")
results = rec.recommend(user_id="u123", top_k=10)
```

## TypeScript 建造者模式

```typescript
// ✅ TypeScript 建造者：同样适用链式调用
interface QueryFilter {
  field: string;
  operator: "eq" | "gt" | "lt" | "in" | "contains";
  value: unknown;
}

interface QueryConfig {
  index: string;
  filters: QueryFilter[];
  pageSize: number;
  sortBy?: string;
  highlight?: boolean;
}

class QueryBuilder {
  private config: Partial<QueryConfig> = {
    filters: [],
    pageSize: 20,
  };

  constructor(index: string) {
    this.config.index = index;
  }

  filter(field: string, operator: QueryFilter["operator"], value: unknown): this {
    this.config.filters!.push({ field, operator, value });
    return this;
  }

  pageSize(size: number): this {
    this.config.pageSize = size;
    return this;
  }

  sortBy(field: string): this {
    this.config.sortBy = field;
    return this;
  }

  highlight(): this {
    this.config.highlight = true;
    return this;
  }

  build(): QueryConfig {
    if (!this.config.index) throw new Error("index is required");
    return this.config as QueryConfig;
  }
}

// 使用
const query = new QueryBuilder("user_behavior")
  .filter("gender", "eq", "female")
  .filter("age", "in", [18, 25, 30])
  .filter("tags", "contains", "美妆")
  .pageSize(50)
  .sortBy("last_active")
  .highlight()
  .build();
```

## 常见误区

- **误区1：所有类都要建造者**。只有参数多（>4个）+ 可选参数混杂时才值得用。2-3个参数的类直接用构造函数即可。
- **误区2：建造者中放业务逻辑**。建造者只负责对象组装，构建逻辑与业务逻辑要严格分离。
- **误区3：工厂和建造者混用**。工厂管"创建哪个类"，建造者管"如何配置这个类"，职责不同，不要混在一起。
- **误区4：简单工厂越来越大**。当工厂类里出现大量 if/elif，要及时改造为"注册式工厂"（如上例），避免修改工厂代码。

## 适用场景

- **模型训练配置**：超参多、类型复杂、需要验证和默认值
- **推荐管道配置**：特征提取 → 排序 → 重排，多步骤配置
- **数据库连接配置**：host/port/credentials/poolSize/timeout 等参数
- **API 请求构建**：URL 参数、Header、Body 的组装
- **日志/监控配置**：多个开关、阈值、采样率的组合

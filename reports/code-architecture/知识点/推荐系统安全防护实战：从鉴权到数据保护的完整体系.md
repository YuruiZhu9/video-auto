## 一、推荐系统安全全景：威胁模型

### 1.1 为什么推荐系统更容易成为攻击目标？

普通 Web 应用攻击的是"能不能进来"，推荐系统攻击的是"进去能拿什么"。

```
┌──────────────────────────────────────────────────────────────┐
│                  推荐系统独特攻击面                            │
├──────────────────────────────────────────────────────────────┤
│  ① 特征数据 → 用户行为偏好 / 商业机密（谁对什么感兴趣）         │
│  ② 模型能力 → 召回/排序黑盒（算法工程师的核心资产）            │
│  ③ 实验结果 → A/B Test数据（商业决策依据）                    │
│  ④  Embedding → 用户/商品向量（可反推相似用户/商品）           │
│  ⑤  实时信号 → 点击流/购买流（实时竞品分析）                   │
└──────────────────────────────────────────────────────────────┘
```

### 1.2 推荐系统威胁矩阵

| 威胁类型 | 攻击目标 | 后果 | 推荐系统案例 |
|---------|---------|------|------------|
| **特征数据窃取** | 用户画像/Embedding | 隐私泄露、竞品获取 | 通过大量请求反推用户特征 |
| **AB实验数据窃取** | 实验组/对照组流量 | 商业机密泄露 | 抓取实验组用户ID分析策略 |
| **模型逆向攻击** | 排序模型权重/特征 | 模型能力泄露 | 梯度攻击反推训练数据 |
| **缓存投毒** | Redis推荐缓存 | 推荐结果被污染 | 写入恶意推荐结果 |
| **特征注入攻击** | 用户行为特征 | 模型预测被操纵 | 伪造点击行为操纵推荐 |
| **接口滥用** | 推荐API | 资源损耗/数据爬取 | 爬虫批量抓取商品推荐 |
| **模型服务DoS** | 推理服务 | 推荐服务不可用 | 构造特殊请求耗尽GPU |
| **向量数据库攻击** | Milvus/Qdrant | Embedding数据泄露 | 未授权查询向量索引 |

### 2.1 JWT 鉴权：超越"能验证"的安全实践

上一份文档讲了 JWT 的基本实现，这里补充**生产级安全加固**。

#### 反例 vs 正例对比

```python
# ❌ 问题1：算法未指定，依赖库自动选择（RS256 vs HS256 安全差异巨大）
# ❌ 问题2：Token 永不过期
# ❌ 问题3：Payload 包含敏感信息
def create_token_basic(user_id: int) -> str:
    payload = {"user_id": user_id, "username": "alice"}
    return jwt.encode(payload, SECRET_KEY, algorithm="HS256")


# ✅ 生产级 JWT 实现
import jwt
import hashlib
import time
import secrets
from dataclasses import dataclass
from typing import Optional

@dataclass
class JWTConfig:
    secret_key: str
    algorithm: str = "HS256"          # 显式指定，不依赖默认值
    access_token_expire_minutes: int = 15
    refresh_token_expire_days: int = 7
    issuer: str = "recommend-service"
    audience: str = "recommend-api"


class SecureJWTSerializer:
    """生产级 JWT 序列化器"""

    def __init__(self, config: JWTConfig):
        self.config = config

    def create_access_token(
        self,
        user_id: int,
        scene: str,                   # 业务场景（首页/详情页/搜索）
        scopes: list[str],            # 权限范围
        extra_claims: Optional[dict] = None
    ) -> str:
        now = int(time.time())
        payload = {
            "iss": self.config.issuer,
            "aud": self.config.audience,
            "iat": now,                          # 签发时间
            "exp": now + self.config.access_token_expire_minutes * 60,  # 过期时间
            "nbf": now,                          # 生效时间（防止"未来签发"攻击）
            "jti": self._generate_jti(),         # 唯一ID，防重放
            "sub": str(user_id),                 # 不含敏感信息
            "scene": scene,
            "scopes": scopes,
        }
        if extra_claims:
            payload.update(extra_claims)
        return jwt.encode(payload, self.config.secret_key, algorithm=self.config.algorithm)

    def _generate_jti(self) -> str:
        return secrets.token_urlsafe(32)

    def verify_token(self, token: str) -> dict:
        """验签"""
        try:
            payload = jwt.decode(
                token,
                self.config.secret_key,
                algorithms=[self.config.algorithm],  # 白名单算法，防算法混淆攻击
                issuer=self.config.issuer,
                audience=self.config.audience,
                options={
                    "require": ["exp", "iat", "jti", "sub", "scopes"],
                    "verify_exp": True,
                    "verify_iat": True,
                    "verify_nbf": True,
                    "verify_iss": True,
                    "verify_aud": True,
                }
            )
            return payload
        except jwt.ExpiredSignatureError:
            raise TokenExpiredError("Token 已过期")
        except jwt.InvalidAudienceError:
            raise TokenInvalidError("Token 受众不匹配")
        except jwt.InvalidIssuerError:
            raise TokenInvalidError("Token 签发者不匹配")
        except jwt.InvalidSignatureError:
            raise TokenInvalidError("Token 签名无效")
        except Exception as e:
            raise TokenInvalidError(f"Token 验证失败: {e}")


class RefreshTokenStore:
    """Refresh Token 黑名单存储（Redis）"""

    def __init__(self, redis_client):
        self.redis = redis_client

    def store(self, user_id: int, refresh_token: str, ttl_seconds: int):
        key = f"refresh_token:{user_id}:{self._hash_token(refresh_token)}"
        self.redis.setex(key, ttl_seconds, "valid")

    def is_valid(self, user_id: int, refresh_token: str) -> bool:
        key = f"refresh_token:{user_id}:{self._hash_token(refresh_token)}"
        return self.redis.exists(key) == 1

    def revoke(self, user_id: int, refresh_token: str):
        key = f"refresh_token:{user_id}:{self._hash_token(refresh_token)}"
        self.redis.delete(key)

    def revoke_all(self, user_id: int):
        """撤销用户所有 Refresh Token（密码修改/账号异常时）"""
        pattern = f"refresh_token:{user_id}:*"
        for key in self.redis.scan_iter(pattern):
            self.redis.delete(key)

    @staticmethod
    def _hash_token(token: str) -> str:
        """Token 哈希存储，防明文泄露"""
        return hashlib.sha256(token.encode()).hexdigest()
```

### 2.2 防重放攻击：Nonce + 时间戳双重机制

JWT 只能防止伪造，但不能防止**重放攻击**（攻击者截获合法 Token 后反复使用）。

```python
import time

class NonceStore:
    """Nonce 黑名单：记录已使用的 Token jti，防重放"""

    def __init__(self, redis_client, window_seconds: int = 900):
        self.redis = redis_client
        self.window_seconds = window_seconds

    def is_replay(self, jti: str) -> bool:
        key = f"nonce:{jti}"
        if self.redis.exists(key):
            return True
        self.redis.setex(key, self.window_seconds, "used")
        return False


class RequestTimestampValidator:
    """时间戳验证：防止"时间隧道"攻击"""

    def __init__(self, max_age_seconds: int = 300):
        self.max_age = max_age_seconds

    def validate(self, request_timestamp: int) -> bool:
        now = int(time.time())
        return abs(now - request_timestamp) <= self.max_age


# ── 完整的安全中间件 ─────────────────────────────────────────
from fastapi import Request, HTTPException
from starlette.middleware.base import BaseHTTPMiddleware

class RecommendSecurityMiddleware(BaseHTTPMiddleware):
    """推荐系统安全中间件：限流 + JWT验签 + Nonce防重放 + 时间戳"""

    def __init__(self, app, jwt_serializer, nonce_store, rate_limiter):
        super().__init__(app)
        self.jwt_serializer = jwt_serializer
        self.nonce_store = nonce_store
        self.rate_limiter = rate_limiter

    async def dispatch(self, request: Request, call_next):
        client_ip = request.client.host
        # Step 1：限流（IP + UserID 双维度）
        if not self.rate_limiter.allow(client_ip, limit=100, window=60):
            raise HTTPException(429, "请求过于频繁")

        token = self._extract_token(request)
        if not token:
            raise HTTPException(401, "缺少认证令牌")

        # Step 2：JWT 验签
        payload = self.jwt_serializer.verify_token(token)

        # Step 3：Nonce 检查（防重放）
        if self.nonce_store.is_replay(payload["jti"]):
            raise HTTPException(401, "Token 已被使用")

        # Step 4：时间戳检查（防时间隧道）
        ts = request.headers.get("X-Request-Timestamp")
        if ts:
            validator = RequestTimestampValidator(max_age_seconds=300)
            if not validator.validate(int(ts)):
                raise HTTPException(401, "请求时间戳无效")

        request.state.user_id = payload["sub"]
        request.state.scopes = payload["scopes"]
        request.state.scene = payload["scene"]
        return await call_next(request)

    def _extract_token(self, request: Request) -> str | None:
        auth = request.headers.get("Authorization", "")
        if auth.startswith("Bearer "):
            return auth[7:]
        return None
```

## 三、第二层：纵深防护——特征数据保护

### 3.1 ABAC：超越 RBAC 的细粒度权限控制

RBAC（基于角色）只控制"谁能做什么"，ABAC（基于属性）可以控制"在什么条件下对什么资源做什么"。

推荐系统的 ABAC 典型场景：
- 管理员可以查看所有特征，普通算法工程师只能查看自己团队的
- 实验期内的 A/B Test 特征数据对外"不可见"
- 付费客户可以获取更多推荐结果，白嫖用户受限
- Embedding 向量仅限内网访问，禁止导出

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional, Callable

class ResourceType(Enum):
    USER_PROFILE = "user_profile"           # 用户画像
    ITEM_FEATURE = "item_feature"          # 商品特征
    RECOMMEND_RESULT = "recommend_result"   # 推荐结果
    AB_EXPERIMENT = "ab_experiment"          # AB实验数据
    MODEL_ARTIFACT = "model_artifact"       # 模型文件
    EMBEDDING_VECTOR = "embedding_vector"    # Embedding向量

class Action(Enum):
    READ = "read"
    WRITE = "write"
    DELETE = "delete"
    EXPORT = "export"        # 导出权限（最敏感）

@dataclass
class Subject:
    user_id: str
    roles: list[str]
    department: str
    team: str
    security_level: int  # 1-5，涉密等级

@dataclass
class Resource:
    type: ResourceType
    owner_id: Optional[str] = None
    sensitivity_level: int = 1
    experiment_id: Optional[str] = None
    team: Optional[str] = None

@dataclass
class Environment:
    ip_address: str
    request_scene: str
    time_window: str  # working_hours / off_hours
    is_internal: bool

@dataclass
class ABACPolicy:
    name: str
    description: str
    condition: Callable
    allowed: bool

class ABACEngine:
    """ABAC 权限引擎"""

    def __init__(self):
        self.policies: list[ABACPolicy] = []
        self._register_default_policies()

    def _register_default_policies(self):
        # 策略1：用户画像按团队隔离
        self.policies.append(ABACPolicy(
            name="user_profile_team_isolation",
            description="用户画像按团队隔离，非本团队成员不可访问",
            condition=lambda s, r, e: (
                r.type == ResourceType.USER_PROFILE and s.team == r.team
            ),
            allowed=True
        ))

        # 策略2：AB实验数据仅实验参与团队可见
        self.policies.append(ABACPolicy(
            name="ab_experiment_confidentiality",
            description="实验期数据仅实验负责人可见",
            condition=lambda s, r, e: (
                r.type == ResourceType.AB_EXPERIMENT
                and r.experiment_id is not None
                and self._is_experiment_team_member(s, r.experiment_id)
            ),
            allowed=True
        ))

        # 策略3：导出权限需高级安全等级
        self.policies.append(ABACPolicy(
            name="export_requires_high_security",
            description="特征导出需 security_level >= 4",
            condition=lambda s, r, e: (
                e.action == Action.EXPORT and s.security_level >= 4
            ),
            allowed=True
        ))

        # 策略4：推荐结果按用户等级分级
        self.policies.append(ABACPolicy(
            name="recommend_result_tiered_access",
            description="推荐结果按用户等级开放不同粒度",
            condition=lambda s, r, e: (
                r.type == ResourceType.RECOMMEND_RESULT
                and s.roles[0] in ["vip_user", "internal_debug"]
            ),
            allowed=True
        ))

        # 策略5：非工作时间仅管理员可访问模型文件
        self.policies.append(ABACPolicy(
            name="model_artifact_off_hours",
            description="非工作时间禁止下载模型文件",
            condition=lambda s, r, e: (
                r.type == ResourceType.MODEL_ARTIFACT
                and ("admin" in s.roles or e.time_window == "working_hours")
            ),
            allowed=True
        ))

        # 策略6：Embedding 向量仅内部访问
        self.policies.append(ABACPolicy(
            name="embedding_internal_only",
            description="Embedding 向量仅限内网访问",
            condition=lambda s, r, e: (
                r.type == ResourceType.EMBEDDING_VECTOR and e.is_internal
            ),
            allowed=True
        ))

        # 默认拒绝（兜底）
        self.policies.append(ABACPolicy(
            name="default_deny",
            description="未匹配任何策略则默认拒绝",
            condition=lambda s, r, e: True,
            allowed=False
        ))

    def authorize(
        self,
        subject: Subject,
        resource: Resource,
        environment: Environment,
        action: Action = Action.READ
    ) -> bool:
        for policy in self.policies:
            if policy.condition(subject, resource, environment):
                return policy.allowed
        return False

    def _is_experiment_team_member(self, subject: Subject, exp_id: str) -> bool:
        return True  # 实际应查 Redis 实验成员列表
```

### 3.2 实验隔离：防止 A/B 实验数据泄露

A/B 实验数据是推荐系统最敏感的商业机密之一。

```python
from dataclasses import dataclass
from enum import Enum
import time
from typing import Optional
import numpy as np

class ExperimentStage(Enum):
    RUNNING = "running"        # 实验中（严格保密）
    CONCLUDED = "concluded"    # 已结束（可公开）

@dataclass
class Experiment:
    id: str
    name: str
    stage: ExperimentStage
    owner_team: str
    start_time: int
    end_time: Optional[int] = None

class ExperimentDataIsolation:
    """
    实验数据隔离策略：
    1. 实验期间，实验组用户ID不暴露给 API 响应
    2. 实验期间，实验相关指标不返回给普通用户
    """

    def __init__(self, redis_client):
        self.redis = redis_client

    def mask_experiment_in_response(
        self,
        user_id: str,
        recommend_results: list[dict],
        include_experiment_info: bool = False
    ) -> list[dict]:
        """
        在推荐结果中隐藏实验相关信息
        防止通过批量请求反推实验策略
        """
        masked_results = []
        for item in recommend_results:
            masked_item = {
                "item_id": item["item_id"],
                "score": item["score"],
                "position": item["position"],
                "experiment_group": item.get("experiment_group") if include_experiment_info else None,
                "strategy_id": item.get("strategy_id") if include_experiment_info else None,
            }
            masked_results.append(masked_item)
        return masked_results

    def check_experiment_access(self, user_id: str, experiment_id: str) -> bool:
        key = f"experiment_member:{experiment_id}"
        return self.redis.sismember(key, user_id)


class TrafficAntiTheft:
    """流量防盗：防止竞争对手通过大量请求分析推荐策略"""

    def __init__(self, redis_client):
        self.redis = redis_client

    def detect_anomalous_access(
        self,
        user_id: str,
        time_window_seconds: int = 3600
    ) -> dict:
        """
        检测异常访问模式：
        - 短时间内大量请求（爬虫特征）
        - 请求覆盖大量不同用户（数据抓取特征）
        - 固定时间间隔请求（脚本特征）
        """
        now = time.time()
        window_start = now - time_window_seconds

        freq_key = f"access_freq:{user_id}"
        request_count = self.redis.zcount(freq_key, window_start, now)

        users_key = f"queried_users:{user_id}"
        unique_users = self.redis.scard(users_key)

        timestamps_key = f"access_timestamps:{user_id}"
        timestamps = self.redis.lrange(timestamps_key, -20, -1)
        interval_variance = self._calculate_interval_variance(timestamps)

        return {
            "request_count": request_count,
            "unique_users_queried": unique_users,
            "interval_variance": interval_variance,
            "is_suspicious": (
                request_count > 1000
                or unique_users > 500
                or interval_variance < 0.1  # 时间间隔几乎相同 → 脚本
            )
        }

    @staticmethod
    def _calculate_interval_variance(timestamps: list) -> float:
        if len(timestamps) < 2:
            return 1.0
        intervals = []
        for i in range(1, len(timestamps)):
            try:
                t1 = float(timestamps[i-1])
                t2 = float(timestamps[i])
                intervals.append(t2 - t1)
            except (ValueError, TypeError):
                continue
        if not intervals:
            return 1.0
        mean = sum(intervals) / len(intervals)
        variance = sum((x - mean) ** 2 for x in intervals) / len(intervals)
        return variance
```

### 4.1 GDPR / PIPL 合规：推荐系统的特殊义务

推荐系统处理的个人数据种类多、量大，是隐私合规的重点领域：

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional
from datetime import datetime

class DataCategory(Enum):
    PERSONAL_IDENTIFICATION = "个人身份信息"
    BEHAVIORAL_DATA = "行为数据"
    PREFERENCE_DATA = "偏好数据"
    LOCATION_DATA = "位置数据"
    DEVICE_DATA = "设备数据"

@dataclass
class ConsentRecord:
    user_id: str
    consent_type: str           # 数据处理目的
    granted: bool
    timestamp: int
    withdrawal_timestamp: Optional[int] = None
    legal_basis: str = "consent"

class RecommendSystemPrivacyEngine:
    """
    推荐系统隐私引擎：
    满足 GDPR / PIPL 合规要求
    """

    def __init__(self, consent_store, data_repository):
        self.consent_store = consent_store
        self.data_repo = data_repository

    def check_consent(self, user_id: str, purpose: str) -> bool:
        record = self.consent_store.get(user_id, purpose)
        if not record:
            return False
        return record.granted and record.withdrawal_timestamp is None

    def apply_right_to_access(self, user_id: str) -> dict:
        """数据访问权：用户查看自己被收集的所有数据"""
        user_data = self.data_repo.get_all_user_data(user_id)
        return {
            "user_id": user_id,
            "profile": self._sanitize_profile(user_data.get("profile", {})),
            "preferences": user_data.get("preferences", []),
            "behavior_history": self._sanitize_behavior(user_data.get("history", [])),
            "consent_records": self._get_consent_summary(user_id),
            "data_categories": list(user_data.keys()),
        }

    def apply_right_to_erasure(self, user_id: str) -> dict:
        """
        删除权：用户要求删除个人数据
        推荐系统的特殊挑战：
        - 用户画像：匿名化处理（而非完全删除，保持推荐能力）
        - 行为记录：完全删除（无统计价值，用户隐私优先）
        """
        deleted_items = []

        # 行为日志：完全删除
        self.data_repo.delete_behavior_data(user_id)
        deleted_items.append("behavior_history")

        # 用户画像：匿名化处理（k-匿名化，保留统计价值）
        self.data_repo.anonymize_profile(user_id, k=50)
        deleted_items.append("profile (anonymized)")

        # 推荐结果缓存：立即删除
        self.data_repo.delete_cached_recommendations(user_id)
        deleted_items.append("cached_recommendations")

        # 特征向量：匿名化
        self.data_repo.anonymize_embeddings(user_id)
        deleted_items.append("embeddings (anonymized)")

        return {
            "user_id": user_id,
            "deleted_items": deleted_items,
            "retention_note": "监管要求的最低保留期限数据除外",
            "request_id": self._generate_request_id()
        }

    def apply_right_to_portability(self, user_id: str) -> dict:
        """可携带权：导出用户数据"""
        user_data = self.apply_right_to_access(user_id)
        return {
            "export_format": "JSON",
            "generated_at": int(datetime.now().timestamp()),
            "data": user_data,
            "metadata": {
                "data_controller": "推荐系统服务",
                "contact": "privacy@company.com",
                "purpose": "用户数据可携带权行使"
            }
        }

    def _sanitize_profile(self, profile: dict) -> dict:
        """脱敏用户画像"""
        return {
            "age_range": profile.get("age_range"),     # 保留年龄段，不保留精确年龄
            "city_level": profile.get("city_level"),  # 不保留具体城市
            "interest_tags": profile.get("interest_tags"),
        }

    def _sanitize_behavior(self, history: list) -> list:

        for event in history:

                "event_type": event["event_type"],
```

### 4.2 差分隐私 + 联邦学习

差分隐私可以在保护个体隐私的同时，提供群体行为统计；联邦学习则让数据不出本地设备：

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class PrivacyBudget:
    epsilon: float        # 隐私预算，越小越隐私（推荐系统典型值：1.0~10.0）
    delta: float           # 失败概率上限（典型值：1e-5）
    sensitivity: float    # 敏感度（单用户数据改变时输出变化的最大值）


class LaplaceNoiseMechanism:
    """拉普拉斯机制：向数值结果添加拉普拉斯噪声"""

    @staticmethod
    def add_noise(true_value: float, sensitivity: float, epsilon: float) -> float:
        """
        ε-差分隐私保证：单用户数据变化对输出分布的影响被限制在 e^ε
        """
        scale = sensitivity / epsilon
        noise = np.random.laplace(0, scale)
        return max(0, true_value + noise)

    @staticmethod
    def privacy_preserving_count(true_count: int, epsilon: float = 2.0) -> float:
        """带差分隐私保护的计数查询"""
        sensitivity = 1
        return LaplaceNoiseMechanism.add_noise(true_count, sensitivity, epsilon)


class FederatedLearningAggregator:
    """
    联邦学习聚合器：用户行为数据不出本地设备，只上传梯度
    适用于：多方联合建模（如跨APP用户行为联合推荐）
    """

    def __init__(self, epsilon: float = 5.0, clip_threshold: float = 5.0):
        self.epsilon = epsilon
        self.clip_threshold = clip_threshold
        self.client_updates: list[dict] = []
        self.privacy_spent: float = 0.0

    def receive_gradient_update(
        self,
        user_id: str,
        gradient: dict,
        sample_count: int
    ) -> float:
        """接收单个客户端的梯度更新"""
        clipped_gradient = self._clip_gradient(gradient)
        noisy_gradient = self._add_gaussian_noise(clipped_gradient, sample_count)

        self.client_updates.append({
            "user_id": user_id,
            "gradient": noisy_gradient,
            "sample_count": sample_count
        })

        privacy_cost = self._compute_privacy_cost(sample_count)
        self.privacy_spent += privacy_cost
        return privacy_cost

    def aggregate(self) -> dict:
        """聚合所有客户端更新，返回全局梯度"""
        total_samples = sum(u["sample_count"] for u in self.client_updates)
        aggregated = {}

        for key in self.client_updates[0]["gradient"].keys():
            weighted_sum = sum(
                u["gradient"][key] * u["sample_count"]
                for u in self.client_updates
            )
            aggregated[key] = weighted_sum / total_samples

        self.client_updates.clear()
        return aggregated

    def _clip_gradient(self, gradient: dict) -> dict:
        """梯度裁剪：限制每个参数的 L2 范数"""
        l2_norm = np.sqrt(sum(np.sum(np.array(g) ** 2) for g in gradient.values()))
        if l2_norm > self.clip_threshold:
            scale = self.clip_threshold / l2_norm
            return {k: np.array(g) * scale for k, g in gradient.items()}
        return gradient

    def _add_gaussian_noise(self, gradient: dict, sample_count: int) -> dict:
        """添加高斯噪声（用于 (ε, δ)-差分隐私）"""
        noise_std = self.clip_threshold * np.sqrt(2 * np.log(1.25 / 1e-5)) / self.epsilon
        noise_std_per_sample = noise_std / np.sqrt(sample_count)
        return {
            k: np.array(g) + np.random.normal(0, noise_std_per_sample, np.array(g).shape)
            for k, g in gradient.items()
        }

    def _compute_privacy_cost(self, sample_count: int) -> float:
        return 1.0 / sample_count


class EmbeddingPrivacyProtection:
    """Embedding 隐私保护：防止通过向量反推个体信息"""

    @staticmethod
    def add_embedding_noise(
        embedding: list[float],
        epsilon: float = 3.0,
        sensitivity: float = 1.0
    ) -> list[float:
        """向 Embedding 添加噪声（差分隐私保护）"""
        scale = sensitivity / epsilon
        noise = np.random.laplace(0, scale, len(embedding))
        return [max(-1.0, min(1.0, e + n)) for e, n in zip(embedding, noise)]

    @staticmethod
    def check_embedding_similarity_attack(
        embedding_a: list[float],
        embedding_b: list[float],
        threshold: float = 0.99
    ) -> bool:
        """
        检测 Embedding 相似度攻击：
        如果两个 Embedding 余弦相似度极高，存在隐私风险
        （例如：同一用户两个时间点的向量几乎相同 → 数据可关联）
        """
        ea = np.array(embedding_a)
        eb = np.array(embedding_b)
        cos_sim = np.dot(ea, eb) / (np.linalg.norm(ea) * np.linalg.norm(eb))
        return cos_sim >= threshold
```

### 5.1 输入过滤：防止 Prompt 注入

推荐系统引入大模型（LLM）生成推荐理由或结果排序时，面临 Prompt 注入攻击：

```python
import re
from dataclasses import dataclass
from enum import Enum

class InjectionType(Enum):
    PROMPT_INJECTION = "prompt_injection"
    CONTEXT_MANIPULATION = "context_manip"
    PRIVACY_PROBING = "privacy_probing"

@dataclass
class InjectionResult:
    detected: bool
    injection_type: InjectionType | None
    risk_score: float
    sanitized_input: str

class PromptInjectionDetector:
    """
    Prompt 注入检测器：
    防止用户输入操纵推荐系统（大模型部分）
    """

    # 典型的 Prompt 注入模式
    INJECTION_PATTERNS = [
        re.compile(r"ignore\s+(previous|above|all)\s+instructions", re.I),
        re.compile(r"forget\s+(everything|all|what)", re.I),
        re.compile(r"#\s*system\s*|#\s*instruction\s*", re.I),
        re.compile(r"act\s+as\s+(?:a|an)\s+different", re.I),
        re.compile(r"you\s+are\s+now\s+", re.I),
        re.compile(r"\[INST\]\s*.*\[/INST\]", re.I),  # 微调模型注入标记
        re.compile(r"{{.*}}"),  # 模板注入
        re.compile(r"<\|.*\|>", re.I),  # 分隔符注入
    ]

    # 隐私探测模式
    PRIVACY_PATTERNS = [
        re.compile(r"show\s+(?:me\s+)?all\s+(?:your|the)\s+(?:users?|profiles?)", re.I),
        re.compile(r"export\s+(?:all\s+)?training\s+data", re.I),
        re.compile(r"what\s+(?:is|are)\s+the\s+(?:other|current)\s+recommendations", re.I),
    ]

    def detect(self, user_input: str) -> InjectionResult:
        """检测 Prompt 注入"""
        user_input_lower = user_input.lower()
        detected_type = None
        risk_score = 0.0

        # 检查注入模式
        for pattern in self.INJECTION_PATTERNS:
            if pattern.search(user_input):
                detected_type = InjectionType.PROMPT_INJECTION
                risk_score = max(risk_score, 0.8)
                break

        # 检查隐私探测
        for pattern in self.PRIVACY_PATTERNS:
            if pattern.search(user_input):
                detected_type = InjectionType.PRIVACY_PROBING
                risk_score = max(risk_score, 0.9)
                break

        # 检查上下文字符异常（Unicode 同形攻击）
        if self._has_homoglyph_attack(user_input):
            detected_type = InjectionType.CONTEXT_MANIPULATION
            risk_score = max(risk_score, 0.7)

        # 脱敏处理
        sanitized = self._sanitize(user_input, detected_type)

        return InjectionResult(
            detected=risk_score > 0.5,
            injection_type=detected_type,
            risk_score=risk_score,
            sanitized_input=sanitized
        )

    def _sanitize(self, text: str, injection_type: InjectionType | None) -> str:
        """脱敏处理：移除注入内容"""
        if injection_type == InjectionType.PROMPT_INJECTION:
            # 移除可疑的指令片段
            for pattern in self.INJECTION_PATTERNS:
                text = pattern.sub("[filtered]", text)
        return text.strip()

    @staticmethod
    def _has_homoglyph_attack(text: str) -> bool:
        """检测 Unicode 同形攻击（用相似字符替换）"""
        suspicious_chars = [
            '\u0430',  # 西里尔 а（替代拉丁 a）
            '\u0435',  # 西里尔 е（替代拉丁 e）
            '\u043e',  # 西里尔 о（替代拉丁 o）
        ]
        return any(c in text for c in suspicious_chars)
```

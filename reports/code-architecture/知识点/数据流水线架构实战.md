# 数据流水线架构实战

## 概念解释

**什么是数据流水线？** 将数据从原始状态，经过一系列处理步骤，转化为最终可用形式的全流程。

**推荐系统中的典型流水线**：

```
用户行为日志（原始）
  ↓ 采集
Kafka 消息队列
  ↓ 消费
特征工程（ETL）
  ↓ 写入
FeatureStore / Redis / HBase
  ↓ 读取
模型训练 / 在线推理
  ↓ 输出
推荐结果
  ↓ 排序、重排
前端展示
```

**为什么需要流水线架构？**
- 数据处理步骤多，任何一步都可能出错
- 步骤之间有依赖关系
- 需要监控每一步的吞吐量、延迟、异常
- 需要支持断点重跑、数据回填

---

## 核心模式：管道-过滤器（Pipeline-Filter）

**Unix/Linux 管道哲学**：`cat file | grep "ERROR" | sort | uniq -c`

**软件工程对应**：数据流经一系列处理器，每一步只做一件事。

```
原始数据 → FilterA → FilterB → FilterC → 最终结果
```

**设计原则**：
- 每个 Filter 独立、无状态（理想情况）
- Filter 之间通过标准格式传递（JSON / Protobuf / DataFrame）
- 新增 Filter 不影响现有流程（开闭原则）

## 代码示例

### 反例（坏味道）：大函数式 ETL

```python
# ❌ 单个函数处理所有数据转换，违反单一职责
def process_user_behavior(raw_data: list[dict]) -> list[dict]:
    results = []
    for record in raw_data:
        # 1. 数据清洗
        if not record.get("user_id") or not record.get("item_id"):
            continue
        # 2. 类型转换
        record["timestamp"] = int(record["timestamp"])
        record["behavior"] = str(record["behavior"]).lower()
        # 3. 数据标准化
        if record["behavior"] == "click":
            record["weight"] = 1.0
        elif record["behavior"] == "buy":
            record["weight"] = 5.0
        elif record["behavior"] == "fav":
            record["weight"] = 3.0
        else:
            record["weight"] = 0.5
        # 4. 字段提取
        record["date"] = datetime.fromtimestamp(record["timestamp"]).strftime("%Y-%m-%d")
        # 5. 数据增强
        record["hour"] = datetime.fromtimestamp(record["timestamp"]).hour
        # 6. 过滤异常
        if record["timestamp"] < 0 or record["hour"] < 0:
            continue
        results.append(record)
    return results
# 问题：想单独调试"清洗"步骤？改一行试试？很难。
```

### 正例（改进后）：管道-过滤器架构

```python
from abc import ABC, abstractmethod
from typing import TypeVar, Generic
from dataclasses import dataclass
import json

T = TypeVar("T")
R = TypeVar("R")


class Pipeline(Generic[T]):
    """数据流水线，支持任意 Filter 组合"""

    def __init__(self):
        self._filters: list[Callable[[list[T]], list[T]]] = []

    def add(self, filter_fn: Callable[[list[T]], list[T]]) -> "Pipeline[T]":
        self._filters.append(filter_fn)
        return self

    def execute(self, data: list[T]) -> list[T]:
        """顺序执行所有过滤器"""
        result = data
        for filter_fn in self._filters:
            result = filter_fn(result)
            if not result:  # 早退：数据流中断时停止
                break
        return result


@dataclass
class UserBehavior:
    user_id: str
    item_id: str
    behavior: str  # click / buy / fav / view
    timestamp: int
    weight: float = 0.0
    date: str = ""
    hour: int = 0


class DataFrame:
    """流水线中的数据结构，类似 pandas 但简化"""
    def __init__(self, data: list[dict]):
        self._data = data

    def filter(self, fn: Callable[[dict], bool]) -> "DataFrame":
        return DataFrame([r for r in self._data if fn(r)])

    def map(self, fn: Callable[[dict], dict]) -> "DataFrame":
        return DataFrame([fn(r) for r in self._data])

    def to_list(self) -> list[dict]:
        return self._data


# ====== 具体 Filter 实现（每个只做一件事）======

def filter_invalid_records(records: list[dict]) -> list[dict]:
    """Filter 1: 过滤无效记录"""
    return [
        r for r in records
        if r.get("user_id") and r.get("item_id") and r.get("timestamp")
    ]


def normalize_behavior(records: list[dict]) -> list[dict]:
    """Filter 2: 行为类型标准化"""
    weight_map = {"click": 1.0, "buy": 5.0, "fav": 3.0, "view": 0.5}
    for r in records:
        behavior = str(r.get("behavior", "")).lower()
        r["behavior"] = behavior
        r["weight"] = weight_map.get(behavior, 0.5)
    return records


def add_time_features(records: list[dict]) -> list[dict]:
    """Filter 3: 添加时间特征"""
    for r in records:
        ts = r["timestamp"]
        dt = datetime.fromtimestamp(ts)
        r["date"] = dt.strftime("%Y-%m-%d")
        r["hour"] = dt.hour
        r["weekday"] = dt.weekday()
    return records


def filter_suspicious(records: list[dict]) -> list[dict]:
    """Filter 4: 过滤可疑数据（刷单、爬虫等）"""
    suspicious_users = load_suspicious_user_set()  # 从外部加载
    suspicious_items = load_suspicious_item_set()
    return [
        r for r in records
        if r["user_id"] not in suspicious_users
        and r["item_id"] not in suspicious_items
        and 0 <= r.get("hour", -1) <= 23
    ]


def enrich_item_features(records: list[dict]) -> list[dict]:
    """Filter 5: 关联商品维度信息"""
    item_db = load_item_database()  # 外部数据源
    for r in records:
        item_info = item_db.get(r["item_id"], {})
        r["category"] = item_info.get("category", "unknown")
        r["price"] = item_info.get("price", 0.0)
        r["brand"] = item_info.get("brand", "unknown")
    return records


# ====== 灵活组合：构建不同场景的 Pipeline ======

def build_etl_pipeline(mode: str = "train") -> Pipeline[dict]:
    """工厂方法：根据场景构建不同的流水线"""
    p = Pipeline()

    if mode == "train":
        # 训练模式：全量数据，包含所有 Filter
        p.add(filter_invalid_records) \
         .add(normalize_behavior) \
         .add(add_time_features) \
         .add(filter_suspicious) \
         .add(enrich_item_features)
    elif mode == "online":
        # 在线模式：跳过耗时的外部数据关联
        p.add(filter_invalid_records) \
         .add(normalize_behavior) \
         .add(add_time_features)
    elif mode == "debug":
        # 调试模式：每步单独输出，方便查问题
        p.add(filter_invalid_records)

    return p


# ====== 执行流水线 ======
raw_logs = load_raw_logs("/data/behavior_2026_04_01.jsonl")
pipeline = build_etl_pipeline(mode="train")
clean_data = pipeline.execute(raw_logs)
print(f"原始记录数: {len(raw_logs)}, 清洗后: {len(clean_data)}")
```

## 带错误处理和监控的进阶版

```python
import logging
from dataclasses import dataclass, field
from typing import Callable, Any
from datetime import datetime

logger = logging.getLogger(__name__)


@dataclass
class ProcessingStats:
    """每步处理统计"""
    stage: str
    input_count: int = 0
    output_count: int = 0
    errors: int = 0
    start_time: datetime = field(default_factory=datetime.now)

    def duration_ms(self) -> float:
        return (datetime.now() - self.start_time).total_seconds() * 1000


class MonitoredFilter:
    """带监控的 Filter 包装器"""

    def __init__(self, name: str, fn: Callable[[list[T]], list[T]]):
        self.name = name
        self.fn = fn

    def __call__(self, data: list[T]) -> list[T]:
        stats = ProcessingStats(stage=self.name, input_count=len(data))
        try:
            result = self.fn(data)
            stats.output_count = len(result)
            logger.info(
                f"[{self.name}] in={stats.input_count}, "
                f"out={stats.output_count}, "
                f"dropped={stats.input_count - stats.output_count}, "
                f"time={stats.duration_ms():.1f}ms"
            )
            return result
        except Exception as e:
            stats.errors = 1
            logger.error(f"[{self.name}] FAILED: {e}")
            # 根据策略决定：抛出异常 or 返回空列表
            return []


class ResilientPipeline:
    """带重试 + 监控的数据流水线"""

    def __init__(self, max_retries: int = 3):
        self._filters: list[MonitoredFilter] = []
        self._max_retries = max_retries

    def add(self, name: str, fn: Callable[[list[T]], list[T]]) -> "ResilientPipeline":
        self._filters.append(MonitoredFilter(name, fn))
        return self

    def execute(self, data: list[T]) -> tuple[list[T], list[ProcessingStats]]:
        """返回 (处理后数据, 各步骤统计)"""
        result = data
        all_stats: list[ProcessingStats] = []

        for filter_wrapper in self._filters:
            if not result:  # 空数据直接短路
                break

            result = filter_wrapper(result)
            # 从日志提取 stats（简化版，实际可用结构化传递）
            # 实际项目中建议每个 Filter 直接返回 (result, stats)

        return result, all_stats


# 使用
pipeline = (
    ResilientPipeline(max_retries=3)
    .add("invalid_filter", filter_invalid_records)
    .add("normalize", normalize_behavior)
    .add("time_features", add_time_features)
    .add("suspicious_filter", filter_suspicious)
    .add("item_enrich", enrich_item_features)
)

result, stats = pipeline.execute(raw_logs)
```

## 适用场景

| 场景 | 说明 |
|------|------|
| **推荐系统特征工程** | 原始日志 → 清洗 → 特征提取 → 聚合 → 写入 FeatureStore |
| **数据同步管道** | MySQL → Binlog → Kafka → Flink → ClickHouse |
| **模型推理管道** | 请求 → 参数校验 → 特征查询 → 模型推理 → 结果排序 → 返回 |
| **ETL 作业** | S3 文件 → Lambda/DataProc → 清洗 → 转换 → 入库 |
| **日志处理** | 服务日志 → 采集 → 解析 → 过滤 ERROR → 告警 |

## 常见误区

- **误区1：Filter 之间共享状态**。每个 Filter 应该是独立的，共享状态会导致难以调试和并行化。
- **误区2：一条流水线处理所有场景**。训练、推理、debug 场景需求不同，用工厂模式构建不同 Pipeline。
- **误区3：没有监控，不知道哪一步慢了**。用 `MonitoredFilter` 包装每个步骤，量化每步耗时和数据量。
- **误区4：异常吞掉不处理**。数据流水线中异常要有明确策略：重试？跳过？告警？不要默默 return []。

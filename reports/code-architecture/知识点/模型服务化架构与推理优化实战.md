# 模型服务化架构与推理优化实战

> 推荐系统的"最后一公里"：如何把训练好的模型高效、稳定地部署为在线服务

---

## 概念解释

训练模型是起点，让模型在生产环境稳定服务才是终点。模型服务化要解决的核心问题：

1. **吞吐量**：QPS 够不够高
2. **延迟**：P99 能不能压到 200ms 以内
3. **稳定性**：模型挂了怎么办
4. **可迭代**：怎么灰度 A/B 测试新模型

---

## 代码示例

### 反例（裸 Flask 直接加载模型）

```python
# ❌ 问题代码：所有请求共用一个模型实例，无并发，无降级
from flask import Flask, request, jsonify
import pickle
import numpy as np

app = Flask(__name__)

# 启动时加载模型（阻塞 + 无异常处理）
model = pickle.load(open('model.pkl', 'rb'))

@app.route('/recommend', methods=['POST'])
def recommend():
    user_id = request.json['user_id']
    item_ids = request.json['item_ids']

    # 每个请求串行推理，无批处理
    scores = [model.predict(user_id, item_id) for item_id in item_ids]

    return jsonify({'scores': scores})

# 问题：
# 1. 单线程串行，QPS 极低
# 2. 模型推理慢（每个请求都调一遍）
# 3. 模型崩溃 → 整个服务崩溃
# 4. 无法灰度切换模型
# 5. 无法动态扩缩容
```

### 正例一：基础批处理推理服务（Python）

```python
# ✅ 改进：FastAPI + 批量推理 + 进程隔离
import asyncio
import numpy as np
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from concurrent.futures import ProcessPoolExecutor
import multiprocessing as mp
import pickle
import os

app = FastAPI()

# 模型加载（进程池，隔离加载）
_executor = ProcessPoolExecutor(max_workers=4)
_model = None

def _load_model():
    global _model
    _model = pickle.load(open(os.getenv('MODEL_PATH', 'model.pkl'), 'rb'))
    return _model

class RecommendRequest(BaseModel):
    user_id: int
    item_ids: list[int]

class RecommendResponse(BaseModel):
    scores: list[float]
    model_version: str
    latency_ms: float

@app.on_event("startup")
async def startup():
    # 启动时预加载模型
    _load_model()
    print(f"模型加载完成，版本: {os.getenv('MODEL_VERSION', 'unknown')}")

@app.post("/recommend", response_model=RecommendResponse)
async def recommend(req: RecommendRequest):
    """批量推理接口"""
    import time
    start = time.perf_counter()

    # 异步批量推理（真实场景用 Ray/Triton）
    loop = asyncio.get_event_loop()
    scores = await loop.run_in_executor(
        _executor,
        _predict_batch,
        req.user_id,
        req.item_ids
    )

    return RecommendResponse(
        scores=scores,
        model_version=os.getenv('MODEL_VERSION', 'v1.0'),
        latency_ms=(time.perf_counter() - start) * 1000
    )

def _predict_batch(user_id: int, item_ids: list[int]) -> list[float]:
```

```python
# Python Client：推荐引擎调用 Triton
import tritonclient.http as httpclient
from tritonclient.utils import np_to_triton_dtype
import numpy as np

class TritonRecommender:
    """Triton 推理客户端封装"""

    def __init__(self, url: str = "localhost:8000"):
        self.client = httpclient.InferenceServerClient(url=url)
        self.model_recall = "recall_model"
        self.model_ranking = "ranking_model"

    def batch_recall(self, user_ids: list[int], top_k: int = 100) -> dict[int, list[int]]:
        """批量召回（GPU 并行）"""
        # 构造批量输入
        user_tensor = np.array(user_ids, dtype=np.int32).reshape(-1, 1)

        inputs = [
            httpclient.InferInput('user_ids', user_tensor.shape, np_to_triton_dtype(user_tensor.dtype)),
        ]
        inputs[0].set_data_from_numpy(user_tensor)

        outputs = [
            httpclient.InferRequestedOutput('item_ids', binary_data=True),
            httpclient.InferRequestedOutput('scores', binary_data=True),
        ]

        response = self.client.infer(self.model_recall, inputs, outputs=outputs)

        result = {}
        for i, uid in enumerate(user_ids):
            items = response.as_numpy('item_ids')[i][:top_k].tolist()
            result[uid] = items
        return result

    def batch_ranking(self, user_ids: list[int], item_ids: list[list[int]]) -> dict[int, list[tuple[int, float]]]:
        """批量排序（精排）"""
        # flatten 便于批量处理
        flat_user = np.array([u for u, items in zip(user_ids, item_ids) for _ in items])
        flat_item = np.array([item for items in item_ids for item in items])

        u_tensor = flat_user.reshape(-1, 1).astype(np.int32)
        i_tensor = flat_item.reshape(-1, 1).astype(np.int32)

        inputs = [
            httpclient.InferInput('user_ids', u_tensor.shape, np_to_triton_dtype(u_tensor.dtype)),
            httpclient.InferInput('item_ids', i_tensor.shape, np_to_triton_dtype(i_tensor.dtype)),
        ]
        inputs[0].set_data_from_numpy(u_tensor)
        inputs[1].set_data_from_numpy(i_tensor)

        outputs = [httpclient.InferRequestedOutput('scores', binary_data=True)]
        response = self.client.infer(self.model_ranking, inputs, outputs=outputs)

        scores = response.as_numpy('scores')
        result = {}
        idx = 0
        for u, items in zip(user_ids, item_ids):
            n = len(items)
            score_pairs = list(zip(item_ids[idx:idx+n], scores[idx:idx+n].tolist()))
            score_pairs.sort(key=lambda x: x[1], reverse=True)
            result[u] = score_pairs
            idx += n
        return result
```

### 正例三：Ray Serve 在线学习架构

```python
# ✅ Ray Serve：推荐系统在线学习 + 实时更新
import ray
from ray import serve
import numpy as np
import pickle
import time

ray.init(address="auto")
serve.start(detached=True)

@serve.deployment(
    num_replicas=2,
    ray_actor_options={"num_gpus": 0.5},
    max_concurrent_queries=100,
)
class OnlineLearningRecommender:
    """
    在线学习推荐模型：
    - 实时接收用户反馈
    - 增量更新模型参数（无需全量重训练）
    - 指数衰减旧数据（Recency Bias）
    """

    def __init__(self):
        self.model = self._load_init_model()
        self.update_count = 0
        self.last_update = time.time()

    def _load_init_model(self):
        # 启动时加载基础模型
        try:
            return pickle.load(open('base_model.pkl', 'rb'))
        except Exception:
            return self._create_dummy_model()

    @serve.batch(max_batch_size=64, batch_wait_timeout_s=0.01)
    async def batch_predict(self, user_ids: list[int], item_ids: list[int]) -> list[float]:
        """批量预测（Ray Serve 自动批处理）"""
        scores = [self.model.predict(u, i) for u, i in zip(user_ids, item_ids)]
        return scores

    async def __call__(self, request):
        user_id = request.json['user_id']
        item_id = request.json['item_id']
        action = request.json.get('action')  # 'predict' | 'feedback'

        if action == 'feedback':
            # 在线学习：收到用户反馈后增量更新
            reward = request.json['reward']  # 1=点击, 0=未点击
            self._online_update(user_id, item_id, reward)
            return {"status": "updated", "update_count": self.update_count}
        else:
            score = await self.batch_predict([user_id], [item_id])
            return {"user_id": user_id, "item_id": item_id, "score": score[0]}

    def _online_update(self, user_id: int, item_id: int, reward: float):
        """
        增量更新策略：UCB / Thompson Sampling / 简单的滑动平均
        这里用滑动平均作为示例
        """
        # 简化：更新用户-物品偏好矩阵
        lr = 0.01  # 学习率
        decay = 0.99

        self.model.weights[user_id, item_id] = (
            decay * self.model.weights[user_id, item_id] + lr * reward
        )
        self.update_count += 1
        self.last_update = time.time()

        # 每 1000 次更新持久化一次（异步）
        if self.update_count % 1000 == 0:
            self._async_save()

    def _async_save(self):
        import asyncio
        asyncio.create_task(self._save_checkpoint())

    async def _save_checkpoint(self):
        # 真实场景：保存到 S3/HDFS
        pickle.dump(self.model, open('checkpoint.pkl', 'wb'))
```

### 正例四：降级策略（模型不可用时的兜底）

```python
# ✅ 三级降级策略：模型推理 → 缓存结果 → 静态规则兜底
from functools import lru_cache
import time

class RecommenderWithFallback:
    """推荐系统三级降级（模型优先，规则兜底）"""

    def __init__(self, triton_client, redis_client):
        self.triton = triton_client
        self.redis = redis_client
        self.fallback_rules = self._init_rules()

    def _init_rules(self):
        """静态兜底规则：热度榜 / 新品榜"""
        return {
            "popular": [1001, 1002, 1003, 1004, 1005],  # 热搜
            "newest": [2001, 2002, 2003, 2004, 2005],  # 新品
            "personalized_warm": [3001, 3002, 3003],   # 冷启动通用
        }

    def recommend(self, user_id: int, scene: str, count: int = 10) -> list[int]:
        """推荐主流程：缓存 → 模型推理 → 规则兜底"""
        cache_key = f"rec:{user_id}:{scene}"

        # 第一级：读缓存
        cached = self.redis.get(cache_key)
        if cached:
            return cached[:count]

        # 第二级：模型推理
        try:
            candidates = self._model_inference(user_id, count=count * 3)
            results = candidates[:count]

            # 异步写缓存（Write-Behind）
            self._async_cache_set(cache_key, results, ttl=300)

        except Exception as e:
            print(f"模型推理失败，降级到规则兜底: {e}")
            results = self._fallback_by_scene(scene, count)

        # 第三级：规则兜底
        if not results:
            results = self._fallback_by_scene(scene, count)

        return results

    def _model_inference(self, user_id: int, count: int) -> list[int]:
        """调用 Triton 模型推理"""
        # 简化：实际通过 triton_client.batch_recall 调用
        return list(range(1, count * 3 + 1))

    def _fallback_by_scene(self, scene: str, count: int) -> list[int]:
        """根据场景选择兜底规则"""
        rules = {
            "home": "popular",
            "detail": "newest",
            "search": "popular",
            "cold_start": "personalized_warm",
        }
        rule_key = rules.get(scene, "popular")
        return self.fallback_rules[rule_key][:count]

    def _async_cache_set(self, key: str, value: list, ttl: int):
        """异步写缓存（不阻塞主流程）"""
        import asyncio
        asyncio.create_task(self._do_cache_set(key, value, ttl))

    async def _do_cache_set(self, key: str, value: list, ttl: int):
        self.redis.setex(key, ttl, value)
```

## 常见误区

### ❌误区1：所有请求单独推理
一个请求10个候选物品，串行调用10次模型 → QPS极低
**正确**：批量推理，一次输入所有候选，最大化GPU利用率

### ❌误区2：模型和业务代码混在一起
```python
# 错误：直接在前端服务里写推理代码
app = Flask()
model = load_model()
@app.route('/predict'): model.predict(...)
```
**正确**：用 Triton/Seldon/Core 做模型推理，Bypass 服务只负责路由

### ❌误区3：模型更新需要重启服务
**正确**：Triton 支持模型版本热加载，通过 `model_version` 参数灰度切换

### ❌误区4：没有降级预案
**正确**：始终保留规则/热度兜底，模型不可用时优雅降级

### ❌误区5：GPU 显存估算错误
**正确**：评估阶段就测量好模型占用的显存（含 batch 处理峰值），防止 OOM

## 模型服务框架对比

| 框架 | 适用场景 | 优点 | 缺点 |
|------|----------|------|------|
| **Triton** | 推荐/广告/图像 | GPU批处理好，版本管理强，多框架支持 | 部署配置复杂 |
| **Ray Serve** | 在线学习/动态图 | 原生支持在线更新，Python优先 | 生态比Triton弱 |
| **vLLM** | LLM推理 | PagedAttention，吞吐极高 | 仅支持LLM |
| **TensorFlow Serving** | TensorFlow模型 | 成熟稳定 | 只支持TF |
| **BentoML** | 快速原型 | 易上手，多框架 | 生产级需配置 |

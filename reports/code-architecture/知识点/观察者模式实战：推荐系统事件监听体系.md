# 观察者模式实战：推荐系统事件监听体系

## 概念解释

**观察者模式（Observer Pattern）**：定义对象间一对多的依赖关系，当一个对象（Subject）状态改变时，所有依赖它的对象（Observer）都会自动收到通知并更新。

**核心思想**：发布-订阅（Pub/Sub），解耦事件的"发出者"和"处理者"。

```
┌─────────────────────────────────────────────────────────┐
│  Subject（被观察者）                                     │
│  ┌──────────────────────────────────┐                   │
│  │ 观察者列表：[O1, O2, O3, ...]    │                   │
│  │ attach(observer)                 │                   │
│  │ detach(observer)                 │ 通知              │
│  │ notify() ────────────────────────┼──► Observer1      │
│  └──────────────────────────────────┘              ──► Observer2
│                                                        ──► Observer3
└─────────────────────────────────────────────────────────┘
```

**为什么推荐系统需要它？**

用户点击一个商品 → 触发一系列连锁反应：
- 更新用户兴趣画像
- 刷新推荐缓存
- 上报 BI 数据分析
- 触发相似商品推送
- 更新实时 CTR 统计

不用观察者模式，这段逻辑会写成：
```python
# 一个 click 函数里塞进所有处理逻辑 —— 违反 SRP
def handle_click(user_id, item_id):
    update_user_profile(user_id, item_id)   # 画像服务
    invalidate_recommendation_cache(user_id) # 缓存服务
    send_to_bi_analytics(user_id, item_id)  # BI服务
    trigger_similar_item_push(user_id)      # 推送服务
    update_realtime_ctr(user_id, item_id)   # 统计服务
    # ...未来再加10个需求，这个函数就爆炸了
```

用观察者模式：每个服务独立注册，互不感知，Subject 只负责通知。

## 代码示例

### 基础实现（Python）

```python
from abc import ABC, abstractmethod
from typing import List
import logging

logger = logging.getLogger(__name__)


# ============================================================
# 1. 抽象层：定义观察者接口
# ============================================================
class UserEventObserver(ABC):
    """用户事件观察者抽象接口"""

    @abstractmethod
    def on_user_action(self, event: "UserActionEvent") -> None:
        """当用户行为事件发生时，自动调用此方法"""
        pass


# ============================================================
# 2. 事件定义：统一事件格式
# ============================================================
@dataclass
class UserActionEvent:
    """用户行为事件"""
    event_type: str          # "click" / "view" / "purchase" / "like"
    user_id: str
    item_id: str
    timestamp: float
    extra_data: dict = field(default_factory=dict)

    def __str__(self):
        return (f"[{self.event_type.upper()}] user={self.user_id} "
                f"item={self.item_id} @ {datetime.fromtimestamp(self.timestamp)}")


# ============================================================
# 3. Subject（被观察者）：事件发布中心
# ============================================================
class UserEventSubject:
    """用户事件发布中心（Subject）"""

    def __init__(self):
        self._observers: List[UserEventObserver] = []
        self._event_history: List[UserActionEvent] = []
        self._logger = logging.getLogger("UserEventSubject")

    def attach(self, observer: UserEventObserver) -> None:
        """注册观察者"""
        if observer not in self._observers:
            self._observers.append(observer)
            self._logger.info(f"✅ 观察者注册: {observer.__class__.__name__}")

    def detach(self, observer: UserEventObserver) -> None:
        """注销观察者"""
        if observer in self._observers:
            self._observers.remove(observer)
            self._logger.info(f"❌ 观察者注销: {observer.__class__.__name__}")

    def notify(self, event: UserActionEvent) -> None:
        """通知所有观察者"""
        self._event_history.append(event)
        self._logger.debug(f"📢 广播事件: {event}")
        for observer in self._observers:
            try:
                observer.on_user_action(event)
            except Exception as e:
                # 单个 observer 失败不影响其他 observer
                self._logger.error(
                    f"⚠️  {observer.__class__.__name__} 处理失败: {e}"
                )

    @property
    def observer_count(self) -> int:
        return len(self._observers)


# ============================================================
# 4. Concrete Observer：具体的观察者实现
# ============================================================

class UserProfileObserver(UserEventObserver):
    """观察者1：更新用户兴趣画像"""

    def __init__(self, profile_service: "UserProfileService"):
        self.profile_service = profile_service

    def on_user_action(self, event: UserActionEvent) -> None:
        if event.event_type == "click":
            self.profile_service.add_interest(event.user_id, event.item_id)
            logger.info(f"[画像] 用户 {event.user_id} 点击了 {event.item_id}")
        elif event.event_type == "purchase":
            self.profile_service.confirm_interest(event.user_id, event.item_id)
            logger.info(f"[画像] 用户 {event.user_id} 购买了 {event.item_id}")


class RecommendationCacheObserver(UserEventObserver):
    """观察者2：刷新推荐结果缓存"""

    CACHE_KEY_TEMPLATE = "rec:user:{user_id}:home"

    def __init__(self, redis_client):
        self.redis = redis_client

    def on_user_action(self, event: UserActionEvent) -> None:
        if event.event_type in ("click", "purchase", "like"):
            cache_key = self.CACHE_KEY_TEMPLATE.format(user_id=event.user_id)
            self.redis.delete(cache_key)
            logger.info(f"[缓存] 已清除用户 {event.user_id} 的推荐缓存")


class BIAnalyticsObserver(UserEventObserver):
    """观察者3：上报 BI 数据分析"""

    def __init__(self, kafka_producer):
        self.kafka = kafka_producer

    def on_user_action(self, event: UserActionEvent) -> None:
        topic = f"user_action_{event.event_type}"
        self.kafka.send(topic, {
            "user_id": event.user_id,
            "item_id": item_id,
            "timestamp": event.timestamp,
            "extra": event.extra_data
        })
        logger.info(f"[BI] 事件上报Kafka: {topic}")


class SimilarItemPushObserver(UserEventObserver):
    """观察者4：触发相似商品实时推送"""

    def __init__(self, push_service: "PushService"):
        self.push_service = push_service

    def on_user_action(self, event: UserActionEvent) -> None:
        if event.event_type == "purchase":
            similar_items = self.push_service.find_similar(event.item_id)
            self.push_service.send_to_user(
                event.user_id,
                f"购买了 {event.item_id}，你可能也喜欢: {similar_items}"
            )


class RealtimeCTRStatsObserver(UserEventObserver):
    """观察者5：实时 CTR 统计"""

    def __init__(self, clickhouse_client):
        self.clickhouse = clickhouse_client

    def on_user_action(self, event: UserActionEvent) -> None:
        if event.event_type in ("click", "view"):
            self.clickhouse.execute(
                "INSERT INTO ctr_stats (event_type, item_id, ts) VALUES",
                [(event.event_type, event.item_id, int(event.timestamp))]
            )


# ============================================================
# 5. 事件总线：全局单例（可选，简化调用）
# ============================================================
class UserEventBus:
    """全局事件总线，单例模式 + 观察者模式组合"""

    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._subject = UserEventSubject()
        return cls._instance

    @property
    def subject(self) -> UserEventSubject:
        return self._subject

    # 便捷方法
    def publish(self, event: UserActionEvent) -> None:
        self._subject.notify(event)

    def register(self, observer: UserEventObserver) -> None:
        self._subject.attach(observer)

    def unregister(self, observer: UserEventObserver) -> None:
        self._subject.detach(observer)


# ============================================================
# 6. 使用示例：推荐系统中集成
# ============================================================
class RecommendationService:
    """推荐服务：集成事件总线"""

    def __init__(self, redis_client, kafka_producer, profile_service, push_service):
        # 初始化事件总线并注册所有观察者
        self.event_bus = UserEventBus()
        self.event_bus.register(UserProfileObserver(profile_service))
        self.event_bus.register(RecommendationCacheObserver(redis_client))
        self.event_bus.register(BIAnalyticsObserver(kafka_producer))
        self.event_bus.register(SimilarItemPushObserver(push_service))
        self.event_bus.register(RealtimeCTRStatsObserver(...))

    def get_recommendations(self, user_id: str) -> List[str]:
        """获取推荐结果"""
        return self._recall_and_rank(user_id)

    def handle_user_action(self, user_id: str, item_id: str, action: str) -> None:
        """处理用户行为，发布事件"""
        event = UserActionEvent(
            event_type=action,
            user_id=user_id,
            item_id=item_id,
            timestamp=time.time(),
            extra_data={}
        )
        # 只需一行：发布事件，所有观察者自动响应
        self.event_bus.publish(event)
```

### TypeScript / NestJS 实现

```typescript
// ============================================================
// 1. 事件类型定义
// ============================================================
type EventType = 'click' | 'view' | 'purchase' | 'like' | 'unlike';

interface UserActionEvent {
  eventType: EventType;
  userId: string;
  itemId: string;
  timestamp: number;
  extraData?: Record<string, unknown>;
}

// ============================================================
// 2. 观察者接口
// ============================================================
interface IEventObserver {
  onUserAction(event: UserActionEvent): Promise<void> | void;
}

// ============================================================
// 3. Subject：事件发布器
// ============================================================
@Injectable()
export class UserEventSubject {
  private observers: Set<IEventObserver> = new Set();
  private logger = new Logger('UserEventSubject');

  attach(observer: IEventObserver): void {
    this.observers.add(observer);
    this.logger.log(`✅ Observer registered: ${observer.constructor.name}`);
  }

  detach(observer: IEventObserver): void {
    this.observers.delete(observer);
  }

  notify(event: UserActionEvent): void {
    for (const observer of this.observers) {
      try {
        const result = observer.onUserAction(event);
        // 支持 async observer
        if (result instanceof Promise) {
          result.catch((err) =>
            this.logger.error(`⚠️ ${observer.constructor.name} failed: ${err.message}`)
          );
        }
      } catch (err) {
        this.logger.error(`⚠️ ${observer.constructor.name} error: ${err}`);
        // 单个 observer 失败不影响其他 observer
      }
    }
  }
}

// ============================================================
// 4. NestJS Module：全局事件总线
// ============================================================
@Global()
@Module({
  providers: [UserEventSubject],
  exports: [UserEventSubject],
})
export class EventBusModule {}

// ============================================================
// 5. 具体观察者实现
// ============================================================
@Injectable()
export class UserProfileObserver implements IEventObserver {
  constructor(private profileService: UserProfileService) {}

  async onUserAction(event: UserActionEvent): Promise<void> {
    if (event.eventType === 'click') {
      await this.profileService.addInterest(event.userId, event.itemId);
    } else if (event.eventType === 'purchase') {
      await this.profileService.confirmInterest(event.userId, event.itemId);
    }
  }
}

@Injectable()
export class CacheInvalidationObserver implements IEventObserver {
  constructor(@Inject('REDIS_CLIENT') private redis: Redis) {}

  async onUserAction(event: UserActionEvent): Promise<void> {
    if (['click', 'purchase', 'like'].includes(event.eventType)) {
      await this.redis.del(`rec:user:${event.userId}:home`);
    }
  }
}

@Injectable()
export class AnalyticsObserver implements IEventObserver {
  constructor(private kafka: KafkaService) {}

  async onUserAction(event: UserActionEvent): Promise<void> {
    await this.kafka.emit(`user_action_${event.eventType}`, {
      userId: event.userId,
      itemId: event.itemId,
      timestamp: event.timestamp,
    });
  }
}

// ============================================================
// 6. Module 注册
// ============================================================
@Module({
  imports: [EventBusModule],
  providers: [
    UserProfileObserver,
    CacheInvalidationObserver,
    AnalyticsObserver,
    {
      provide: APP_INITIALIZER,
      useFactory: (subject: UserEventSubject,
                   obs: UserProfileObserver[],
                   cacheObs: CacheInvalidationObserver,
                   analyticsObs: AnalyticsObserver) => {
        subject.attach(cacheObs);
        subject.attach(analyticsObs);
        obs.forEach(o => subject.attach(o));
      },
      inject: [UserEventSubject, UserProfileObserver,
               CacheInvalidationObserver, AnalyticsObserver],
    },
  ],
})
export class RecommendationModule {}
```

## 观察者 vs Kafka：什么时候用哪个？

| 维度 | 进程内观察者模式 | Kafka 消息队列 |
|------|----------------|--------------|
| **范围** | 同一进程内 | 跨服务、跨机器 |
| **实时性** | 同步毫秒级 | 异步毫秒~秒级 |
| **可靠性** | 进程挂了消息丢 | 持久化，可重消费 |
| **解耦程度** | 代码层面解耦 | 服务层面解耦 |
| **适用场景** | 画像更新、缓存刷新、本地统计 | BI 分析、跨系统通知、数据湖同步 |
| **复杂度** | 低（几行代码） | 高（Kafka 运维、Consumer Group） |

**实际推荐系统架构：两者组合使用**

```
用户点击
    │
    ├─── 同步（进程内 Observer）─────┐
    │    ├── UserProfileObserver    │  → 更新用户画像（Redis）
    │    └── CacheInvalidObserver   │  → 清除推荐缓存
    │
    └─── 异步（Kafka）───────────────┐
         ├── AnalyticsObserver      │  → BI 数据湖
         └── SimilarPushObserver    │  → 跨服务推送
```

## 装饰器模式 + 观察者模式：给事件处理加拦截

有时候需要在事件处理前后加横切逻辑：日志记录、权限校验、性能监控。用**装饰器模式**包装 Observer：

```python
def log_event(observer: UserEventObserver) -> UserEventObserver:
    """给 Observer 加日志装饰器"""
    original_on_action = observer.on_user_action

    def wrapped(event: UserActionEvent):
        logger.info(f"📥 {observer.__class__.__name__} 收到事件: {event}")
        start = time.time()
        result = original_on_action(event)
        elapsed = time.time() - start
        logger.info(f"📤 {observer.__class__.__name__} 处理完成，耗时 {elapsed*1000:.1f}ms")
        return result

    observer.on_user_action = wrapped
    return observer


def rate_limit(max_calls: int, window_seconds: float):
    """限流装饰器：防止高频事件压垮下游"""
    def decorator(observer: UserEventObserver) -> UserEventObserver:
        calls: deque = deque()

        original = observer.on_user_action
        def wrapped(event: UserActionEvent):
            now = time.time()
            # 清理过期记录
            while calls and calls[0] < now - window_seconds:
                calls.popleft()
            if len(calls) >= max_calls:
                logger.warning(f"⏳ {observer.__class__.__name__} 限流中，丢弃事件")
                return
            calls.append(now)
            return original(event)

        observer.on_user_action = wrapped
        return observer

    return decorator
```

## 适用场景

- ✅ 用户行为触发多系统联动（点击 → 画像/缓存/BI）
- ✅ 推荐结果实时更新（用户操作后即时刷新推荐）
- ✅ 插件式功能扩展（新增观察者无需改 Subject）
- ✅ 日志/监控横切关注点（装饰器包装 Observer）
- ✅ 需要支持动态注册/注销观察者（运行时开关功能）

## 常见误区

- ❌ **过度使用**：不是所有地方都要用观察者，简单场景直接调函数即可
- ❌ **Observer 中抛出异常导致通知中断**：必须 try-catch，单个失败不影响整体
- ❌ **Observer 之间有依赖顺序但没有处理**：可以用 `@depends` 装饰器或 OrderedObserver 接口
- ❌ **内存泄漏**：动态注册后忘记 detach，长期运行的进程会导致 Observer 列表无限增长
- ❌ **把同步 Observer 当异步用**：做 Kafka/网络IO的 Observer 要么 async，要么用线程池

## 总结：观察者模式在推荐系统中的位置

```
┌────────────────────────────────────────────────────────┐
│                   推荐系统事件驱动架构                    │
├────────────────────────────────────────────────────────┤
│                                                        │
│   用户行为 ──► EventBus.notify()                       │
│                    │                                   │
│         ┌──────────┼──────────┬──────────┐             │
│         ▼          ▼          ▼          ▼             │
│    Profile    Cache      BI-Analytics  Push           │
│    Observer  Observer    Observer    Observer        │
│                                                        │
│   ⚠️ 注意：                                             │
│   · 同一进程用 Observer，跨服务用 Kafka                 │
│   · Observer 要幂等（Kafka 可能重复投递）               │
│   · 装饰器统一加日志/限流/重试                          │
└────────────────────────────────────────────────────────┘
```

## 练习一：单文件推荐服务 → 整洁四层架构

### 📋 场景

你接手了一个"能用"的推荐系统，单文件 `recommend.py` 共 400 行，所有逻辑混在一起：
- 直接连 MySQL 查用户画像
- 直接发 HTTP 请求给模型推理服务
- 直接写 Redis 缓存
- 直接调 Kafka 发事件
- 路由层代码直接操作数据库连接

产品说"要加一个新召回策略"，你发现要改 8 个地方，还可能改坏已有的。

### ❌ 反例：单文件耦合地狱

```python
# recommend.py - 400行单文件（节选核心问题）
from flask import Flask, request
import pymysql
import redis
import requests
import json
from kafka import KafkaProducer

app = Flask(__name__)

# 全局连接（启动时初始化）
mysql_conn = pymysql.connect(host='127.0.0.1', port=3306, user='root', password='xxx', database='rec_db')
redis_client = redis.Redis(host='127.0.0.1', port=6379, db=0)
kafka_producer = KafkaProducer(bootstrap_servers=['127.0.0.1:9092'], value_serializer=lambda v: json.dumps(v).encode())

MODEL_URL = "http://127.0.0.1:8001/predict"  # 硬编码URL

def get_recommendations(user_id: int, scene: str, page: int = 1, page_size: int = 10):
    """推荐主函数 - 400行中的冰山一角"""
    
    # 问题1: 直接查MySQL，跨层调用
    cursor = mysql_conn.cursor()
    cursor.execute("SELECT age, gender, city FROM users WHERE id = %s", (user_id,))
    user_row = cursor.fetchone()
    if not user_row:
        return []
    user = {'age': user_row[0], 'gender': user_row[1], 'city': user_row[2]}
    
    # 问题2: 逻辑和SQL混在一起，Service层职责不明
    cursor.execute("""
        SELECT item_id, score FROM user_history 
        WHERE user_id = %s AND action = 'click' 
        ORDER BY created_at DESC LIMIT 100
    """, (user_id,))
    history = [row[0] for row in cursor.fetchall()]
    
    # 问题3: 硬编码模型URL
    payload = {'user_id': user_id, 'history': history, 'scene': scene}
    try:
        resp = requests.post(MODEL_URL, json=payload, timeout=3)
    except:
        resp = None  # bare except，吞掉所有异常
    
    if resp and resp.status_code == 200:
        ranked = resp.json().get('items', [])
    else:
        # 问题4: 降级逻辑硬编码在主函数里
        cursor.execute("SELECT id FROM items ORDER BY hot_score DESC LIMIT 20")
        ranked = [row[0] for row in cursor.fetchall()]
    
    # 问题5: 缓存直接操作，不抽象
    cache_key = f"rec:{user_id}:{scene}:{page}"
    cached = redis_client.get(cache_key)
    if cached:
        return json.loads(cached)
    
    result = ranked[(page-1)*page_size: page*page_size]
    redis_client.setex(cache_key, 300, json.dumps(result))
    
    # 问题6: 事件发布直接调Kafka，耦合
    kafka_producer.send('recommendation_events', {
        'user_id': user_id, 'scene': scene, 'items': result
    })
    
    return result

@app.route('/api/rec', methods=['GET'])
def api_rec():
    uid = request.args.get('user_id', type=int)
    scene = request.args.get('scene', 'home')
    return json.dumps(get_recommendations(uid, scene))
```

### 🔍 问题分析

| 问题 | 违反原则 | 根因 |
|------|---------|------|
| 全局连接在模块加载时初始化 | 可测试性差 | 硬编码依赖 |
| Service 层直接操作 MySQL | SRP、OCP | 没有分层 |
| 硬编码 MODEL_URL | OCP | 依赖具体实现 |
| bare except | 错误处理规范 | 不知道怎么处理 |
| 降级逻辑硬编码 | OCP | 没有策略模式 |
| 缓存直接操作 Redis | DIP、OCP | 没有端口抽象 |
| Kafka 直接调用 | DIP | 没有事件端口 |
| 路由和业务混在一起 | 单一职责 | MVC 分离不清 |

# ===== domain/ports/recall_port.py =====
from abc import ABC, abstractmethod
from typing import List

class RecallPort(ABC):
    """召回端口：抽象"如何召回"，不关心实现"""
    @abstractmethod
    async def recall(self, user_id: int, scene: str, candidates: int = 100) -> List[int]:
        pass

# ===== domain/services/recommendation_domain_service.py =====
class RecommendationDomainService:
    """推荐领域服务：只包含业务规则"""
    
    def __init__(self, recall_port: RecallPort, ranking_port: RankingPort):
        self._recall = recall_port
        self._ranker = ranking_port
    
    async def recommend(self, ctx) -> List[ItemScore]:
        candidates = await self._recall.recall(ctx.user_id, ctx.scene, candidates=100)
        if not candidates:
            return self._ranker.fallback_by_scene(ctx.scene)
        
        ranked = await self._ranker.rank(ctx.user_id, candidates)
        return self._apply_business_rules(ranked, ctx.scene)
    
    def _apply_business_rules(self, items, scene):
        # 去重/曝光过滤/商业化过滤，规则可配置
        return items

# ===== infrastructure/di/container.py =====
class DIContainer:
    @staticmethod
    def initialize():
        config = Config.load()
        
        milvus = MilvusClient(config.milvus_uri)
        recall_adapter = EmbeddingRecallAdapter(milvus, config.model_url)
        ranking_adapter = HTTPRankingAdapter(config.model_url)
        cache_adapter = RedisCacheAdapter(config.redis_url)
        event_adapter = KafkaEventAdapter(config.kafka_brokers)
        profile_adapter = MySQLUserProfileAdapter(config.db_url)
        
        domain = RecommendationDomainService(recall_adapter, ranking_adapter, profile_adapter)
        return GetRecommendationsUseCase(domain, cache_adapter, event_adapter)

# ===== interfaces/controllers/recommendation_controller.py =====
class RecommendationController:
    async def get(self, user_id: int, scene: str, page: int, page_size: int):
        """Controller 只做请求解析 + 响应封装"""
        try:
            result = await container.execute(user_id, scene, page, page_size)
            return SuccessResponse(result)
        except UserNotFoundError:
            return ErrorResponse(code=404, message="用户不存在")
        # 不在 Controller 里写业务逻辑！

## 练习二：if-else 召回策略 → 策略模式 + 工厂模式

### 📋 场景

产品要求：为不同场景（首页/详情页/搜索页/关注页）提供不同召回策略，且策略要能动态配置切换。现有代码用 if-else 硬编码。

### ❌ 反例：策略硬编码

```python
# 反例：每次加新场景都要改主函数
def do_recall(user_id, scene):
    if scene == 'home':
        return home_recall(user_id)
    elif scene == 'detail':
        return detail_recall(user_id)
    elif scene == 'search':
        return search_recall(user_id)
    elif scene == 'follow':
        return follow_recall(user_id)
    else:
        return default_recall(user_id)
```

### ✅ 正例：策略模式

```python
# ===== recall_strategies/base.py =====
from abc import ABC, abstractmethod
from typing import List

class RecallStrategy(ABC):
    """召回策略抽象接口"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        pass
    
    @abstractmethod
    async def recall(self, user_id: int, limit: int = 100) -> List[int]:
        pass

# ===== recall_strategies/implementations.py =====
class HomeRecallStrategy(RecallStrategy):
    name = "home"
    
    async def recall(self, user_id: int, limit: int = 100) -> List[int]:
        user_vec = await self._get_user_embedding(user_id)
        similar = await self._milvus.search(user_vec, top_k=limit)
        hot_items = await self._mysql.query(
            "SELECT id FROM items ORDER BY hot_score DESC LIMIT 20"
        )
        return self._mix(similar, hot_items, ratio=0.8)

class DetailRecallStrategy(RecallStrategy):
    name = "detail"
    
    async def recall(self, user_id: int, limit: int = 100) -> List[int]:
        recent_item = await self._mysql.query("""
            SELECT item_id FROM user_history 
            WHERE user_id=%s AND action='view' ORDER BY ts DESC LIMIT 1
        """, (user_id,))
        if not recent_item:
            return []
        category = await self._mysql.query(
            "SELECT category_id FROM items WHERE id=%s", (recent_item[0],)
        )
        return await self._mysql.query("""
            SELECT id FROM items WHERE category_id=%s AND id!=%s
            ORDER BY score DESC LIMIT %s
        """, (category[0], recent_item[0], limit))

class SearchRecallStrategy(RecallStrategy):
    name = "search"
    
    async def recall(self, user_id: int, limit: int = 100) -> List[int]:
        es_results = await self._es.search(keywords=self._query, size=limit)
        return [r['item_id'] for r in es_results]

class FollowRecallStrategy(RecallStrategy):
    name = "follow"
    
    async def recall(self, user_id: int, limit: int = 100) -> List[int]:
        follows = await self._mysql.query(
            "SELECT followed_id FROM follows WHERE user_id=%s", (user_id,)
        )
        if not follows:
            return []
        ids = [f[0] for f in follows]
        return await self._mysql.query(f"""
            SELECT id FROM items WHERE user_id IN ({','.join(['%s']*len(ids))})
            ORDER BY created_at DESC LIMIT %s
        """, ids + [limit])

# ===== recall_strategies/factory.py =====
class RecallStrategyFactory:
    """工厂：根据场景名称创建策略实例"""
    
    def __init__(self, strategies: List[RecallStrategy]):
        self._map = {s.name: s for s in strategies}
    
    def create(self, scene: str) -> RecallStrategy:
        return self._map.get(scene, self._map['home'])
    
    @classmethod
    def from_config(cls, config: dict):
        """从功能开关配置加载策略，修改配置不用发版"""
        all_strategies = [HomeRecallStrategy(), SearchRecallStrategy(), 
                          DetailRecallStrategy(), FollowRecallStrategy()]
        enabled = [s for s in all_strategies if s.name in config.get('enabled_scenes', [])]
        return cls(enabled)

# ===== recall_engine.py =====
class RecallEngine:
    def __init__(self, factory: RecallStrategyFactory):
        self._factory = factory
    
    async def multi_recall(self, user_id: int, scene: str) -> List[int]:
        """多路召回：并行执行多策略"""
        strategies = [HomeRecallStrategy(), HotRecallStrategy(), NewRecallStrategy()]
        tasks = [s.recall(user_id, 50) for s in strategies]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        merged = []
        for r in results:
            if isinstance(r, list):
                merged.extend(r)
        return list(set(merged))
```

### ✅ 验收标准

| 检查项 | 验收方法 |
|--------|---------|
| ✅ 新增策略不修改主流程 | 新写一个类实现 `RecallStrategy`，加到工厂注册 |
| ✅ 策略可独立测试 | `pytest MyNewStrategy` 无需启动 MySQL/Redis |
| ✅ 策略可运行时切换 | 改功能开关配置，Flask reload，新策略生效 |

## 练习三：裸调用模型推理 → 超时/重试/熔断三件套

### 📋 场景

推荐系统中的模型推理服务经常超时、抖动。一台下游模型服务挂了，整个推荐跟着雪崩。产品要求 P99 稳定在 300ms 以内。

### ❌ 反例：裸调用无保护

```python
# 危险：直接 requests.post，没有超时/重试/熔断
def call_model_ranker(user_id, candidates):
    resp = requests.post(MODEL_URL, json={'user_id': user_id, 'items': candidates})
    return resp.json()['scores']
```

### ✅ 正例：三层韧性保护

```python
# ===== resilience/circuit_breaker.py =====
import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"       # 正常：所有请求通过
    OPEN = "open"           # 熔断：拒绝所有请求
    HALF_OPEN = "half_open" # 半开：放一个请求试试

class CircuitBreaker:
    def __init__(self, failure_threshold=5, timeout_seconds=30):
        self._state = CircuitState.CLOSED
        self._failure_count = 0
        self._failure_threshold = failure_threshold
        self._timeout = timeout_seconds
        self._opened_at = None
    
    def call(self, func, *args, **kwargs):
        if self._state == CircuitState.OPEN:
            if time.time() - self._opened_at > self._timeout:
                self._state = CircuitState.HALF_OPEN
            else:
                raise CircuitOpenError("熔断中，快速失败")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception:
            self._on_failure()
            raise
    
    def _on_success(self):
        self._failure_count = 0
        if self._state == CircuitState.HALF_OPEN:
            self._state = CircuitState.CLOSED
    
    def _on_failure(self):
        self._failure_count += 1
        if self._failure_count >= self._failure_threshold:
            self._state = CircuitState.OPEN
            self._opened_at = time.time()

# ===== resilience/resilient_ranker.py =====
import asyncio
import random
import httpx

class ResilientRanker:
    def __init__(self, model_url: str):
        self._url = model_url
        self._breaker = CircuitBreaker(failure_threshold=3, timeout_seconds=30)
        self._semaphore = asyncio.Semaphore(10)  # 每服务并发上限
    
    async def rank(self, user_id: int, item_ids: List[int]) -> Dict[int, float]:
        async with self._semaphore:
            return await self._call_with_retry(user_id, item_ids)
    
    async def _call_with_retry(self, user_id, item_ids):
        last_error = None
        for attempt in range(3):
            try:
                async with httpx.AsyncClient() as client:
                    resp = await client.post(
                        self._url,
                        json={'user_id': user_id, 'items': item_ids},
                        timeout=httpx.Timeout(connect=1.0, read=5.0)  # 双超时
                    )
                    resp.raise_for_status()
                    return resp.json()['scores']
            except (httpx.TimeoutException, httpx.ConnectError) as e:
                last_error = e
                delay = (2 ** attempt) + random.uniform(-0.5, 0.5)
                logger.warning(f"模型调用失败（尝试{attempt+1}/3），{delay:.1f}s后重试")
                await asyncio.sleep(delay)
            except httpx.HTTPStatusError:
                raise  # 业务异常（4xx/5xx）不重试
        
        logger.error(f"模型推理全部失败，触发降级")
        return self._fallback(item_ids)
    
    def _fallback(self, item_ids: List[int]) -> Dict[int, float]:
        """三级降级：模型 → 缓存 → 静态分"""
        cached = self._score_cache.get_many(item_ids)
        return {
            item_id: cached.get(item_id, 0.0)  # 没有缓存则返回0
            for item_id in item_ids
        }
```

### ✅ 验收标准

| 检查项 | 验收方法 |
|--------|---------|
| ✅ 每个外部调用有超时 | `timeout=5.0`，无 timeout 的 requests 被代码审查拒绝 |
| ✅ 重试有指数退避 | 日志中看到 `delay: 1.0s → 2.1s → 4.3s` |
| ✅ 业务异常不重试 | `4xx` 响应码不重试，只有网络异常重试 |
| ✅ 熔断器独立保护每个下游 | 模型服务 / Redis / Kafka 各有独立 CircuitBreaker |
| ✅ 降级有兜底，不返回空结果 | `self._fallback(item_ids)` 确保总有数 |

## 练习四：同步写库 → Kafka 事件驱动

### 📋 场景

用户点击商品后，推荐结果要实时更新（点击了不推同类），但同步写库拖慢了推荐接口 P99 到 500ms。需要改为异步事件驱动。

### ❌ 反例：同步写库

```python
@app.route('/api/click', methods=['POST'])
def record_click():
    data = request.json
    # 问题：三个同步操作，一个慢全链路慢
    db.execute("INSERT INTO clicks ...", (data['user_id'], data['item_id'], ...))
    redis_client.delete(f"rec:{data['user_id']}:*")  # 同步刷新缓存
    redis_client.incr(f"ctr:{data['item_id']}")       # 同步更新统计
    return {'status': 'ok'}
```

### ✅ 正例：Kafka 事件驱动

```python
# ===== events/click_event.py =====
from dataclasses import dataclass
from datetime import datetime

@dataclass
class UserClickedItem:
    user_id: int
    item_id: int
    scene: str
    position: int
    timestamp: datetime
    
    def to_kafka(self) -> dict:
        return {
            'event_type': 'user_clicked_item',
            'user_id': self.user_id,
            'item_id': self.item_id,
            'scene': self.scene,
            'position': self.position,
            'ts': self.timestamp.isoformat(),
        }

# ===== application/use_cases/record_click.py =====
class RecordClickUseCase:
    def __init__(self, event_port: EventPort, cache_port: CachePort):
        self._events = event_port
        self._cache = cache_port
    
    async def execute(self, user_id: int, item_id: int, scene: str, position: int):
        await self._cache.set(f"clicked:{user_id}:{item_id}", 1, ttl=86400)
        
        # 发布领域事件（异步，不阻塞响应）
        event = UserClickedItem(user_id, item_id, scene, position, datetime.now())
        await self._events.publish(event)
        
        return {'status': 'ok'}  # P99 < 10ms

# ===== infrastructure/consumers/profile_consumer.py =====
class ProfileUpdateConsumer:
    """消费者1：更新用户画像（异步，不影响推荐主流程）"""
    async def handle(self, event: UserClickedItem):
        await self._profile_service.increment_interest(
            user_id=event.user_id,
            category_id=await self._get_item_category(event.item_id),
            weight=0.1
        )

class CacheRefreshConsumer:
    """消费者2：刷新推荐缓存"""
    async def handle(self, event: UserClickedItem):
        for scene in ['home', 'detail']:
            await self._cache.delete(f"rec:{event.user_id}:{scene}:*")

class AnalyticsConsumer:
    """消费者3：数据分析（完全解耦）"""
    async def handle(self, event: UserClickedItem):
        await self._redis.incr(f"ctr:hourly:{event.item_id}:{event.scene}")
        await self._click_dw.write(event)  # 写数据仓库
```

### ✅ 验收标准

| 检查项 | 验收方法 |
|--------|---------|

| ✅ 事件发布幂等 | Kafka 开启 `enable_idempotence=True` |
| ✅ 各 Consumer 独立扩缩容 | Consumer Group 隔离 |
| ✅ 事件消费有失败重试 | Consumer 异常时重试 3 次，DLQ 兜底 |

## 练习六：容量规划 + 演进路线图

### 📋 场景

公司推荐系统当前 QPS 500，团队 3 人。产品预计 3 个月后 DAU 从 50 万涨到 200 万（4倍），流量峰值 4 倍。主管问你："架构要不要改？怎么改？"

### ✅ 系统化分析

```python
# ===== capacity_analysis.py =====

"""
容量规划步骤：

Step 1: 现状基线
  - 当前 QPS: 500
  - P99 延迟: 150ms
  - DAU: 50万
  - 服务器: 4台 8C16G（API层）+ 2台 32C64G（模型推理）

Step 2: 增长预测
  - 3个月后 DAU: 200万（4倍）
  - 峰值 QPS = 500 × 4 × 2（峰值系数）= 4000 QPS

Step 3: 各层容量计算
"""

# 推荐系统容量估算
def calculate_capacity(current_qps=500, dau_growth=4, peak_factor=2):
    target_qps = current_qps * dau_growth * peak_factor
    print(f"目标 QPS: {target_qps}")  # 4000 QPS
    
    # API层容量
    # 每台 8C16G Flask 服务约能扛 125 QPS（QPS 500/4台 = 125）
    # 目标 4000 QPS → 需要 4000/125 = 32台（留 20% 余量 → 38台）
    api_servers = target_qps / 125 * 1.2
    print(f"API服务器需求: {api_servers:.0f} 台")
    
    # 缓存层容量
    # DAU 200万，每用户推荐结果（20 items × 50 bytes × 3场景）= 3KB
    # 缓存总量: 200万 × 3KB = 6GB（Redis 单实例可扛，留一主两从）
    cache_size_gb = dau_growth * 50 * 3 * 1024 / 1024 / 1024
    print(f"缓存容量需求: {cache_size_gb:.1f} GB")
    
    # 模型推理容量
    # P99 < 300ms，每批次最多 500 个候选
    # 4000 QPS × 平均 100 个候选 = 40万 items/秒
    # 单卡 A10G (24GB) 每秒推理 ~2000 次 → 需要 200 个实例（留余量 250个）
    model_instances = target_qps / 2000 * 1.3
    print(f"模型推理实例: {model_instances:.0f} 个 GPU 实例")
    
    return {
        'api_servers': int(api_servers),
        'cache_size_gb': cache_size_gb,
        'model_instances': int(model_instances),
    }

# ===== evolution_roadmap.py =====

"""
架构演进路线：

阶段1（当前 ~ 3个月）：分层单体
────────────────────────────────
QPS 500 → 4000，团队 3 人
├─ 重点：分层架构 + Redis 缓存 + 熔断保护
├─ 目标：不拆微服务，靠垂直扩展（加机器）扛住 4000 QPS
└─ 决策：每台 API 服务器扩到 16C32G，用 K8s HPA 自动扩缩

阶段2（3个月 ~ 6个月）：垂直拆分
────────────────────────────────
DAU 200万 → 500万（预估），QPS 4000 → 10000
├─ 重点：服务垂直拆分（用户域 / 内容域 / 推荐引擎域）
├─ 新增团队成员到 8 人（各域独立维护）
└─ 决策：推荐引擎域先拆（压力最大，收益最高）

阶段3（6个月 ~ 1年）：水平拆分
────────────────────────────────
特征工程 + 模型服务独立部署，引入 Feature Store
├─ 重点：训练-服务特征一致性，AB 测试平台
├─ 技术：Flink 实时特征 / Milvus 向量召回 / Triton 模型服务
└─ 决策：全量重构代价太高，用 Strangler Fig 灰度替换

演进决策树：

流量预估 < 6个月 4倍增长？
├─ 是 → 阶段1分层 + 垂直扩展（不改架构）
└─ 否 → 继续判断：
    团队 > 5人 且 各域独立发布需求强烈？
    ├─ 是 → 阶段2垂直拆分（按领域边界拆服务）
    └─ 否 → 继续单体，积累领域知识

微服务 > 100台服务器？
├─ 是 → 阶段3引入服务网格（Istio）
└─ 否 → 继续当前规模
"""
```

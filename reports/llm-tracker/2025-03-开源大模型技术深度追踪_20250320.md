### 1.3 GRPO数学原理详解

#### 核心公式

```python
def grpo_loss(policy_logps, ref_logps, rewards, group_size, beta=0.1):
    """
    GRPO损失函数
    
    参数:
    - policy_logps: 当前策略的log概率 [batch, seq_len]
    - ref_logps: 参考策略的log概率 [batch, seq_len]
    - rewards: 每条样本的奖励值 [batch]
    - group_size: 每组的样本数
    - beta: KL散度惩罚系数
    """
    # 1. 计算概率比
    ratio = torch.exp(policy_logps - ref_logps)
    
    # 2. 按组归一化计算优势（核心创新）
    # 假设group_size=4，每个问题采样4个答案
    rewards_grouped = rewards.view(-1, group_size)
    
    # 组内相对优势：减去组内均值，除以组内标准差
    advantages = (rewards_grouped - rewards_grouped.mean(dim=-1, keepdim=True))
    advantages = advantages / (rewards_grouped.std(dim=-1, keepdim=True) + 1e-8)
    advantages = advantages.view(-1, 1)  # [batch, 1]
    
    # 3. PPO风格的裁剪目标函数
    surr1 = ratio * advantages
    surr2 = torch.clamp(ratio, 0.9, 1.1) * advantages
    policy_loss = -torch.min(surr1, surr2).mean()
    
    # 4. KL惩罚项
    kl_loss = F.kl_div(
        policy_logps.exp(), 
        ref_logps.exp(), 
        reduction='batchmean'
    )
    
    return policy_loss + beta * kl_loss
```

#### 优势分析

**为什么不需要Value函数？**

1. **组内归一化替代基线**：用同一问题的多个采样输出的均值作为"基线"
2. **奖励归一化**：组内std归一化使得优势估计更稳定
3. **降低复杂度**：省去了训练和存储Value模型的开销

```math
\hat{A}_i = \frac{r_i - \text{mean}(\mathbf{r}_{\text{group}})}{\text{std}(\mathbf{r}_{\text{group}}) + \epsilon}
```

1. **解耦裁剪上下界**：上下界使用不同的epsilon值
2. **动态采样**：根据难度调整采样数量
3. **token级损失**：更细粒度的梯度更新

## 二、Mamba-2与状态空间模型

### 2.1 Mamba-2核心架构

Mamba-2在ICML 2024发表，提出**结构化状态空间对偶性(SSSD)**，统一Transformer和SSM：

```python
class Mamba2Block(nn.Module):
    """
    Mamba-2核心块
    特点：支持并行训练 + 高效推理
    """
    def __init__(self, d_model, d_state=128, nheads=8):
        super().__init__()
        self.d_model = d_model
        self.d_state = d_state
        self.nheads = nheads
        
        # 投影层
        self.in_proj = nn.Linear(d_model, d_model * 2)
        
        # 状态空间参数
        self.x_proj = nn.Linear(d_model, d_state * 2)
        self.dt_proj = nn.Linear(d_state, d_model)
        
        # 状态矩阵（可学习，对角结构）
        self.A_log = nn.Parameter(torch.randn(nheads, d_state))
        self.D = nn.Parameter(torch.ones(nheads))
        
        # 输出投影
        self.out_proj = nn.Linear(d_model, d_model)
        
    def forward(self, x):
        """
        前向传播：结合卷积和递归表示
        """
        # 门控投影
        gate, x_inner = self.in_proj(x).chunk(2, dim=-1)
        gate = F.silu(gate)
        
        # 状态参数
        s, dt = self.x_proj(x_inner).chunk(2, dim=-1)
        dt = F.softplus(self.dt_proj(dt))
        
        # 状态矩阵
        A = -torch.exp(self.A_log.float())  # [nheads, d_state]
        
        # 并行状态计算（核心创新）
        # 使用块展开矩阵乘法实现并行扫描
        output = self.ssm_parallel(x_inner, A, dt, s)
        
        # 门控输出
        output = output * gate
        return self.out_proj(output)
    
    def ssm_parallel(self, x, A, dt, s):
        """
        并行状态空间计算
        复杂度: O(L * d_state) 而非 O(L^2)
        """
        # 生成卷积核
        L = x.size(1)
        device = x.device
        
        # 创建A的幂序列用于并行计算
        # 这是一个简化的并行扫描实现
        A_exp = torch.exp(A.unsqueeze(2) * dt.unsqueeze(1))  # [nheads, d_state, d_state]
        
        # 状态累积
        # 实际实现使用parallel scan算法
        h = torch.zeros(x.size(0), self.nheads, self.d_state, device=device)
        
        for t in range(L):
            h = A_exp @ h + x[:, t].unsqueeze(-1)
            
        # 状态到输出
        C = torch.randn(self.nheads, self.d_state, device=device)
        output = (h @ C.unsqueeze(2)).squeeze(-1)
        
        return output + self.D.unsqueeze(0) * x
```

### 2.2 结构化状态空间对偶性

Mamba-2的核心创新是证明SSM可以表示为**结构化线性注意力**：

```math
\text{SSM}(A, B, C)(x)_t = \sum_{i=0}^{t} C^T A^{t-i} B x_i

\text{等价于线性注意力形式：}
\text{Attention}(Q, K, V)_t = \frac{\sum_{i=0}^{t} \phi(Q_t)^T \phi(K_i) V_i}{\sum_{i=0}^{t} \phi(Q_t)^T \phi(K_i)}
```

这意味着可以用Transformer的优化方法来加速SSM！

### 3.2 位置编码外推技术

#### 3.2.1 旋转位置编码(RoPE)基础

```python
class RoPE(nn.Module):
    """
    旋转位置编码
    核心思想：将位置信息编码为旋转矩阵
    """
    def __init__(self, dim, base=10000):
        super().__init__()
        self.dim = dim
        self.base = base
        # 预计算频率
        self.register_buffer('inv_freq', 1.0 / (base ** (torch.arange(0, dim, 2).float() / dim)))
        
    def forward(self, seq_len, device):
        # 生成位置
        positions = torch.arange(seq_len, device=device).float()
        # 计算角度
        angles = positions.unsqueeze(1) * self.inv_freq.unsqueeze(0)
        # 复数形式
        angles = torch.cat([angles, angles], dim=-1)
        return angles.cos(), angles.sin()
    
    def apply_rope(self, x, cos, sin):
        """
        应用旋转编码到查询和键
        x: [batch, heads, seq, dim]
        """
        # 分割实部和虚部
        x1, x2 = x[..., :x.size(-1)//2], x[..., x.size(-1)//2:]
        
        # 旋转
        return torch.cat([
            x1 * cos - x2 * sin,
            x1 * sin + x2 * cos
        ], dim=-1)
```

```python
def position_interpolation(rope, old_seq_len, new_seq_len):
    """
    位置编码插值
    将[new_seq_len]映射到[old_seq_len]
    """
    scale = old_seq_len / new_seq_len
    
    # 重新计算频率
    new_inv_freq = rope.inv_freq / scale
    
    # 位置索引也需要缩放
    positions = torch.arange(new_seq_len).float()
    scaled_positions = positions * scale
    
    # 计算新的角度
    angles = scaled_positions.unsqueeze(1) * new_inv_freq.unsqueeze(0)
    angles = torch.cat([angles, angles], dim=-1)
    
    return angles.cos(), angles.sin()
```

### 3.3 注意力机制优化

#### 3.3.1 稀疏注意力模式

```python
class SparseAttention(nn.Module):
    """
    滑动窗口 + 全局注意力
    典型配置：窗口大小512，每隔64个token一个全局token
    """
    def __init__(self, d_model, window_size=512, global_interval=64):
        super().__init__()
        self.window_size = window_size
        self.global_interval = global_interval
        
        self.qkv = nn.Linear(d_model, d_model * 3)
        self.o_proj = nn.Linear(d_model, d_model)
        
    def forward(self, x):
        B, N, C = x.shape
        qkv = self.qkv(x).reshape(B, N, 3, -1, C // 3).permute(2, 0, 3, 1, 4)
        q, k, v = qkv[0], qkv[1], qkv[2]
        
        # 识别全局token位置
        global_mask = (torch.arange(N, device=x.device) % self.global_interval == 0)
        
        # 局部窗口注意力
        output = torch.zeros_like(q)
        for i in range(N):
            # 窗口范围
            start = max(0, i - self.window_size)
            end = min(N, i + self.window_size)
            
            # 局部注意力
            q_i = q[:, :, i:i+1, :]
            k_local = k[:, :, start:end, :]
            v_local = v[:, :, start:end, :]
            
            attn_local = (q_i @ k_local.transpose(-2, -1)) / math.sqrt(C)
            attn_local = F.softmax(attn_local, dim=-1)
            out_local = attn_local @ v_local
            
            # 全局注意力（所有全局token参与）
            if global_mask.any():
                q_global = q[:, :, i:i+1, :]
                k_global = k[:, :, global_mask, :]
                v_global = v[:, :, global_mask, :]
                
                attn_global = (q_global @ k_global.transpose(-2, -1)) / math.sqrt(C)
                attn_global = F.softmax(attn_global, dim=-1)
                out_global = attn_global @ v_global
                
                # 合并
                out_local = out_local + out_global * 0.3
            
            output[:, :, i:i+1, :] = out_local
        
        return self.o_proj(output.permute(0, 2, 1, 3).reshape(B, N, C))
```

#### 3.3.2 Flash Attention长序列优化

```python
def flash_attention_long_seq(Q, K, V, block_size=128):
    """
    分块Flash Attention
    适用于超长序列
    """
    B, H, N, D = Q.shape
    n_blocks = (N + block_size - 1) // block_size
    
    # 存储输出和归一化因子
    output = torch.zeros_like(Q)
    l = torch.zeros(B, H, N, device=Q.device, dtype=Q.dtype)
    m = torch.full((B, H, N), -float('inf'), device=Q.device)
    
    for i in range(n_blocks):
        # 获取当前块
        start_idx = i * block_size
        end_idx = min(start_idx + block_size, N)
        
        Q_block = Q[:, :, start_idx:end_idx, :]
        
        # 与所有K,V计算（简化版）
        attn_block = torch.matmul(Q_block, K.transpose(-2, -1)) / math.sqrt(D)
        
        # 块内softmax
        attn_block = F.softmax(attn_block, dim=-1)
        
        # 累计输出
        output[:, :, start_idx:end_idx, :] = torch.matmul(attn_block, V)
        
        # 更新归一化因子
        m[:, :, start_idx:end_idx] = attn_block.max(dim=-1)[0]
        l[:, :, start_idx:end_idx] = attn_block.sum(dim=-1)
    
    return output
```

## 四、量化技术深度解析

### 4.1 量化基础理论

#### 4.1.1 量化映射

```python
def quantize_tensor(x, bits=4, group_size=128):
    """
    量化一个张量
    
    参数:
    - x: 输入张量
    - bits: 量化位数（2, 3, 4, 8）
    - group_size: 每组共享缩放因子的token数
    """
    # 确定形状
    original_shape = x.shape
    if len(original_shape) > 1:
        # 对最后一个维度分块
        n_groups = original_shape[-1] // group_size
        x_flat = x.reshape(-1, n_groups, group_size)
    else:
        x_flat = x.unsqueeze(0)
    
    # 计算每组的范围
    x_min = x_flat.min(dim=-1, keepdim=True)[0]
    x_max = x_flat.max(dim=-1, keepdim=True)[0]
    
    # 量化范围
    n_levels = 2 ** bits - 1  # 4bit = 15级
    
    # 缩放因子和零点
    scale = (x_max - x_min) / n_levels
    zero = -x_min / scale
    
    # 量化
    x_quant = torch.round(x_flat / scale + zero)
    x_quant = torch.clamp(x_quant, 0, n_levels)
    
    # 反量化（用于训练）
    x_dequant = (x_quant - zero) * scale
    
    return x_quant, scale, zero, original_shape
```

### 4.2 AWQ量化原理

#### 4.2.1 激活感知权重量化

AWQ (Activation-aware Weight Quantization) 的核心发现：**只有1%的显著权重需要保持高精度**

```python
class AWQLayer(nn.Module):
    """
    AWQ: 激活感知权重量化
    核心：保护显著权重，量化普通权重
    """
    def __init__(self, in_features, out_features, bits=4):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.bits = bits
        
        # 原始权重（FP16）
        self.weight = nn.Parameter(torch.randn(out_features, in_features))
        
        # 缩放因子（每output channel一组）
        self.scale = nn.Parameter(torch.ones(out_features))
        
        # 显著权重掩码（可学习）
        self.alpha = nn.Parameter(torch.ones(out_features, 1) * 0.4)
        
    def forward(self, x):
        # 计算每个输出通道的"重要性"（基于激活）
        # 使用小的校准集来估计
        with torch.no_grad():
            # 简化的重要性计算
            importance = (x.abs().mean(dim=0) @ self.weight.abs().mean(dim=1))
            importance = importance / importance.sum()
        
        # 确定显著权重阈值
        threshold = torch.quantile(importance, 1 - self.alpha)
        
        # 分离显著权重和普通权重
        mask = importance > threshold
        weight_sig = self.weight * mask.float()
        weight_nonsig = self.weight * (~mask).float()
        
        # 显著权重保持高精度
        # 普通权重量化
        weight_nonsig_q = self.quantize(weight_nonsig)
        
        # 合并
        weight_final = weight_sig + weight_nonsig_q
        
        return F.linear(x, weight_final, None)
    
    def quantize(self, w):
        """量化权重"""
        # 简化的4bit量化
        w_max = w.abs().max()
        scale = (2 ** self.bits - 1) / w_max
        w_q = torch.round(w * scale) / scale
        return w_q
```

#### 4.2.2 AWQ数学原理

```math
\min_{\alpha} \| W \odot M \cdot x + \text{quant}(W \odot (1-M)) \cdot x \|_2^2

其中:
- M \in \{0,1\}^{d_{out} \times d_{in}}: 显著权重掩码
- \alpha: 掩码阈值参数
- quant(\cdot): 量化函数
```

### 4.3 GPTQ量化原理

#### 4.3.1 基于Hessian的逐通道量化

```python
class GPTQLayer(nn.Module):
    """
    GPTQ: 基于Hessian的量化
    核心：利用二阶信息优化量化误差
    """
    def __init__(self, layer, bits=4):
        super().__init__()
        self.layer = layer
        self.bits = bits
        
        # 提取权重
        self.weight = layer.weight.data.clone()
        self.out_features, self.in_features = self.weight.shape
        
        # 存储量化后的权重和缩放
        self.weight_quant = torch.zeros_like(self.weight)
        self.scale = torch.zeros(self.out_features, device=self.weight.device)
        self.offset = torch.zeros(self.out_features, device=self.weight.device)
        
    def quantize(self, calib_x):
        """
        校准量化
        
        calib_x: 校准数据集的输入
        """
        # 计算Hessian近似（对角近似）
        # H ≈ (X^T X) 的对角元素
        H = calib_x.T @ calib_x / calib_x.shape[0]
        H_diag = H.diag()  # 简化的对角Hessian
        
        # 逐输出通道量化
        for i in range(self.out_features):
            # 该通道的权重
            w = self.weight[i]
            
            # 量化误差权重
            err_weight = 1.0 / (H_diag + 1e-8)
            
            # 排序权重（按重要性）
            idx = torch.argsort(err_weight)
            
            # 逐列量化
            for j_idx, j in enumerate(idx):
                if j_idx >= 2 ** self.bits:
                    break
                    
                # 更新该列
                w[j] = self.quantize_value(w[j], self.bits)
                
                # 计算误差并传播
                error = (self.weight[i, j] - w[j]) * err_weight[j]
                self.weight[i, idx[j_idx+1:]] += error
        
        # 存储结果
        self.weight_quant = self.weight.clone()
        
    def quantize_value(self, v, bits):
        """单值量化"""
        n_levels = 2 ** bits - 1
        v_max = max(v.abs(), 1e-8)
        scale = n_levels / v_max
        v_q = torch.round(v * scale) / scale
        return v_q
```

```python
class ModularRAG(nn.Module):
    """

    包含：路由、检索、重排序、生成
    """
    def __init__(self, 
                 embed_model,

        self.generator = generator           # LLM生成器
        
    def forward(self, query, top_k=10, rerank_top=3):
        # 1. 查询理解/路由
        query_embedding = self.embed_model.encode(query)
        
        # 2. 检索阶段
        initial_results = self.retriever.search(
            query_embedding, 
            top_k=top_k
        )
        
        # 3. 重排序阶段
        reranked_results = self.reranker.rerank(
            query, 
            initial_results,
            top_n=rerank_top
        )
        
        # 4. 构建prompt
        context = self.build_context(reranked_results)
        prompt = f"基于以下资料回答问题：\n\n{context}\n\n问题：{query}\n回答："
        
        # 5. 生成答案
        response = self.generator.generate(prompt)
        
        return response
    
    def build_context(self, results):
        """构建检索上下文"""
        contexts = []
        for i, doc in enumerate(results):
            contexts.append(f"[{i+1}] {doc.page_content}\n来源: {doc.metadata.get('source', '未知')}")
        return "\n\n".join(contexts)
```

### 5.3 混合检索策略

```python
class HybridRetriever(nn.Module):
    """
    混合检索：向量 + 关键词 + 知识图谱
    """
    def __init__(self):
        self.dense_retriever = DenseRetriever()    # 向量检索
        self.sparse_retriever = BM25Retriever()    # 关键词检索
        self.kg_retriever = KGRetriever()         # 知识图谱检索
        
    def search(self, query, weights=[0.5, 0.3, 0.2]):
        # 并行执行三种检索
        dense_results = self.dense_retriever.search(query)
        sparse_results = self.sparse_retriever.search(query)
        kg_results = self.kg_retriever.search(query)
        
        # 融合结果（倒数排名融合）
        fused_scores = self.rrf_fusion(
            dense_results, 
            sparse_results, 
            kg_results,
            weights
        )
        
        return sorted(fused_results, key=lambda x: x.score, reverse=True)
    
    def rrf_fusion(self, *result_lists, weights):
        """倒数排名融合"""
        k = 60  # 调和参数
        all_docs = {}
        
        for results, weight in zip(result_lists, weights):
            for rank, doc in enumerate(results):
                doc_id = doc.id
                if doc_id not in all_docs:
                    all_docs[doc_id] = {'doc': doc, 'score': 0}
                # RRF公式: 1/(k+rank)
                all_docs[doc_id]['score'] += weight * (1 / (k + rank + 1))
        
        return [v['doc'] for v in all_docs.values()]
```

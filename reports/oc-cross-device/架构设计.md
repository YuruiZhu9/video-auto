#### 2.1.3 Trigger - 触发器
```python
from abc import ABC, abstractmethod

class Trigger(ABC):
    """触发器基类"""
    
    def __init__(self, task: Task):
        self.task = task
    
    @abstractmethod
    def check_condition(self) -> bool:
        """检查触发条件"""
        pass
    
    def execute(self) -> dict:
        """执行触发"""
        if self.check_condition():
            return self.task.execute()
        return {"status": "skipped", "reason": "condition_not_met"}

class HTTPTrigger(Trigger):
    """HTTP 触发器 - 通过 HTTP 请求触发"""
    
    def __init__(self, task: Task, method: str = "POST", path: str = None):
        super().__init__(task)
        self.method = method.upper()
        self.path = path or f"/api/v1/tasks/{task.id}"

class CronTrigger(Trigger):
    """定时触发器 - Cron 表达式"""
    
    def __init__(self, task: Task, cron_expr: str):
        super().__init__(task)
        self.cron_expr = cron_expr
    
    def check_condition(self) -> bool:
        """检查是否到达执行时间"""
        # 使用 croniter 库解析 Cron 表达式
        return croniter.is_now(self.cron_expr)

class WebhookTrigger(Trigger):
    """Webhook 触发器 - 接收外部回调"""
    
    def __init__(self, task: Task, secret: str = None):
        super().__init__(task)
        self.secret = secret
    
    def verify_signature(self, payload: str, signature: str) -> bool:
        """验证 Webhook 签名"""
        # 实现 HMAC 签名验证
```

#### 2.1.4 Auth - 认证模块
```python
import hashlib
import hmac
import secrets
from typing import Optional

class Auth:
    """认证模块"""
    
    def __init__(self, api_keys: dict):
        self.api_keys = api_keys  # {key: level} level: read/execute/admin
    
    def verify_api_key(self, api_key: str) -> Optional[str]:
        """验证 API Key，返回权限级别"""
        return self.api_keys.get(api_key)
    
    def check_permission(self, api_key: str, required_level: str) -> bool:
        """检查权限"""
        level = self.verify_api_key(api_key)
        if not level:
            return False
        levels = {"read": 1, "execute": 2, "admin": 3}
        return levels.get(level, 0) >= levels.get(required_level, 0)
    
    def generate_api_key(self, level: str = "read") -> str:
        """生成新的 API Key"""
        key = secrets.token_urlsafe(32)
        self.api_keys[key] = level
        return key
    
    def verify_webhook_signature(self, payload: str, signature: str, secret: str) -> bool:
        """验证 Webhook 签名"""
        expected = hmac.new(
            secret.encode(), 
            payload.encode(), 
            hashlib.sha256
        ).hexdigest()
        return hmac.compare_digest(expected, signature)
```

### 2.3 HTTP 处理器

#### 2.3.1 API 路由设计
```python
from fastapi import FastAPI, HTTPException, Depends, Header
from typing import Optional

app = FastAPI(title="OpenClaw Control API")

# 依赖项：验证 API Key
async def verify_api_key(x_api_key: str = Header(...)):
    auth = get_auth_instance()
    level = auth.verify_api_key(x_api_key)
    if not level:
        raise HTTPException(status_code=401, detail="Invalid API Key")
    return level

# 任务管理 API
@app.post("/api/v1/tasks", dependencies=[Depends(verify_api_key)])
async def create_task(task_data: TaskCreate):
    """创建新任务"""
    
@app.get("/api/v1/tasks/{task_id}", dependencies=[Depends(verify_api_key)])
async def get_task(task_id: str):
    """获取任务状态"""
    
@app.delete("/api/v1/tasks/{task_id}", dependencies=[Depends(verify_api_key)])
async def cancel_task(task_id: str):
    """取消任务"""

# Webhook 入口
@app.post("/api/v1/webhook")
async def webhook_handler(payload: WebhookPayload):
    """处理 Webhook 回调"""

# 系统状态
@app.get("/api/v1/status", dependencies=[Depends(verify_api_key)])
async def get_status():
    """获取系统状态"""

# 任务模板
@app.get("/api/v1/templates", dependencies=[Depends(verify_api_key)])
async def list_templates():
    """获取任务模板列表"""
```

## 3. 数据流设计

### 3.1 任务执行流程

```
1. 用户触发
   │
   ▼
2. API 接收请求
   │  • 验证 API Key
   │  • 解析请求参数
   │  • 创建任务记录
   │
   ▼
3. 任务入队列
   │  • 检查任务依赖
   │  • 分配执行优先级
   │  • 记录开始时间
   │
   ▼
4. 执行任务
   │  • 调用 OpenClaw API
   │  • 记录执行日志
   │  • 处理超时/异常
   │
   ▼
5. 结果处理
   │  • 更新任务状态
   │  • 渲染消息模板
   │  • 触发通知
   │
   ▼
6. 通知用户
   • 钉钉/Telegram 推送
   • Webhook 回调
   • 邮件通知（可选）
```

## 6. 技术选型

| 组件 | 选型 | 理由 |
|------|------|------|
| Web 框架 | FastAPI | 高性能、自动生成 API 文档 |
| 任务队列 | Python asyncio | 轻量级、无额外依赖 |
| 消息队列 | Redis (可选) | 高可用部署时需要 |
| 定时任务 | APScheduler | 功能完善、易于集成 |
| 配置管理 | Pydantic | 类型安全、自动验证 |
| 消息推送 | httpx | 异步 HTTP 客户端 |

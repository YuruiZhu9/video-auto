```bash

  libgstreamer-plugins-base1.0-dev gstreamer1.0-plugins-* \

./bin/vp_example_video_file \
  --input ../data/test.mp4 \
  --algo object_detect \
  --output result.mp4
```

### Python 管道式解析（自定义实现）

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List
import cv2

@dataclass
class VideoFrame:
    index: int
    timestamp: float
    image: any
    metadata: dict

@dataclass
class ParsedResult:
    frame: VideoFrame
    detected_objects: List[str]
    transcribed_text: str
    summary: str

class VideoParserPipeline:
    """自定义视频解析管道（可扩展节点）"""
    
    def __init__(self):
        self.nodes = []
    
    def add_node(self, node: 'PipelineNode'):
        """添加处理节点"""
        self.nodes.append(node)
        return self
    
    def process(self, video_path: str) -> List[ParsedResult]:
        cap = cv2.VideoCapture(video_path)
        fps = cap.get(cv2.CAP_PROP_FPS)
        results = []
        frame_idx = 0
        
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret: break
            
            video_frame = VideoFrame(
                index=frame_idx,
                timestamp=frame_idx / fps,
                image=frame,
                metadata={"width": frame.shape[1], "height": frame.shape[0]}
            )
            
            current_result = video_frame
            for node in self.nodes:
                current_result = node.process(current_result)
            
            results.append(current_result)
            frame_idx += 1
        
        cap.release()
        return results
```

## 避坑指南

### ⚠️ 问题1：VideoPipe 编译依赖复杂
**解决方案**：推荐使用 Docker 镜像运行，Ubuntu 20.04+ 系统最佳。详细文档：https://github.com/sherlockchou86/VideoPipe

### ⚠️ 问题2：GStreamer 学习曲线陡峭
**解决方案**：使用 Python 绑定（gstreamer-python）降低门槛，从简单 Pipeline 起步

### ⚠️ 问题3：性能优化
**解决方案**：批处理多帧后再送入推理引擎（batch inference），合理分配GPU/CPU节点

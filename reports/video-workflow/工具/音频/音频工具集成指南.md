#### 语音克隆流程

```python
import elevenlabs
from elevenlabs import Voice, VoiceSettings

# === 方法1: 上传音频创建克隆 ===
voice = elevenlabs.voices.add(
    name="my-voice-01",
    files=["reference.wav"],  # 5-30秒，清晰人声，无背景音乐
    description="我的克隆声音"
)

voice_id = voice.voice_id

# === 方法2: 即时语音（无需上传）===
# 使用官方声音，通过参数调优达到自然效果
voice = Voice(
    voice_id="EXAVITQu4vr4xnSDxMaL",  # Bella（女声）
    settings=VoiceSettings(
        stability=0.5,
        similarity_boost=0.75,
        style=0.0,
        use_speaker_boost=True
    )
)

# === 生成配音 ===
audio = elevenlabs.generate(
    text="今天告诉你，AI能帮你完成80%的职场任务，你信吗？",
    voice=voice,
    model="eleven_v3",
    language="zh"  # 中文
)
elevenlabs.save(audio, "output.mp3")
```

#### 批量生成脚本

```python
def batch_generate(script_file, voice_id, output_dir):
    """批量生成配音"""
    import os
    os.makedirs(output_dir, exist_ok=True)
    
    with open(script_file, "r", encoding="utf-8") as f:
        segments = f.read().strip().split("\n\n")
    
    for i, text in enumerate(segments):
        audio = elevenlabs.generate(
            text=text.strip(),
            voice=Voice(voice_id=voice_id),
            model="eleven_v3"
        )
        elevenlabs.save(audio, f"{output_dir}/seg_{i:02d}.mp3")
        print(f"✅ 片段 {i+1}/{len(segments)} 完成")

batch_generate("script.txt", "your-voice-id", "audio/segments")
```

## 二、背景音乐生成

### 2.1 Mureka V8（昆仑万维，全球第一）

**官网**：mureka.com  
**亮点**：Artificial Analysis双榜冠军，中文韵律最优  
**免费额度**：注册送积分，API按次计费

```python
import requests

def generate_bgm(prompt, duration=60):
    response = requests.post(
        "https://api.mureka.com/v1/generate",
        headers={"Authorization": f"Bearer {MUREKA_API_KEY}"},
        json={
            "model": "mureka-v8",
            "prompt": prompt,
            "duration": duration,
            "has_vocals": False,  # 纯音乐
            "genre": "ambient-corporate",
            "mood": "professional"
        }
    )
    return response.json()["data"]["audio_url"]

# 生成示例
bgm_url = generate_bgm(
    prompt="upbeat technology background music, inspiring, 
            no vocals, 60 seconds",
    duration=60
)
```

## 五、字幕生成（语音识别）

### MAI-Transcribe-1（微软，2026年新品）

```python
import azure.cognitiveservices.speech as speechsdk

# 语音转字幕（SRT格式）
speech_config = speech_sdk.SpeechConfig(
    subscription="MAI_KEY",
    region="eastus"
)
speech_config.speech_recognition_language = "zh-CN"

recognizer = speech_sdk.SpeechRecognizer(speech_config)

# 输出SRT格式
all_results = []
result = recognizer.recognize_once()
while result.text:
    all_results.append(result)
    result = recognizer.recognize_once()

def to_srt(results):
    srt = ""
    for i, r in enumerate(results, 1):
        start = format_timestamp(r.offset, r.offset + r.duration)
        srt += f"{i}\n{start}\n{r.text}\n\n"
    return srt

with open("subtitles.srt", "w", encoding="utf-8") as f:
    f.write(to_srt(all_results))
```

## 六、音频混音脚本（FFmpeg）

```bash
#!/bin/bash
# audio_mix.sh - 音频混音脚本

# 参数配置
VOICEOVER="audio/voiceover.mp3"
BGM="audio/bgm.mp3"
SFX="audio/sfx.mp3"
OUTPUT="audio/final_mix.mp3"

# 混音：语音(100%) + BGM(25%) + 音效(30%)
ffmpeg -i "$VOICEOVER" -i "$BGM" -i "$SFX" \
  -filter_complex "
    [1:a]volume=0.25[bgm];
    [2:a]volume=0.3[sfx];
    [0:a][bgm][sfx]amix=inputs=3:duration=first:normalize=0
  " \
  -codec:a libmp3lame -q:a 2 \
  "$OUTPUT"

echo "✅ 混音完成: $OUTPUT"
```

### 2.2 API接入（Azure AI Speech）

```python
import requests
import os

# Azure AI Speech - 文本转语音
def text_to_speech(text, voice_name="zh-CN-XiaoxiaoNeural", output_file="output.mp3"):
    """微软Azure TTS"""
    api_key = os.getenv("AZURE_SPEECH_KEY")
    region = "eastus"
    
    url = f"https://{region}.tts.speech.microsoft.com/cognitiveservices/v1"
    
    headers = {
        "Ocp-Apim-Subscription-Key": api_key,
        "Content-Type": "application/ssml+xml",
        "X-Microsoft-OutputFormat": "audio-16khz-128kbitrate-mono-mp3"
    }
    
    ssml = f"""
    <speak version='1.0' xmlns='http://www.w3.org/2001/10/synthesis' 
           xml:lang='zh-CN'>
        <voice name='{voice_name}'>
            {text}
        </voice>
    </speak>
    """
    
    response = requests.post(url, headers=headers, data=ssml.encode("utf-8"))
    
    if response.status_code == 200:
        with open(output_file, "wb") as f:
            f.write(response.content)
        print(f"✅ 音频已保存: {output_file}")
    else:
        print(f"❌ 错误: {response.status_code}")
        print(response.text)

# 中文声音推荐列表
zh_voices = {
```

```python
# 使用 ElevenLabs 进行语音克隆
import elevenlabs

# 设置API Key
elevenlabs.api_key = os.getenv("ELEVENLABS_API_KEY")

# 克隆声音（需要5-30分钟处理）
def clone_voice(audio_file_path, voice_name="MyClonedVoice"):
    """克隆自定义声音"""
    with open(audio_file_path, "rb") as f:
        voice = elevenlabs.clone(
            name=voice_name,
            description="目标克隆声音",
            files=[audio_file_path]
        )
    print(f"✅ 声音克隆完成，Voice ID: {voice.voice_id}")
    return voice.voice_id

# 生成克隆声音的音频
def generate_with_cloned_voice(voice_id, text, output_file="cloned.mp3"):
    """使用克隆声音生成音频"""
    audio = elevenlabs.generate(
        text=text,
        voice=voice_id,
        model="eleven_v2"
    )
    elevenlabs.save(audio, output_file)
    print(f"✅ 已生成: {output_file}")

# 使用示例
vid = clone_voice("my_voice_sample.mp3", "我的声音")
generate_with_cloned_voice(
    vid,
    "这是一段使用我自己声音的AI配音测试。",
    "test_output.mp3"
)
```

### 3.2 SRT字幕生成脚本

```python
import requests
import json
import os
from datetime import datetime

def transcribe_to_srt(audio_file, output_srt="subtitle.srt"):
    """将音频转为SRT字幕文件"""
    api_key = os.getenv("AZURE_SPEECH_KEY")
    region = "eastus"
    
    # 提交 transcription job
    url = f"https://{region}.api.cognitive.microsoft.com/speechtotext/v3.1/transcriptions"
    
    headers = {
        "Ocp-Apim-Subscription-Key": api_key,
        "Content-Type": "application/json"
    }
    
    data = {
        "contentUrls": [f"blob:{audio_file}"],  # 或本地文件路径
        "locale": "zh-CN",
        "displayRightMargin": True
    }
    
    # 提交任务（异步）
    response = requests.post(url, headers=headers, json=data)
    transcription_id = response.json()["self"]
    
    # 轮询结果
    while True:
        status_resp = requests.get(transcription_id, headers=headers)
        status = status_resp.json()["status"]
        if status == "Succeeded":
            result_url = status_resp.json()["resultsUrls"]["channel_0"]
            break
    
    # 获取结果并转为SRT
    result_resp = requests.get(result_url)
    segments = result_resp.json()["recognizedPhrases"]
    
    srt_content = []
    for i, seg in enumerate(segments, 1):
        start = seg["offsetInTicks"] // 10000  # 毫秒
        end = start + seg["durationInTicks"] // 10000
        text = seg["nBest"][0]["display"]
        
        start_str = ms_to_srt_time(start)
        end_str = ms_to_srt_time(end)
        srt_content.append(f"{i}\n{start_str} --> {end_str}\n{text}\n")
    
    with open(output_srt, "w", encoding="utf-8") as f:
        f.write("\n".join(srt_content))
    
    print(f"✅ 字幕文件已生成: {output_srt}")

def ms_to_srt_time(ms):
    """毫秒 → SRT时间格式 (00:00:00,000)"""
    hours = ms // 3600000
    minutes = (ms % 3600000) // 60000
    seconds = (ms % 60000) // 1000
    millis = ms % 1000
    return f"{hours:02d}:{minutes:02d}:{seconds:02d},{millis:03d}"
```

### 4.2 API接入

```python
import requests
import os
import time

MUREKA_API_KEY = os.getenv("MUREKA_API_KEY")

def generate_music(prompt, duration=30, output_file="bgm.mp3"):
    """生成AI背景音乐"""
    response = requests.post(
        "https://api.mureka.com/v1/music/generate",
        headers={
            "Authorization": f"Bearer {MUREKA_API_KEY}",
            "Content-Type": "application/json"
        },
        json={
            "model": "mureka-v8",
            "prompt": prompt,
            "duration": duration,  # 30秒或完整歌曲
            "genre": "ambient",
            "mood": "peaceful",
            "instrumental": True,  # True=纯音乐，False=带人声
            "language": "zh"  # 中文歌曲
        }
    )
    
    result = response.json()
    task_id = result["task_id"]
    
    # 轮询任务状态
    while True:
        status = requests.get(
            f"https://api.mureka.com/v1/music/status/{task_id}",
            headers={"Authorization": f"Bearer {MUREKA_API_KEY}"}
        ).json()
        
        if status["status"] == "completed":
            audio_url = status["audio_url"]
            # 下载音频
            audio_resp = requests.get(audio_url)
            with open(output_file, "wb") as f:
                f.write(audio_resp.content)
            print(f"✅ 音乐已生成: {output_file}")
            break
        elif status["status"] == "failed":
            print(f"❌ 生成失败: {status['error']}")
            break
        else:
            print(f"生成中... {status['progress']}%")
            time.sleep(10)

# 使用示例
generate_music(
    prompt="武侠电影配乐，笛子与古筝交织，宁静中带有紧张感，适合竹林打斗场景",
    duration=30,
    output_file="wuxia_bgm.mp3"
)
```

### 5.2 背景音乐自动混音脚本

```python
import subprocess
import os

def mix_audio(video_path, voice_path, bgm_path, output_path, 
              voice_vol=0.8, bgm_vol=0.2, fade_in=1, fade_out=3):
    """混音脚本：人声为主，背景音乐为辅，自动淡入淡出"""
    
    # Step 1: BGM淡入淡出
    subprocess.run([
        "ffmpeg", "-y", "-i", bgm_path,
        "-af", f"afade=t=in:st=0:d={fade_in},afade=t=out:st=-{fade_out}:d={fade_out}",
        "bgm_fade.mp3"
    ], capture_output=True)
    
    # Step 2: 混音
    subprocess.run([
        "ffmpeg", "-y",
        "-i", video_path,          # 纯视频流
        "-i", voice_path,           # 人声
        "-i", "bgm_fade.mp3",       # 背景音乐
        "-filter_complex",
        f"[1:a]volume={voice_vol}[voice];[2:a]volume={bgm_vol}[bgm];[voice][bgm]amix=inputs=2:duration=first[audio]",
        "-map", "0:v", "-map", "[audio]",
        "-c:v", "copy", "-c:a", "aac",
        output_path
    ], capture_output=True)
    
    print(f"✅ 混音完成: {output_path}")

# 使用示例
mix_audio(
    video_path="video_no_audio.mp4",
    voice_path="narration.mp3",
    bgm_path="bgm.mp3",
    output_path="final_video.mp4",
    voice_vol=0.85,
    bgm_vol=0.18
)
```

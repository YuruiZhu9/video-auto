### Step 2: 创建 Skill

创建文件 `/workspace/skills/voice_clone/index.js`：

```javascript
const { spawn } = require('child_process');
const path = require('path');
const fs = require('fs');

const SKILL = {
  name: 'voice_clone',
  version: '1.0.0',

  // 主入口：接收文本和参考音频
  async infer({ text, voice_file, model = 'cosyvoice', voice_seed = 42 }) {
    const outputFile = `/tmp/voice_${Date.now()}.wav`;

    if (model === 'cosyvoice') {
      await this._cosyvoice(text, voice_file, outputFile);
    } else if (model === 'gptsovits') {
      await this._gptsovits(text, voice_file, outputFile);
    } else if (model === 'chattts') {
      await this._chattts(text, outputFile, voice_seed);
    }

    return { audio_path: outputFile };
  },

  async _cosyvoice(text, voiceFile, outputFile) {
    const script = `
from cosyvoice import CosyVoice
c = CosyVoice('cosyvoice2-0.5B')
r = c.inference_sft('${text.replace(/'/g, "\\'")}', prompt_wav='${voiceFile}')
r.save('${outputFile}')
`;
    return new Promise((resolve, reject) => {
      const proc = spawn('python', ['-c', script]);
      let stderr = '';
      proc.stderr.on('data', d => stderr += d.toString());
      proc.on('close', code => code === 0 ? resolve(outputFile) : reject(new Error(stderr)));
    });
  },

  async _chattts(text, outputFile, seed) {
    const script = `
import ChatTTS
chat = ChatTTS.Chat()
chat.load(source='huggingface', compile=True)
wave = chat.generate('${text.replace(/'/g, "\\'")}', seed=${seed})
with open('${outputFile}', 'wb') as f:
    f.write(wave[0]['audio'])
`;
    return new Promise((resolve, reject) => {
      const proc = spawn('python', ['-c', script]);
      proc.on('close', code => code === 0 ? resolve(outputFile) : reject(new Error('ChatTTS failed')));
    });
  },
};

module.exports = SKILL;
```

## 方案二：API 服务模式（推荐生产环境）

### 启动 CosyVoice API 服务

```python
# cosyvoice_api.py
from fastapi import FastAPI, UploadFile, File, HTTPException
from cosyvoice import CosyVoice
import tempfile
import os

app = FastAPI()
cosyvoice = CosyVoice('cosyvoice2-0.5B')

@app.post("/clone")
async def clone_voice(
    text: str,
    file: UploadFile = File(...),
):
    # 保存上传的参考音频
    with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
        content = await file.read()
        f.write(content)
        temp_path = f.name

    try:
        result = cosyvoice.inference_sft(text=text, prompt_wav=temp_path)
        output_path = f"/tmp/clone_{os.urandom(4).hex()}.wav"
        result.save(output_path)
        return {"file": output_path, "size": os.path.getsize(output_path)}
    finally:
        os.unlink(temp_path)

@app.post("/clone_instruct")
async def clone_with_emotion(
    text: str,
    instruct: str,
    file: UploadFile = File(...),
):
    with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
        f.write(await file.read())
        temp_path = f.name

    try:
        result = cosyvoice.inference_instruct(
            text=text,
            prompt_wav=temp_path,
            instruct_text=instruct  # 情感指令，如"开心的语气"
        )
        output_path = f"/tmp/clone_{os.urandom(4).hex()}.wav"
        result.save(output_path)
        return {"file": output_path}
    finally:
        os.unlink(temp_path)
```

```bash
# 启动服务
uvicorn cosyvoice_api:app --host 0.0.0.0 --port 8080 --workers 2
```

### OpenClaw 中调用 API

```javascript
// 在 skill 中调用 API
async function cloneViaAPI(text, voiceFilePath, emotion = '') {
  const FormData = require('form-data');
  const fs = require('fs');

  const form = new FormData();
  form.append('text', text);
  form.append('file', fs.createReadStream(voiceFilePath));

  const endpoint = emotion 
    ? 'http://localhost:8080/clone_instruct'
    : 'http://localhost:8080/clone';
  
  if (emotion) form.append('instruct', emotion);

  const response = await fetch(endpoint, { method: 'POST', body: form });
  return response.json();
}
```

## 音频文件管理

```javascript
// 音频文件生命周期管理
const fs = require('fs');
const path = require('path');

const AUDIO_DIR = '/tmp/openclaw_audio/';

function cleanupOldFiles(maxAgeHours = 24) {
  const now = Date.now();
  const maxAge = maxAgeHours * 60 * 60 * 1000;
  
  if (!fs.existsSync(AUDIO_DIR)) return;
  
  fs.readdirSync(AUDIO_DIR).forEach(file => {
    const filePath = path.join(AUDIO_DIR, file);
    const stat = fs.statSync(filePath);
    if (now - stat.mtimeMs > maxAge) {
      fs.unlinkSync(filePath);
    }
  });
}

// 定时清理
setInterval(cleanupOldFiles, 60 * 60 * 1000); // 每小时
```

## 常见问题排查

| 问题 | 检查项 |
|------|--------|
| 模型下载慢/失败 | 配置 HuggingFace 镜像：`HF_ENDPOINT=https://hf-mirror.com` |
| 显存不足 | 减小模型（用 0.5B 而非 7B），或加 `--offload` |
| 参考音频格式错误 | 统一转换为 16kHz mono WAV：`ffmpeg -i input.mp3 -ar 16000 -ac 1 output.wav` |
| 生成超时 | 增加 subprocess timeout，或使用后台任务队列 |
| API 服务无响应 | 检查端口占用，确保 GPU 驱动正常 |

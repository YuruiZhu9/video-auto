## 二、统一 Skill 封装

```python
# ===== voice_clone_skill.py =====
"""
统一语音克隆接口
支持：Qwen3-TTS / F5-TTS / ChatTTS / OpenVoice
"""

import os
import uuid
from enum import Enum
from typing import Optional

class TTSEngine(Enum):
    QWEN3 = "qwen3"      # 3秒克隆，高质量中文
    F5 = "f5"            # 2秒零样本，多语言
    CHAT = "chat"        # 无需克隆，对话场景
    OPENVOICE = "openvoice"  # 快速即时克隆

class VoiceCloneSkill:
    """统一语音克隆技能"""
    
    def __init__(self, output_dir: str = "/workspace/voice_output"):
        self.output_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)
        self._engines = {}

    def _get_engine(self, engine: TTSEngine):
        """懒加载引擎"""
        if engine not in self._engines:
            if engine == TTSEngine.QWEN3:
                from f5_tts import load_model  # 复用
                self._engines[engine] = ("qwen3", None)
            elif engine == TTSEngine.F5:
                # F5-TTS 加载
                self._engines[engine] = ("f5", None)
            elif engine == TTSEngine.CHAT:
                import ChatTTS
                chat = ChatTTS.Chat()
                chat.load(compile=False)
                self._engines[engine] = ("chat", chat)
        return self._engines[engine]

    def generate(
        self,
        text: str,
        ref_audio: Optional[str] = None,
        ref_text: Optional[str] = None,
        engine: TTSEngine = TTSEngine.CHAT,
        output_path: Optional[str] = None,
        **kwargs,
    ) -> str:
        """统一生成接口"""
        
        output_path = output_path or os.path.join(
            self.output_dir, f"voice_{uuid.uuid4().hex[:8]}.wav"
        )

        if engine == TTSEngine.CHAT:
            import ChatTTS, soundfile as sf, torch
            chat = ChatTTS.Chat()
            chat.load(compile=False)
            seed = kwargs.get("seed", 42)
            speed = kwargs.get("speed", 1.0)
            torch.manual_seed(seed)
            marked = f"[speed_{speed}] {text}"
            wavs = chat.infer([marked])
            sf.write(output_path, wavs[0], 24000)

        elif engine in (TTSEngine.QWEN3, TTSEngine.F5):
            # 需要参考音频
            if not ref_audio:
                raise ValueError(f"{engine.value} 引擎需要参考音频路径 (ref_audio)")
            import soundfile as sf

            pass

        return output_path
```

## 四、工作流示例

### 场景 A：用户发送文案，直接生成对话语音（ChatTTS）

```
用户："帮我把这段话转成语音：今天给大家介绍一个新功能"
Agent：
  1. 判断：不需要克隆，用 ChatTTS
  2. 调用 voice_skill.generate(text="...", engine=CHAT)
  3. 获得 output.wav 路径
  4. 上传/发送音频文件
```

### 场景 B：用户发送文案+声音样本，要求克隆（Qwen3-TTS）

```
用户：[语音消息] + "克隆这个声音，朗读以下内容：[文本]"
Agent：
  1. 下载语音消息作为参考音频
  2. 调用 voice_skill.generate(
       text=用户文本,
       ref_audio=下载的音频,
       engine=QWEN3
     )
  3. 获得克隆音频路径
  4. 发送音频
```

### 场景 C：批量有声内容生成（定时任务）

```
定时触发：
1. 读取当天要播报的内容
2. 调用 ChatTTS 生成多个段落
3. 用 ffmpeg 合并成一个音频
4. 发送/发布
```

## 五、音频后处理（ffmpeg）

```bash
# 格式转换
ffmpeg -i input.wav -acodec libmp3lame -b:a 128k output.mp3

# 音量标准化（防止过小/过大）
ffmpeg -i input.wav -af loudnorm=I=-16:TP=-1.5:LRA=11 output_normalized.wav

# 合并多段音频
ffmpeg -i "concat:part1.wav|part2.wav|part3.wav" -acodec copy merged.wav

# 改变采样率
ffmpeg -i input.wav -ar 16000 output_16k.wav

# 添加淡入淡出
ffmpeg -i input.wav -af "afade=t=in:st=0:d=0.5,afade=t=out:st=5:d=0.5" output_fade.wav

# 压缩为最终发布格式
ffmpeg -i input.wav -b:a 192k -ar 44100 final.mp3
```

## 七、完整自动化脚本

```python
#!/usr/bin/env python3
"""
voice_workflow.py - 完整语音克隆自动化流程
"""

import os
import sys

def main():
    # 1. 解析参数
    if len(sys.argv) < 3:
        print("用法: python voice_workflow.py <文案> <参考音频> [引擎]")
        print("引擎: qwen3 (默认), f5, chat")
        sys.exit(1)
    
    target_text = sys.argv[1]
    ref_audio = sys.argv[2]
    engine = sys.argv[3] if len(sys.argv) > 3 else "qwen3"
    
    # 2. 导入技能
    from voice_clone_skill import voice_skill, TTSEngine
    
    engine_map = {
        "qwen3": TTSEngine.QWEN3,
        "f5": TTSEngine.F5,
        "chat": TTSEngine.CHAT,
    }
    
    # 3. 生成语音
    output_path = voice_skill.generate(
        text=target_text,
        ref_audio=ref_audio,
        engine=engine_map[engine],
    )
    
    print(f"✅ 语音已生成: {output_path}")
    
    # 4. 音频后处理
    os.system(f"ffmpeg -i {output_path} -af loudnorm=I=-16 output_norm.wav -y")
    
    return output_path

if __name__ == "__main__":
    main()
```
